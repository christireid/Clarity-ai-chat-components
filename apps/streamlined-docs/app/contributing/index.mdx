## Getting Started

### Prerequisites

Before you begin, ensure you have the following installed:

- **Node.js**: 20.0.0 or higher
- **pnpm**: 10.0.0 or higher (required, enforced by `preinstall` script)
- **Git**: Latest stable version
- **A code editor**: We recommend VS Code with TypeScript and ESLint extensions

### Quick Setup

Run our automated setup script to get started quickly:

```bash
# Clone the repository
git clone https://github.com/christireid/Clarity-ai-chat-components.git
cd Clarity-ai-chat-components

# Run the setup script
./scripts/setup-dev-environment.sh
```

The setup script will:
1. Verify Node.js and pnpm versions
2. Install dependencies
3. Build all packages
4. Run type checking
5. Run tests to ensure everything works
6. Initialize git hooks

## Code Style Guidelines

### TypeScript Guidelines

#### 1. Use Strict TypeScript

We enforce strict TypeScript with the following settings:

```json
{
  "strict": true,
  "noImplicitAny": true,
  "strictNullChecks": true,
  "strictFunctionTypes": true,
  "noUnusedLocals": true,
  "noUnusedParameters": true
}
```

#### 2. Type Over Any

```tsx
// ✅ Good: Properly typed
interface ChatMessage {
  id: string
  content: string
  role: 'user' | 'assistant'
}

function sendMessage(message: ChatMessage): Promise<void> {
  // Implementation
}

// ❌ Bad: Using any
function sendMessage(message: any): any {
  // Implementation
}
```

#### 3. Use Discriminated Unions

```tsx
// ✅ Good: Discriminated union
type Message =
  | { type: 'text'; content: string }
  | { type: 'image'; imageUrl: string; caption?: string }
  | { type: 'file'; fileName: string; fileSize: number }

// ❌ Bad: Optional properties without discrimination
interface Message {
  type: string
  content?: string
  imageUrl?: string
  fileName?: string
}
```

### React Component Guidelines

#### 1. Component Structure

All component files follow this structure:

```tsx
// 1. Imports
import React, { useState, useEffect } from 'react'
import type { ComponentProps } from './types'
import { cn } from '@clarity-chat/primitives'

// 2. Type definitions
interface MyComponentProps {
  title: string
  onAction?: () => void
  className?: string
}

// 3. Component implementation
export function MyComponent({ title, onAction, className }: MyComponentProps) {
  // Hooks (state, refs, context, custom hooks)
  const [isActive, setIsActive] = useState(false)

  // Effects
  useEffect(() => {
    // Effect logic
  }, [])

  // Event handlers
  const handleClick = () => {
    setIsActive(true)
    onAction?.()
  }

  // Render
  return (
    &lt;div className={cn('base-classes', className)}&gt;
      &lt;h2&gt;{title}&lt;/h2&gt;
      &lt;button onClick={handleClick}&gt;Click me&lt;/button&gt;
    &lt;/div&gt;
  )
}

// 4. Display name
MyComponent.displayName = 'MyComponent'
```

#### 2. Naming Conventions

- **Components**: PascalCase - `ChatWindow.tsx`, `MessageList.tsx`
- **Hooks**: camelCase with `use` prefix - `useClarityChat.ts`
- **Utilities**: camelCase - `formatMessage.ts`
- **Types**: PascalCase - `Message`, `ChatConfig`
- **Constants**: UPPER_SNAKE_CASE - `MAX_MESSAGE_LENGTH`

#### 3. Props Guidelines

```tsx
// ✅ Good: Destructured props with defaults
interface ButtonProps {
  variant?: 'primary' | 'secondary'
  size?: 'sm' | 'md' | 'lg'
  disabled?: boolean
  children: React.ReactNode
}

export function Button({
  variant = 'primary',
  size = 'md',
  disabled = false,
  children,
}: ButtonProps) {
  // Implementation
}

// ❌ Bad: Props object without destructuring
export function Button(props: ButtonProps) {
  return &lt;button disabled={props.disabled}&gt;{props.children}&lt;/button&gt;
}
```

#### 4. Memoization

```tsx
// ✅ Good: Memoized component
export const MessageList = React.memo(function MessageList({ messages }: Props) {
  return (
    &lt;div&gt;
      {messages.map((msg) => (
        &lt;Message key={msg.id} message={msg} /&gt;
      ))}
    &lt;/div&gt;
  )
})

// ✅ Good: Memoized callback
const handleDelete = useCallback((id: string) => {
  deleteMessage(id)
}, [deleteMessage])

// ✅ Good: Memoized value
const sortedMessages = useMemo(() => {
  return messages.sort((a, b) => a.timestamp - b.timestamp)
}, [messages])
```

### ESLint Configuration

Our ESLint configuration enforces:

- **Security rules**: No `eval()`, no implied eval, no script URLs
- **React Hooks rules**: Correct dependency arrays (error level)
- **Accessibility rules**: ARIA attributes, alt text, semantic HTML
- **Code complexity**: Max 15 cyclomatic complexity, max 100 lines per function
- **Animation rules**: Required reduced motion support
- **React 19 patterns**: Prefer ref-as-prop over forwardRef

### Prettier Configuration

```json
{
  "semi": false,
  "singleQuote": true,
  "tabWidth": 2,
  "trailingComma": "es5",
  "printWidth": 80,
  "arrowParens": "always"
}
```

Code is auto-formatted on commit via lint-staged.

### Accessibility Requirements

All components must meet **WCAG 2.1 AA** standards (we aim for AAA):

```tsx
// ✅ Good: Accessible component
&lt;button
  onClick={handleClick}
  aria-label="Delete message"
  aria-disabled={isLoading}
&gt;
  &lt;TrashIcon aria-hidden="true" /&gt;
&lt;/button&gt;

// ✅ Good: Keyboard navigation
&lt;div
  role="button"
  tabIndex={0}
  onClick={handleClick}
  onKeyDown={(e) =&gt; {
    if (e.key === 'Enter' || e.key === ' ') {
      handleClick()
    }
  }}
&gt;
  Click me
&lt;/div&gt;

// ✅ Good: Focus indicators
&lt;button className="focus-visible:ring-2 focus-visible:ring-blue-500"&gt;
  Click me
&lt;/button&gt;
```

## Testing Requirements

### Unit Tests

Write unit tests for:
- All utility functions
- All custom hooks
- Component logic (non-UI)

**Coverage requirement**: 85%+

```tsx
// Example: Hook test
import { renderHook, act } from '@testing-library/react'
import { useClarityChat } from '@/hooks/use-clarity-chat'

describe('useClarityChat', () => {
  it('initializes with empty messages', () => {
    const { result } = renderHook(() =&gt;
      useClarityChat({ api: '/api/chat' })
    )

    expect(result.current.messages).toEqual([])
    expect(result.current.isLoading).toBe(false)
  })

  it('appends messages correctly', async () => {
    const { result } = renderHook(() =&gt;
      useClarityChat({ api: '/api/chat' })
    )

    await act(async () => {
      await result.current.append({
        role: 'user',
        content: 'Hello'
      })
    })

    expect(result.current.messages).toHaveLength(1)
    expect(result.current.messages[0].content).toBe('Hello')
  })
})
```

### Component Tests

Write component tests for:
- Rendering behavior
- User interactions
- State changes
- Accessibility

**Coverage requirement**: 80%+

```tsx
// Example: Component test
import { render, screen, fireEvent } from '@testing-library/react'
import { ChatMessage } from '@/components/message/ChatMessage'

describe('ChatMessage', () => {
  const mockMessage = {
    id: '1',
    role: 'user' as const,
    content: 'Hello, world!',
  }

  it('renders message content', () => {
    render(&lt;ChatMessage message={mockMessage} /&gt;)
    expect(screen.getByText('Hello, world!')).toBeInTheDocument()
  })

  it('calls onEdit when edit button is clicked', () => {
    const onEdit = vi.fn()
    render(&lt;ChatMessage message={mockMessage} onEdit={onEdit} /&gt;)

    const editButton = screen.getByRole('button', { name: /edit/i })
    fireEvent.click(editButton)

    expect(onEdit).toHaveBeenCalledWith('1', 'Hello, world!')
  })

  it('has no accessibility violations', async () => {
    const { container } = render(&lt;ChatMessage message={mockMessage} /&gt;)
    const results = await axe(container)
    expect(results).toHaveNoViolations()
  })
})
```

### Integration Tests

Write integration tests for:
- Multi-component workflows
- API interactions
- Complex state management
- Cross-component communication

```tsx
// Example: Integration test
describe('Chat Workflow', () => {
  it('sends and receives messages end-to-end', async () => {
    render(&lt;ChatApp /&gt;)

    // Type and send message
    const input = screen.getByPlaceholderText('Type a message...')
    fireEvent.change(input, { target: { value: 'Hello' } })
    fireEvent.click(screen.getByRole('button', { name: /send/i }))

    // Wait for user message to appear
    await screen.findByText('Hello')

    // Wait for AI response
    await screen.findByText(/thinking/i)
    const response = await screen.findByText(/hello/i, {}, { timeout: 5000 })

    expect(response).toBeInTheDocument()
  })
})
```

### End-to-End Tests

Write E2E tests for:
- Critical user journeys
- Cross-browser compatibility
- Real API integration
- Performance benchmarks

```typescript
// Example: E2E test
import { test, expect } from '@playwright/test'

test('complete chat conversation', async ({ page }) => {
  await page.goto('http://localhost:3000')

  // Wait for chat to load
  await expect(page.getByRole('main')).toBeVisible()

  // Send message
  await page.fill('[placeholder="Type a message..."]', 'Hello')
  await page.click('button[aria-label="Send message"]')

  // Verify message appears
  await expect(page.getByText('Hello')).toBeVisible()

  // Wait for response
  await expect(page.getByText(/thinking/i)).toBeVisible()
  await expect(page.getByText(/hello/i)).toBeVisible({ timeout: 10000 })
})
```

## Pull Request Process

### 1. Create a Feature Branch

```bash
# Update main branch
git checkout main
git pull origin main

# Create feature branch
git checkout -b feature/your-feature-name

# Or for bug fixes
git checkout -b fix/bug-description
```

Branch naming conventions:
- `feature/` - New features
- `fix/` - Bug fixes
- `docs/` - Documentation updates
- `refactor/` - Code refactoring
- `test/` - Test additions or fixes
- `chore/` - Maintenance tasks

### 2. Make Your Changes

```bash
# Make changes to code
# Add tests for your changes
# Update documentation if needed

# Run checks locally
pnpm check                    # Type check, lint, and test
```

### 3. Commit Your Changes

We use conventional commits:

```bash
# Stage changes
git add .

# Commit with conventional format
git commit -m "feat: add voice input component"
git commit -m "fix: resolve memory leak in chat hook"
git commit -m "docs: update contributing guide"
git commit -m "test: add tests for message renderer"
git commit -m "refactor: simplify token counter logic"
```

**Commit types:**
- `feat:` - New feature
- `fix:` - Bug fix
- `docs:` - Documentation
- `test:` - Tests
- `refactor:` - Code refactoring
- `perf:` - Performance improvement
- `style:` - Code style (formatting, etc.)
- `chore:` - Maintenance
- `ci:` - CI/CD changes

**Commit scope (optional):**
```bash
git commit -m "feat(chat): add message reactions"
git commit -m "fix(memory): prevent memory overflow"
```

**Breaking changes:**
```bash
git commit -m "feat!: redesign chat API

BREAKING CHANGE: The chat hook now returns an object instead of an array"
```

The pre-commit hook will:
- Check for development artifacts
- Validate workflow files with actionlint
- Run lint-staged (format, lint staged files)

### 4. Push and Create PR

```bash
# Push your branch
git push origin feature/your-feature-name

# Create PR on GitHub
# Fill out the PR template completely
```

When you create a PR, fill out all sections:

## Review Checklist

### For PR Authors

Before requesting review:

- [ ] All tests pass locally (`pnpm test`)
- [ ] Type checking passes (`pnpm typecheck`)
- [ ] Linting passes (`pnpm lint`)
- [ ] Code is formatted (`pnpm format`)
- [ ] No console.log statements (except intentional logging)
- [ ] No commented-out code
- [ ] Tests added for new functionality
- [ ] Documentation updated
- [ ] Accessibility tested
- [ ] Breaking changes documented
- [ ] Changeset added (if needed for version bump)

### For Reviewers

When reviewing a PR:

#### Code Quality
- [ ] Code follows TypeScript best practices
- [ ] No use of `any` type (without justification)
- [ ] Proper error handling
- [ ] No memory leaks (event listeners cleaned up)
- [ ] Efficient algorithms and data structures

#### React Patterns
- [ ] Proper use of hooks (dependencies, cleanup)
- [ ] Components properly memoized (if needed)
- [ ] No unnecessary re-renders
- [ ] Props properly typed
- [ ] Refs used correctly

#### Testing
- [ ] Tests cover new functionality
- [ ] Tests are meaningful (not just for coverage)
- [ ] Edge cases tested
- [ ] Error cases tested
- [ ] Accessibility tested

#### Documentation
- [ ] Public APIs documented
- [ ] Complex logic explained
- [ ] Examples provided (if needed)
- [ ] Breaking changes clearly documented

#### Security
- [ ] No secrets or API keys in code
- [ ] Input validation present
- [ ] No SQL injection risks
- [ ] No XSS vulnerabilities
- [ ] CSRF protection (for API routes)

#### Accessibility
- [ ] ARIA attributes used correctly
- [ ] Keyboard navigation works
- [ ] Focus indicators visible
- [ ] Color contrast meets WCAG AA
- [ ] Screen reader friendly

#### Performance
- [ ] No unnecessary computations
- [ ] Large lists virtualized
- [ ] Images optimized
- [ ] Code splitting used (if needed)
- [ ] Animations respect reduced motion

## Release Process

We use [Changesets](https://github.com/changesets/changesets) for version management and releases.

### 1. Adding a Changeset

When you make changes that should be included in a release:

```bash
# Add a changeset
pnpm changeset

# Follow the prompts:
# 1. Select packages to version bump
# 2. Choose bump type (major, minor, patch)
# 3. Write a summary of changes
```

Example changeset:
```markdown
---
"@clarity-chat/react": minor
---

Add voice input component with browser speech recognition support
```

### 2. Version Bumping

Maintainers will run:

```bash
# Update versions based on changesets
pnpm version-packages

# This creates a PR with version bumps and updated CHANGELOGs
```

### 3. Publishing

After the version PR is merged:

```bash
# Build and publish to npm
pnpm release
```

### Release Types

**Major (x.0.0)**: Breaking changes
```bash
# Example: Redesigned API
pnpm changeset
# Select "major" for affected packages
```

**Minor (0.x.0)**: New features (backward compatible)
```bash
# Example: New component added
pnpm changeset
# Select "minor" for affected packages
```

**Patch (0.0.x)**: Bug fixes
```bash
# Example: Fixed memory leak
pnpm changeset
# Select "patch" for affected packages
```

### Pre-releases

For beta/alpha releases:

```bash
# Enter pre-release mode
pnpm changeset pre enter beta

# Add changesets as usual
pnpm changeset

# Version and publish
pnpm version-packages
pnpm release

# Exit pre-release mode
pnpm changeset pre exit
```

## Documentation Guidelines

- `README.md` - Package overviews

## Development Environment Script

Save this as `scripts/setup-dev-environment.sh`:

```bash
#!/bin/bash

set -e

echo "🚀 Clarity Chat - Development Environment Setup"
echo "================================================"
echo ""

# Color codes for output
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
NC='\033[0m' # No Color

# Function to print colored messages
print_success() {
    echo -e "${GREEN}✓${NC} $1"
}

print_error() {
    echo -e "${RED}✗${NC} $1"
}

print_info() {
    echo -e "${YELLOW}→${NC} $1"
}

# Check Node.js version
print_info "Checking Node.js version..."
NODE_VERSION=$(node -v | cut -d'v' -f2 | cut -d'.' -f1)
if [ "$NODE_VERSION" -lt 20 ]; then
    print_error "Node.js 20+ required. Current: v$(node -v)"
    echo "Install from: https://nodejs.org/"
    exit 1
fi
print_success "Node.js version: $(node -v)"

# Check pnpm installation
print_info "Checking pnpm installation..."
if ! command -v pnpm &gt; /dev/null; then
    print_error "pnpm not found. Installing..."
    npm install -g pnpm@10
fi
PNPM_VERSION=$(pnpm -v | cut -d'.' -f1)
if [ "$PNPM_VERSION" -lt 10 ]; then
    print_error "pnpm 10+ required. Current: $(pnpm -v)"
    echo "Update with: npm install -g pnpm@10"
    exit 1
fi
print_success "pnpm version: $(pnpm -v)"

# Install dependencies
print_info "Installing dependencies..."
if pnpm install; then
    print_success "Dependencies installed"
else
    print_error "Failed to install dependencies"
    exit 1
fi

# Build packages
print_info "Building packages..."
if pnpm build:packages; then
    print_success "Packages built successfully"
else
    print_error "Build failed"
    exit 1
fi

# Run type checking
print_info "Running type checking..."
if pnpm typecheck; then
    print_success "Type checking passed"
else
    print_error "Type checking failed"
    exit 1
fi

# Run linting
print_info "Running linter..."
if pnpm lint; then
    print_success "Linting passed"
else
    print_error "Linting failed"
    exit 1
fi

# Run tests
print_info "Running tests..."
if pnpm test; then
    print_success "Tests passed"
else
    print_error "Tests failed"
    exit 1
fi

# Setup git hooks
print_info "Setting up git hooks..."
if [ -d ".husky" ]; then
    pnpm prepare
    print_success "Git hooks initialized"
else
    print_info "Git hooks directory not found, skipping..."
fi

# Success
echo ""
echo "================================================"
print_success "Development environment ready!"
