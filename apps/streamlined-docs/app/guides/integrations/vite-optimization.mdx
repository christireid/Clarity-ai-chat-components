# Vite Build Optimization for Production

Complete production-ready Vite configuration with tree shaking, code splitting, bundle analysis, and performance optimizations. Achieve 60-80% bundle size reduction with proper configuration.

## Quick Reference

| Optimization | Impact | Effort |

## Complete Optimized Configuration

```typescript
// vite.config.ts - Production-optimized configuration
import { defineConfig, loadEnv, type Plugin } from 'vite'
import react from '@vitejs/plugin-react'
import { visualizer } from 'rollup-plugin-visualizer'
import { compression } from 'vite-plugin-compression2'
import { ViteImageOptimizer } from 'vite-plugin-image-optimizer'
import path from 'path'

export default defineConfig(({ mode }) => {
  const env = loadEnv(mode, process.cwd(), '')
  const isProd = mode === 'production'
  const isAnalyze = process.env.ANALYZE === 'true'

  return {
    // ============================================
    // PLUGINS
    // ============================================
    plugins: [
      // React with SWC for faster builds
      react({
        babel: {
          plugins: isProd
            ? [
                // Remove console.log in production
                ['babel-plugin-transform-remove-console', { exclude: ['error', 'warn'] }],
                // Remove PropTypes in production
                'babel-plugin-transform-react-remove-prop-types',
              ]
            : [],
        },
      }),

      // Bundle size visualization
      isAnalyze &&
        visualizer({
          open: true,
          filename: 'dist/stats.html',
          gzipSize: true,
          brotliSize: true,
          template: 'treemap', // or 'sunburst', 'network'
        }),

      // Gzip compression
      isProd &&
        compression({
          algorithm: 'gzip',
          exclude: [/\.(br)$/, /\.(gz)$/],
          threshold: 1024, // Only compress files > 1KB
          deleteOriginalAssets: false,
        }),

      // Brotli compression (better than gzip)
      isProd &&
        compression({
          algorithm: 'brotliCompress',
          exclude: [/\.(br)$/, /\.(gz)$/],
          threshold: 1024,
          deleteOriginalAssets: false,
        }),

      // Image optimization
      isProd &&
        ViteImageOptimizer({
          png: {
            quality: 80,
          },
          jpeg: {
            quality: 80,
          },
          jpg: {
            quality: 80,
          },
          webp: {
            quality: 80,
          },
        }),
    ].filter(Boolean) as Plugin[],

    // ============================================
    // RESOLVE
    // ============================================
    resolve: {
      alias: {
        '@': path.resolve(__dirname, './src'),
        '@components': path.resolve(__dirname, './src/components'),
        '@lib': path.resolve(__dirname, './src/lib'),
        '@hooks': path.resolve(__dirname, './src/hooks'),
        '@utils': path.resolve(__dirname, './src/utils'),
      },
      // Optimize extension resolution
      extensions: ['.tsx', '.ts', '.jsx', '.js', '.json'],
    },

    // ============================================
    // BUILD OPTIMIZATION
    // ============================================
    build: {
      // Output directory
      outDir: 'dist',
      assetsDir: 'assets',

      // Generate source maps only for production debugging
      sourcemap: isProd ? 'hidden' : true,

      // Target modern browsers for smaller bundles
      target: ['es2020', 'edge88', 'firefox78', 'chrome87', 'safari14'],

      // CSS code splitting
      cssCodeSplit: true,

      // Enable minification
      minify: isProd ? 'terser' : false,

      // Terser options for aggressive minification
      terserOptions: isProd
        ? {
            compress: {
              // Remove console and debugger
              drop_console: true,
              drop_debugger: true,
              // Remove dead code
              dead_code: true,
              // Optimize conditions
              conditionals: true,
              // Evaluate constant expressions
              evaluate: true,
              // Optimize booleans
              booleans: true,
              // Remove unused code
              unused: true,
              // Inline functions
              inline: 2,
              // Optimize loops
              loops: true,
              // Reduce variables
              reduce_vars: true,
              // Pure function calls
              pure_funcs: ['console.log', 'console.info', 'console.debug'],
            },
            mangle: {
              // Mangle variable names
              toplevel: true,
              safari10: true,
            },
            format: {
              // Remove comments
              comments: false,
            },
          }
        : undefined,

      // ============================================
      // ROLLUP OPTIONS - ADVANCED OPTIMIZATION
      // ============================================
      rollupOptions: {
        // External dependencies (for libraries)
        // external: ['react', 'react-dom'],

        output: {
          // ====================================
          // MANUAL CHUNKS - CODE SPLITTING
          // ====================================
          manualChunks: (id) => {
            // Vendor chunks - libraries that rarely change
            if (id.includes('node_modules')) {
              // React ecosystem
              if (id.includes('react') || id.includes('react-dom')) {
                return 'vendor-react'
              }

              // UI libraries
              if (
                id.includes('framer-motion') ||
                id.includes('lucide-react') ||
                id.includes('@radix-ui')
              ) {
                return 'vendor-ui'
              }

              // AI/ML libraries
              if (
                id.includes('@ai-sdk') ||
                id.includes('openai') ||
                id.includes('@anthropic-ai') ||
                id.includes('ai')
              ) {
                return 'vendor-ai'
              }

              // Utilities
              if (
                id.includes('lodash') ||
                id.includes('date-fns') ||
                id.includes('uuid') ||
                id.includes('clsx')
              ) {
                return 'vendor-utils'
              }

              // Markdown/syntax highlighting
              if (
                id.includes('prism') ||
                id.includes('shiki') ||
                id.includes('markdown') ||
                id.includes('remark') ||
                id.includes('rehype')
              ) {
                return 'vendor-markdown'
              }

              // Heavy libraries that should be separate
              if (id.includes('monaco-editor')) {
                return 'vendor-monaco'
              }

              if (id.includes('three')) {
                return 'vendor-three'
              }

              if (id.includes('mermaid')) {
                return 'vendor-mermaid'
              }

              // All other node_modules
              return 'vendor-other'
            }

            // Application code splitting
            // Split by route/feature
            if (id.includes('/routes/')) {
              const routeMatch = id.match(/\/routes\/([^/]+)/)
              if (routeMatch) {
                return `route-${routeMatch[1]}`
              }
            }

            // Split by feature/component group
            if (id.includes('/features/')) {
              const featureMatch = id.match(/\/features\/([^/]+)/)
              if (featureMatch) {
                return `feature-${featureMatch[1]}`
              }
            }

            // Large component libraries
            if (id.includes('/components/')) {
              // Heavy components in separate chunks
              if (id.includes('/components/chat/')) {
                return 'components-chat'
              }
              if (id.includes('/components/editor/')) {
                return 'components-editor'
              }
              if (id.includes('/components/visualizations/')) {
                return 'components-viz'
              }
            }
          },

          // ====================================
          // CHUNK FILE NAMING
          // ====================================
          chunkFileNames: (chunkInfo) => {
            // Hashed names for long-term caching
            const name = chunkInfo.name
            return isProd ? `assets/js/${name}-[hash].js` : `assets/js/${name}.js`
          },

          entryFileNames: isProd ? 'assets/js/[name]-[hash].js' : 'assets/js/[name].js',

          assetFileNames: (assetInfo) => {
            const name = assetInfo.name || ''

            // Organize assets by type
            if (name.endsWith('.css')) {
              return isProd ? 'assets/css/[name]-[hash][extname]' : 'assets/css/[name][extname]'
            }

            if (/\.(png|jpe?g|svg|gif|webp|avif)$/.test(name)) {
              return isProd
                ? 'assets/images/[name]-[hash][extname]'
                : 'assets/images/[name][extname]'
            }

            if (/\.(woff2?|eot|ttf|otf)$/.test(name)) {
              return isProd ? 'assets/fonts/[name]-[hash][extname]' : 'assets/fonts/[name][extname]'
            }

            return isProd ? 'assets/[name]-[hash][extname]' : 'assets/[name][extname]'
          },

          // ====================================
          // ADDITIONAL OPTIMIZATION
          // ====================================
          // Hoist transitive imports for better tree-shaking
          hoistTransitiveImports: true,

          // Preserve module structure for better debugging
          preserveModules: false,

          // Inline dynamic imports below threshold
          inlineDynamicImports: false,

          // Generate compact output
          compact: isProd,

          // Interop for CommonJS modules
          interop: 'auto',
        },

        // ====================================
        // TREE SHAKING CONFIGURATION
        // ====================================
        treeshake: isProd
          ? {
              // Enable all tree-shaking optimizations
              moduleSideEffects: (id) => {
                // Preserve side effects for CSS and specific modules
                if (id.includes('.css') || id.includes('polyfill')) {
                  return true
                }
                // Default: assume no side effects for better tree-shaking
                return false
              },
              propertyReadSideEffects: false,
              tryCatchDeoptimization: false,
              annotations: true, // Respect /*#__PURE__*/ annotations
              unknownGlobalSideEffects: false,
            }
          : false,
      },

      // ============================================
      // CHUNK SIZE WARNINGS
      // ============================================
      chunkSizeWarningLimit: 500, // KB
      reportCompressedSize: true,

      // ============================================
      // ASSET INLINING
      // ============================================
      // Inline assets below 4KB as base64
      assetsInlineLimit: 4096,
    },

    // ============================================
    // OPTIMIZATION
    // ============================================
    optimizeDeps: {
      // Pre-bundle these dependencies
      include: [
        'react',
        'react-dom',
        'react/jsx-runtime',
        'framer-motion',
        'lucide-react',
        'clsx',
      ],

      // Don't pre-bundle these (dynamic imports)
      exclude: ['@clarity-chat/monaco-editor', '@clarity-chat/three-viewer'],

      // ESbuild options for optimization
      esbuildOptions: {
        target: 'es2020',
        supported: {
          bigint: true,
        },
        // Remove unused imports
        treeShaking: true,
      },
    },

    // ============================================
    // CSS OPTIMIZATION
    // ============================================
    css: {
      devSourcemap: !isProd,
      modules: {
        localsConvention: 'camelCase',
        generateScopedName: isProd ? '[hash:base64:5]' : '[name]__[local]__[hash:base64:5]',
      },
      preprocessorOptions: {
        scss: {
          additionalData: `@import "@/styles/variables.scss";`,
        },
      },
    },

    // ============================================
    // SERVER (DEVELOPMENT)
    // ============================================
    server: {
      port: 3000,
      open: true,
      host: true,
      // Enable CORS for development
      cors: true,
      // HMR configuration
      hmr: {
        overlay: true,
      },
    },

    // ============================================
    // PREVIEW (PRODUCTION TEST)
    // ============================================
    preview: {
      port: 4173,
      open: true,
    },

    // ============================================
    // ENVIRONMENT VARIABLES
    // ============================================
    define: {
      __APP_VERSION__: JSON.stringify(process.env.npm_package_version),
      __BUILD_TIME__: JSON.stringify(new Date().toISOString()),
    },

    // ============================================
    // ESBuild (for TS/JS transformation)
    // ============================================
    esbuild: {
      // Drop console in production
      drop: isProd ? ['console', 'debugger'] : [],
      // Legal comments
      legalComments: 'none',
      // Target
      target: 'es2020',
      // JSX
      jsxInject: `import React from 'react'`,
    },

    // ============================================
    // WORKER OPTIMIZATION
    // ============================================
    worker: {
      format: 'es',
      plugins: [],
      rollupOptions: {
        output: {
          format: 'es',
          entryFileNames: 'assets/workers/[name]-[hash].js',
        },
      },
    },
  }
})
```

## Tree Shaking Configuration

### Package.json Setup

```json
{
  "name": "@clarity-chat/components",
  "version": "1.0.0",
  "type": "module",
  "sideEffects": [
    "*.css",
    "*.scss",
    "*.less",
    "*/polyfills.ts"
  ],
  "exports": {
    ".": {
      "import": "./dist/index.js",
      "require": "./dist/index.cjs",
      "types": "./dist/index.d.ts"
    },
    "./styles": "./dist/styles/index.css"
  }
}
```

### ESM-Only Exports

```typescript
// src/index.ts - Tree-shakeable exports
export { Button } from './components/Button'
export { Input } from './components/Input'
export type { ButtonProps, InputProps } from './types'

// ❌ Bad - Barrel exports prevent tree-shaking
export * from './components'

// ✅ Good - Explicit named exports
export { Button, Input, Select } from './components'
```

### Pure Annotations

```typescript
// Mark functions as pure for better tree-shaking
export const createConfig = /*#__PURE__*/ (options: Options) => {
  return { ...defaultOptions, ...options }
}

// Mark class constructors as pure
export class Logger {
  /*#__PURE__*/ constructor(private name: string) {}
}
```

import { lazy, Suspense } from 'react'

// Lazy load routes
const Home = lazy(() => import('./pages/Home'))

### 3. Dynamic Imports

```typescript
// Dynamic imports for conditional features
async function loadAIModel(modelType: string) {
  if (modelType === 'gpt-4') {
    const { GPT4 } = await import('./models/gpt4')
    return new GPT4()
  } else if (modelType === 'claude') {
    const { Claude } = await import('./models/claude')
    return new Claude()
  }
}

// Network-aware loading
async function loadHeavyFeature() {
  if ('connection' in navigator) {
    const connection = (navigator as any).connection
    if (connection.effectiveType === '4g' || connection.effectiveType === 'wifi') {
      return import('./features/heavy-feature')
    }
  }
  // Fallback for slow connections
  return import('./features/light-feature')
}
```

## Deployment Configuration

### GitHub Actions CI/CD

```yaml
# .github/workflows/deploy.yml
name: Build and Deploy

on:
  push:
    branches: [main]

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: '20'
          cache: 'pnpm'

      - name: Install dependencies
        run: pnpm install --frozen-lockfile

      - name: Build
        env:
          VITE_API_URL: ${{ secrets.API_URL }}
          VITE_API_KEY: ${{ secrets.API_KEY }}
          COMMIT_HASH: ${{ github.sha }}
        run: pnpm build

      - name: Size Report
        run: |
          pnpm size:report
          echo "## Bundle Size Report" >> $GITHUB_STEP_SUMMARY
          pnpm size:report --json >> size-report.json
          cat size-report.json >> $GITHUB_STEP_SUMMARY

      - name: Deploy to Vercel
        uses: vercel/action@v2
        with:
          vercel-token: ${{ secrets.VERCEL_TOKEN }}

```

## Troubleshooting

### Common Issues

#### 1. Large Vendor Chunks

```bash
# Problem: vendor-other.js is too large (>500 KB)

# Solution: Analyze and split further
pnpm build:analyze

# Check what's inside the chunk
# Split large libraries into separate chunks
manualChunks: {
  'vendor-charts': ['chart.js', 'd3'],
  'vendor-markdown': ['marked', 'prismjs'],
}
```

#### 2. Slow Build Times

```bash
# Problem: Build takes >30 seconds

# Solution 1: Enable persistent cache
export VITE_CACHE_DIR=.vite

# Solution 2: Disable source maps in dev
build: {
  sourcemap: false
}

# Solution 3: Use SWC instead of Babel
plugins: [react({ babel: false })]
```

#### 3. Tree Shaking Not Working

```bash
# Problem: Bundle includes unused code

# Solution: Check sideEffects in package.json
{
  "sideEffects": false,
  "exports": "./dist/index.js"
}

# Use named imports
import { Button } from '@clarity-chat/react' // ✅
import * as Components from '@clarity-chat/react' // ❌
```
