### Key Configuration Explanations

#### Strict Mode Foundation

```json
{
  "strict": true // Enables ALL strict type checking options
}
```

This single flag enables:
- `noImplicitAny` - No implicit 'any' types
- `strictNullChecks` - Null and undefined must be handled explicitly
- `strictFunctionTypes` - Function parameters are checked contravariantly
- `strictBindCallApply` - Strict checking of bind, call, and apply
- `strictPropertyInitialization` - Class properties must be initialized
- `noImplicitThis` - 'this' must have an explicit type
- `alwaysStrict` - Parse in strict mode and emit "use strict"

#### Enhanced Error Detection

```json
{
```

### Constrained Generic Components

```tsx
import type { GenericMessage } from '@clarity-chat/react/types'

/**
 * Generic message list component with metadata constraints
 * TMetadata must extend Record<string, unknown> for type safety
 */
interface MessageListProps<TMetadata extends Record<string, unknown>> {
  messages: GenericMessage<TMetadata>[]
  onMessageClick?: (message: GenericMessage<TMetadata>) => void
  filterBy?: keyof TMetadata
  sortBy?: keyof TMetadata
}

function MessageList<TMetadata extends Record<string, unknown>>({
  messages,
  onMessageClick,
  filterBy,
  sortBy,
}: MessageListProps<TMetadata>) {
  // Type-safe filtering
  const filtered = filterBy
    ? messages.filter((msg) => msg.metadata?.[filterBy] !== undefined)
    : messages

  // Type-safe sorting
  const sorted = sortBy
    ? [...filtered].sort((a, b) => {
        const aVal = a.metadata?.[sortBy]
        const bVal = b.metadata?.[sortBy]
        return String(aVal).localeCompare(String(bVal))
      })
    : filtered

  return (
    <div>
      {sorted.map((message) => (
        <div
          key={message.id}
          onClick={() => onMessageClick?.(message)}
          role="button"
          tabIndex={0}
        >
          {message.content}
        </div>
      ))}
    </div>
  )
}

// Usage with custom metadata
interface ChatMetadata {
  userId: string
  sentiment: 'positive' | 'negative' | 'neutral'
  priority: number
}

const MyChat = () => {
  const messages: GenericMessage<ChatMetadata>[] = [
    /* ... */
  ]

  return (
    <MessageList
      messages={messages}
      filterBy="sentiment" // Type-safe: must be keyof ChatMetadata
      sortBy="priority" // Type-safe: must be keyof ChatMetadata
      onMessageClick={(msg) => {
        // msg.metadata is typed as ChatMetadata | undefined

      }}
    />
  )
}
```

### Multiple Generic Constraints

```tsx
import type { StrictChatEvent, EventType } from '@clarity-chat/react/types'

/**
 * Event handler with multiple generic constraints
 * TEvent must extend EventType
 * TPayload is inferred from the event type
 */
interface EventHandler<
  TEvent extends EventType,
  TPayload = Extract<StrictChatEvent, { type: TEvent }>
> {
  type: TEvent
  handler: (payload: TPayload) => void | Promise<void>
  once?: boolean
}

/**
 * Type-safe event emitter with generic constraints
 */
class TypeSafeEventEmitter {
  private handlers = new Map<EventType, Set<(payload: any) => void>>()

  /**
   * Register an event handler with full type safety
   */
  on<TEvent extends EventType>(
    type: TEvent,
    handler: (payload: Extract<StrictChatEvent, { type: TEvent }>) => void
  ): () => void {
    if (!this.handlers.has(type)) {
      this.handlers.set(type, new Set())
    }

    this.handlers.get(type)!.add(handler)

    // Return unsubscribe function
    return () => {
      this.handlers.get(type)?.delete(handler)
    }
  }

  /**
   * Emit an event with type-safe payload
   */
  emit<TEvent extends EventType>(
    type: TEvent,
    payload: Extract<StrictChatEvent, { type: TEvent }>
  ): void {
    this.handlers.get(type)?.forEach((handler) => {
      handler(payload)
    })
  }
}

// Usage
const emitter = new TypeSafeEventEmitter()

// Type-safe registration
const unsubscribe = emitter.on('message:sent', (event) => {
  // event is typed as MessageSentEvent
  console.log(event.content, event.messageId)
})

// Type-safe emission
emitter.emit('message:sent', {
  type: 'message:sent' as const,
  timestamp: new Date(),
  messageId: '123',
  content: 'Hello',
  role: 'user',
})

// TypeScript error - wrong payload type
// emitter.emit('message:sent', { type: 'message:sent', invalid: true })
```

## Type Narrowing Techniques

### Discriminated Unions

```tsx
import type { FinishReason } from '@clarity-chat/react/types'

/**
 * Discriminated union for chat responses
 */
type ChatResponse =
  | {
      status: 'success'
      message: string
      finishReason: FinishReason
      usage: { promptTokens: number; completionTokens: number }
    }
  | {
      status: 'error'
      error: Error
      retryable: boolean
    }
  | {
      status: 'pending'
      progress: number
    }

/**
 * Type-safe response handler using discriminated unions
 */
function handleResponse(response: ChatResponse) {
  // TypeScript narrows the type based on status
  switch (response.status) {
    case 'success':
      // response is typed as the success variant
      console.log('Message:', response.message)
      console.log('Tokens:', response.usage.promptTokens)
      break

    case 'error':
      // response is typed as the error variant
      if (response.retryable) {
        console.error('Retryable error:', response.error)
      } else {
        console.error('Fatal error:', response.error)
      }
      break

    case 'pending':
      // response is typed as the pending variant
      console.log('Progress:', response.progress)
      break

    default:
      // Exhaustiveness check - ensures all cases are handled
      const _exhaustive: never = response
      throw new Error(`Unhandled response type: ${_exhaustive}`)
  }
}
```

### Type Guards

```tsx
import type { StrictChatEvent } from '@clarity-chat/react/types'

/**
 * Type guard for message sent events
 */
function isMessageSentEvent(event: StrictChatEvent): event is Extract<
  StrictChatEvent,
  { type: 'message:sent' }
> {
  return event.type === 'message:sent'
}

/**
 * Type guard for tool events
 */
function isToolEvent(
  event: StrictChatEvent
): event is Extract<
  StrictChatEvent,
  { type: 'tool:started' | 'tool:completed' | 'tool:error' }
> {
  return (
    event.type === 'tool:started' ||
    event.type === 'tool:completed' ||
    event.type === 'tool:error'
  )
}

/**
 * Generic type guard factory
 */
function createEventTypeGuard<TType extends StrictChatEvent['type']>(
  type: TType
) {
  return (event: StrictChatEvent): event is Extract<StrictChatEvent, { type: TType }> => {
    return event.type === type
  }
}

// Usage
const isStreamStarted = createEventTypeGuard('stream:started')
const isStreamChunk = createEventTypeGuard('stream:chunk')

function processEvent(event: StrictChatEvent) {
  if (isMessageSentEvent(event)) {
    // event is now MessageSentEvent
    console.log(event.content, event.messageId)
  }

  if (isStreamStarted(event)) {
    // event is now StreamStartedEvent
    console.log('Stream started for message:', event.messageId)
  }
}
```

### Assertion Functions

```tsx
/**
 * Assertion function for non-null values
 */
function assertNonNull<T>(value: T | null | undefined, message?: string): asserts value is T {
  if (value === null || value === undefined) {
    throw new Error(message ?? 'Value is null or undefined')
  }
}

/**
 * Assertion function for specific types
 */
function assertIsString(value: unknown, message?: string): asserts value is string {
  if (typeof value !== 'string') {
    throw new Error(message ?? `Expected string, got ${typeof value}`)
  }
}

/**
 * Assertion function for adapter validation
 */
import { validateAdapter, isFormalizedModelAdapter } from '@clarity-chat/react/types'

function processAdapter(adapter: unknown) {
  // Throws if adapter is invalid, otherwise narrows type
  validateAdapter(adapter)

  // adapter is now typed as FormalizedModelAdapter
  console.log('Provider:', adapter.provider)
  console.log('Capabilities:', adapter.capabilities)
}

// Usage
function processMessage(message: unknown) {
  assertNonNull(message, 'Message cannot be null')
  // message is now typed as unknown (but guaranteed non-null)

  const content = (message as any).content
  assertIsString(content, 'Message content must be a string')
  // content is now typed as string

  console.log(content.toUpperCase()) // ✓ Safe
}
```

## Avoiding 'any' and Type Assertions

### Use 'unknown' Instead of 'any'

```tsx
/**
 * ❌ Bad: Using 'any' loses all type safety
 */
function parseResponseBad(response: any) {
  // No type checking - dangerous!
  return response.data.messages
}

/**
 * ✓ Good: Using 'unknown' requires explicit type checking
 */
function parseResponseGood(response: unknown): Message[] {
  // Must validate before use
  if (
    typeof response === 'object' &&
    response !== null &&
    'data' in response &&
    typeof response.data === 'object' &&
    response.data !== null &&
    'messages' in response.data &&
    Array.isArray(response.data.messages)
  ) {
    return response.data.messages as Message[]
  }

  throw new Error('Invalid response format')
}
```

### Type Guards Instead of Assertions

```tsx
import type { ChatMessage } from '@clarity-chat/react/types'

/**
 * ❌ Bad: Direct type assertion without validation
 */
function processMessageBad(data: unknown) {
  const message = data as ChatMessage // Unsafe!
  return message.content // Could crash if data is invalid
}

/**
 * ✓ Good: Type guard with validation
 */
function isChatMessage(data: unknown): data is ChatMessage {
  return (
    typeof data === 'object' &&
    data !== null &&
    'role' in data &&
    typeof (data as any).role === 'string' &&
    ['system', 'user', 'assistant'].includes((data as any).role) &&
    'content' in data &&
    (typeof (data as any).content === 'string' ||
      Array.isArray((data as any).content))
  )
}

function processMessageGood(data: unknown): string {
  if (!isChatMessage(data)) {
    throw new Error('Invalid message format')
  }

  // data is now typed as ChatMessage
  return typeof data.content === 'string'
    ? data.content
    : data.content.map((part) => part.text).join('')
}
```

### Zod for Runtime Validation

```tsx
import { z } from 'zod'
import type { ChatMessage } from '@clarity-chat/react/types'

/**
 * Runtime validation schema
 */
const ChatMessageSchema = z.object({
  role: z.enum(['system', 'user', 'assistant']),
  content: z.union([
    z.string(),
    z.array(
      z.object({
        type: z.enum(['text', 'image', 'tool_result']),
        text: z.string().optional(),
        imageUrl: z.string().optional(),
        toolCallId: z.string().optional(),
        toolResult: z.unknown().optional(),
      })
    ),
  ]),
  toolCalls: z
    .array(
      z.object({
        id: z.string(),
        type: z.literal('function'),
        function: z.object({
          name: z.string(),
          arguments: z.string(),
        }),
      })
    )
    .optional(),
  finishReason: z
    .enum(['stop', 'length', 'tool-calls', 'content-filter'])
    .optional(),
})

/**
 * Type-safe parser with runtime validation
 */
function parseMessage(data: unknown): ChatMessage {
  // Zod validates and returns typed data
  const validated = ChatMessageSchema.parse(data)

  // No type assertion needed - Zod guarantees the type
  return validated as ChatMessage
}

/**
 * Safe parsing with error handling
 */
function parseMessageSafe(data: unknown): ChatMessage | null {
  const result = ChatMessageSchema.safeParse(data)

  if (!result.success) {
```

### Example 2: Type-Safe Event Dispatcher

```tsx
import type {
  StrictChatEvent,
  EventType,
  EventPayload,
  StrictEventListener,
} from '@clarity-chat/react/types'

/**
 * Advanced type-safe event dispatcher with middleware
 */
class EventDispatcher {
  private listeners = new Map<EventType, Set<StrictEventListener>>()
  private middleware: Array<(event: StrictChatEvent) => StrictChatEvent | null> = []

  /**
   * Add middleware to transform/filter events
   */
  use(fn: (event: StrictChatEvent) => StrictChatEvent | null): void {
    this.middleware.push(fn)
  }

  /**
   * Register event listener with full type safety
   */
  on<T extends EventType>(
    type: T,
    listener: StrictEventListener<T>
  ): () => void {
    if (!this.listeners.has(type)) {
      this.listeners.set(type, new Set())
    }

    this.listeners.get(type)!.add(listener as StrictEventListener)

    // Return unsubscribe function
    return () => {
      this.listeners.get(type)?.delete(listener as StrictEventListener)
    }
  }

  /**
   * Register one-time event listener
   */
  once<T extends EventType>(
    type: T,
    listener: StrictEventListener<T>
  ): () => void {
    const wrappedListener = ((event: EventPayload<T>) => {
      listener(event)
      this.listeners.get(type)?.delete(wrappedListener as StrictEventListener)
    }) as StrictEventListener<T>

    return this.on(type, wrappedListener)
  }

  /**
   * Emit event with middleware pipeline
   */
  emit<T extends EventType>(type: T, payload: EventPayload<T>): void {
    let event: StrictChatEvent | null = payload as StrictChatEvent

    // Run through middleware pipeline
    for (const fn of this.middleware) {
      event = fn(event)
      if (event === null) {
        // Middleware filtered out the event
        return
      }
    }

    // Dispatch to listeners
    const listeners = this.listeners.get(type)
    if (listeners) {
      listeners.forEach((listener) => {
        try {
          listener(event as any)
        } catch (error) {
          console.error(`Error in listener for ${type}:`, error)
        }
      })
    }
  }

  /**
   * Remove all listeners for a type
   */
  removeAllListeners(type?: EventType): void {
    if (type) {
      this.listeners.delete(type)
    } else {
      this.listeners.clear()
    }
  }
}

// Usage with middleware
const dispatcher = new EventDispatcher()

// Add logging middleware
dispatcher.use((event) => {
  console.log('[Event]', event.type, event)
  return event
})

// Add filtering middleware
dispatcher.use((event) => {
  // Filter out error events in production
  if (process.env.NODE_ENV === 'production' && event.type === 'error:occurred') {
    return null
  }
  return event
})

// Type-safe event handling
dispatcher.on('message:sent', (event) => {
  // event is typed as MessageSentEvent
  console.log('Message sent:', event.content)
})

dispatcher.on('tool:completed', (event) => {
  // event is typed as ToolCompletedEvent
  console.log('Tool completed:', event.toolName, event.executionTimeMs)
})
```

### Example 3: Type-Safe State Machine

```tsx
/**
 * Chat state machine with discriminated unions
 */
type ChatState =
  | { status: 'idle' }
  | { status: 'connecting'; attemptCount: number }
  | { status: 'connected'; sessionId: string }
  | {
      status: 'streaming'
      messageId: string
      progress: number
      tokensReceived: number
    }
  | {
      status: 'error'
      error: Error
      retryable: boolean
      lastState: ChatState
    }

type ChatAction =
  | { type: 'CONNECT'; sessionId?: string }
  | { type: 'CONNECTED'; sessionId: string }
  | { type: 'START_STREAM'; messageId: string }
  | { type: 'STREAM_PROGRESS'; progress: number; tokens: number }
  | { type: 'STREAM_COMPLETE' }
  | { type: 'ERROR'; error: Error; retryable: boolean }
  | { type: 'RETRY' }
  | { type: 'RESET' }

/**
 * Type-safe state reducer
 */
function chatReducer(state: ChatState, action: ChatAction): ChatState {
  switch (action.type) {
    case 'CONNECT':
      if (state.status === 'idle') {
        return {
          status: 'connecting',
          attemptCount: 1,
        }
      }
      if (state.status === 'error' && state.retryable) {
        return {
          status: 'connecting',
          attemptCount:
            state.lastState.status === 'connecting'
              ? state.lastState.attemptCount + 1
              : 1,
        }
      }
      return state

    case 'CONNECTED':
      if (state.status === 'connecting') {
        return {
          status: 'connected',
          sessionId: action.sessionId,
        }
      }
      return state

    case 'START_STREAM':
      if (state.status === 'connected') {
        return {
          status: 'streaming',
          messageId: action.messageId,
          progress: 0,
          tokensReceived: 0,
        }
      }
      return state

    case 'STREAM_PROGRESS':
      if (state.status === 'streaming') {
        return {
          ...state,
          progress: action.progress,
          tokensReceived: state.tokensReceived + action.tokens,
        }
      }
      return state

    case 'STREAM_COMPLETE':
      if (state.status === 'streaming') {
        return { status: 'idle' }
      }
      return state

    case 'ERROR':
      return {
        status: 'error',
        error: action.error,
        retryable: action.retryable,
        lastState: state,
      }

    case 'RETRY':
      if (state.status === 'error' && state.retryable) {
        return state.lastState
      }
      return state

    case 'RESET':
      return { status: 'idle' }

    default:
      const _exhaustive: never = action
      return state
  }
}

/**
 * Type-safe state machine hook
 */
function useChatStateMachine() {
  const [state, dispatch] = React.useReducer(chatReducer, { status: 'idle' })

  const connect = React.useCallback((sessionId?: string) => {
    dispatch({ type: 'CONNECT', sessionId })
  }, [])

  const startStream = React.useCallback((messageId: string) => {
    dispatch({ type: 'START_STREAM', messageId })
  }, [])

  const updateProgress = React.useCallback((progress: number, tokens: number) => {
    dispatch({ type: 'STREAM_PROGRESS', progress, tokens })
  }, [])

  const completeStream = React.useCallback(() => {
    dispatch({ type: 'STREAM_COMPLETE' })
  }, [])

  const handleError = React.useCallback((error: Error, retryable: boolean) => {
    dispatch({ type: 'ERROR', error, retryable })
  }, [])

  const retry = React.useCallback(() => {
    dispatch({ type: 'RETRY' })
  }, [])

  const reset = React.useCallback(() => {
    dispatch({ type: 'RESET' })
  }, [])

  return {
    state,
    connect,
    startStream,
    updateProgress,
    completeStream,
    handleError,
    retry,
    reset,
  }
}

// Usage
function ChatComponent() {
  const machine = useChatStateMachine()

  React.useEffect(() => {
    // Type-safe state handling
    switch (machine.state.status) {
      case 'idle':
        console.log('Chat is idle')
        break

      case 'connecting':
        console.log('Connecting, attempt:', machine.state.attemptCount)
        break

      case 'connected':
        console.log('Connected with session:', machine.state.sessionId)
        break

      case 'streaming':
        console.log(
          'Streaming message:',
          machine.state.messageId,
          'Progress:',
          machine.state.progress
        )
        break

      case 'error':
        console.error('Error:', machine.state.error.message)
        if (machine.state.retryable) {
          console.log('Error is retryable')
        }
        break

      default:
        const _exhaustive: never = machine.state
    }
  }, [machine.state])

  return <div>{/* Component JSX */}</div>
}
```
