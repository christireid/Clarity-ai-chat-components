# Vercel AI SDK Adapter

Complete integration guide for using Clarity Chat Components with [Vercel AI SDK](https://sdk.vercel.ai/docs). This adapter provides seamless connection between your AI backend and Clarity's UI components.

## Features

- Streaming responses with real-time updates
- Tool/function calling support
- Multiple provider support (OpenAI, Anthropic, Google)
- Automatic error handling and retries
- Cost tracking integration
- Type-safe message mapping
- Event-driven state management

## Installation

```bash
npm install @clarity-chat/react ai
# or
pnpm add @clarity-chat/react ai
```

## Error Handling and Retries

### Automatic Retry Configuration

```typescript
import { openai } from '@ai-sdk/openai'
import { streamText } from 'ai'

export async function POST(req: Request) {
  const { messages } = await req.json()

  try {
    const result = await streamText({
      model: openai('gpt-4o'),
      messages,
      maxRetries: 3, // Automatic retries on failure
      abortSignal: req.signal, // Support request cancellation
    })

    return result.toDataStreamResponse()
  } catch (error) {
    // Handle different error types
    if (error instanceof Error) {
      if (error.message.includes('rate limit')) {
        return new Response(
          JSON.stringify({
            error: 'Rate limit exceeded. Please try again later.'
          }),
          { status: 429 }
        )
      }

      if (error.message.includes('timeout')) {
        return new Response(
          JSON.stringify({
            error: 'Request timed out. Please try again.'
          }),
          { status: 504 }
        )
      }
    }

    return new Response(
      JSON.stringify({ error: 'An unexpected error occurred' }),
      { status: 500 }
    )
  }
}
```

```tsx
      } else {
        setErrorMessage('Something went wrong. Please try again.')
      }
    },
  })

  const adapter = createVercelAIAdapter({
    chat,
    onError: (error) => {
      // Additional adapter-level error handling
      console.error('Adapter error:', error)
    },
  })

    <div>
      {errorMessage && (
        <div className="error-banner">
          {errorMessage}
          <button onClick={() => setErrorMessage(undefined)}>Dismiss</button>
        </div>
      )}
      <ClarityChatProvider adapter={adapter}>
        <ChatWindow />
      </ClarityChatProvider>
    </div>
  )
}
```

## Cost Tracking Integration

### Server-Side Cost Tracking

```typescript
import { openai } from '@ai-sdk/openai'
import { streamText } from 'ai'
import { CostTracker } from '@clarity-chat/token-optimization'

// Create a cost tracker (persist in your database/cache)
const costTracker = new CostTracker('gpt-4o')

export async function POST(req: Request) {
  const { messages } = await req.json()

  const result = await streamText({
    model: openai('gpt-4o'),
    messages,
    onFinish: async ({ usage }) => {
      // Track cost with token usage
      const cost = costTracker.trackRequest({
        inputTokens: usage.promptTokens,
        outputTokens: usage.completionTokens,
        cachedInputTokens: 0, // Add cache metrics if using provider caching
      })

      // Log cost information
      console.log('Request cost:', {
        totalCost: cost.totalCost,
        savings: cost.savingsPercentage,
        tokens: usage.totalTokens,
      })

      // Get cumulative statistics
      const report = costTracker.getReport()
      console.log('Cumulative savings:', {
        totalSaved: report.cumulative.totalSavings,
        averageSavings: report.cumulative.averageSavingsPercentage,
        requestCount: report.cumulative.requestCount,
      })
    },
  })

  return result.toDataStreamResponse()
}
```

### Advanced Cost Optimization

```tsx
'use client'

import { useChat } from 'ai/react'
import { createVercelAIAdapter } from '@clarity-chat/react/adapters'
import { ClarityChatProvider, ChatWindow } from '@clarity-chat/react'
import { useTokenOptimization } from '@clarity-chat/token-optimization'

export function OptimizedChat() {
  const { optimize, compress, route, getStats } = useTokenOptimization({
    preset: 'production',
    model: 'gpt-4o',
    enableCache: true,
    enableCompression: true,
    enableRouting: true,
  })

  const chat = useChat({
    api: '/api/chat',
    onResponse: async (response) => {
      const stats = getStats()
      console.log('Optimization stats:', {
        tokensSaved: stats.tokensSaved,
        cacheHits: stats.cache?.hits,
        cacheHitRate: stats.cache?.hitRate,
      })
    },
  })

  const adapter = createVercelAIAdapter({
    chat,
    onError: (error) => {
      console.error('Chat error:', error)
    },
  })

  // Optimize messages before sending
  const handleOptimizedSend = async (message: string) => {
    const result = await optimize(message)

    if (result.cachedResponse) {
      // Use cached response
      return result.cachedResponse
    }

    // Send optimized message
    await chat.append({
      role: 'user',
      content: result.compressedPrompt,
    })

    console.log('Optimization results:', {
      tokensSaved: result.tokensSaved,
      estimatedCostSaved: result.estimatedCostSaved,
      recommendedModel: result.recommendedModel,
    })
  }

  return (
    <ClarityChatProvider adapter={adapter}>
      <ChatWindow onSend={handleOptimizedSend} />
    </ClarityChatProvider>
  )
}
```

```typescript
export const runtime = 'edge'

// Initialize cost tracker
const costTracker = new CostTracker('gpt-4o')

// Define tools
const tools = {

      }
    },
  }),
  searchKnowledge: tool({
    description: 'Search the knowledge base',
    parameters: z.object({
      query: z.string().describe('Search query'),
    }),
    execute: async ({ query }) => {
      // Mock implementation
      return {
        results: [
          { title: 'Result 1', content: 'Sample content...' },
          { title: 'Result 2', content: 'More content...' },
        ],
      }
    },
  }),
}

export async function POST(req: Request) {
  try {
    const { messages, model = 'gpt-4o' } = await req.json()

    // Select provider based on model
    const providers = {
      'gpt-4o': openai('gpt-4o'),
      'gpt-4o-mini': openai('gpt-4o-mini'),
      'claude-3-5-sonnet': anthropic('claude-3-5-sonnet-20241022'),
    }

    const selectedModel = providers[model as keyof typeof providers] || providers['gpt-4o']

    // Stream with tools
    const result = await streamText({
      model: selectedModel,
      messages,
      tools,
      maxTokens: 4096,
      temperature: 0.7,
      maxRetries: 3,
      onFinish: async ({ text, usage, toolCalls }) => {
        // Track cost
        const cost = costTracker.trackRequest({
          inputTokens: usage.promptTokens,
          outputTokens: usage.completionTokens,
          cachedInputTokens: 0,
        })

        // Log metrics
        console.log('Request completed:', {
          model,
          tokens: usage.totalTokens,
          cost: cost.totalCost,
          toolCalls: toolCalls?.length || 0,
        })

        // Get savings report
        const report = costTracker.getReport()
        console.log('Cumulative stats:', {
          totalSaved: report.cumulative.totalSavings,
          avgSavings: report.cumulative.averageSavingsPercentage,
          requests: report.cumulative.requestCount,
        })
      },
    })

    return result.toDataStreamResponse()
  } catch (error) {
    console.error('API error:', error)

    // Handle specific error types
    if (error instanceof Error) {
      if (error.message.includes('rate limit')) {
        return new Response(
          JSON.stringify({
            error: 'Rate limit exceeded',
            retryAfter: 60,
          }),
          {
            status: 429,
            headers: { 'Retry-After': '60' },
          }
        )
      }

      if (error.message.includes('timeout')) {
        return new Response(
          JSON.stringify({ error: 'Request timeout' }),
          { status: 504 }
        )
      }

      if (error.message.includes('authentication')) {
        return new Response(
          JSON.stringify({ error: 'Authentication failed' }),
          { status: 401 }
        )
      }
    }

    return new Response(
      JSON.stringify({ error: 'Internal server error' }),
      { status: 500 }
    )
  }
}
```

## Advanced Patterns

### Message Persistence

```tsx
'use client'

import { useChat } from 'ai/react'
import { createVercelAIAdapter } from '@clarity-chat/react/adapters'
import { useEffect } from 'react'

export function PersistentChat() {
  const chat = useChat({
    api: '/api/chat',
    id: 'my-chat-session', // Persist by session ID
  })

  // Save messages to localStorage
  useEffect(() => {
    localStorage.setItem('chat-messages', JSON.stringify(chat.messages))
  }, [chat.messages])

  // Load messages on mount
  useEffect(() => {
    const saved = localStorage.getItem('chat-messages')
    if (saved) {
      const messages = JSON.parse(saved)
      chat.setMessages(messages)
    }
  }, [])

  const adapter = createVercelAIAdapter({ chat })

  return <ClarityChatProvider adapter={adapter}>...</ClarityChatProvider>
}
```

### Conversation Branching

```tsx
'use client'

import { useChat } from 'ai/react'
import { createVercelAIAdapter } from '@clarity-chat/react/adapters'
import { useState } from 'react'

export function BranchingChat() {
  const [branches, setBranches] = useState<Map<string, any[]>>(new Map())

  const chat = useChat({
    api: '/api/chat',
  })

  const createBranch = (fromMessageId: string) => {
    const messageIndex = chat.messages.findIndex((m) => m.id === fromMessageId)
    if (messageIndex === -1) return

    const branchMessages = chat.messages.slice(0, messageIndex + 1)
    const branchId = `branch-${Date.now()}`

    setBranches(new Map(branches.set(branchId, branchMessages)))

    return branchId
  }

  const switchToBranch = (branchId: string) => {
    const branchMessages = branches.get(branchId)
    if (branchMessages) {
      chat.setMessages(branchMessages)
    }
  }

  const adapter = createVercelAIAdapter({ chat })

  return (
    <div>
      <div className="branch-selector">
        {Array.from(branches.keys()).map((branchId) => (
          <button key={branchId} onClick={() => switchToBranch(branchId)}>
            Switch to {branchId}
          </button>
        ))}
      </div>
      <ClarityChatProvider adapter={adapter}>...</ClarityChatProvider>
    </div>
  )
}
```

## Best Practices

### 1. Enable Streaming for Better UX

Always use streaming for real-time feedback:

```typescript
const result = await streamText({
  model: openai('gpt-4o'),
  messages,
  // streaming is enabled by default
})
```

### 2. Implement Proper Error Handling

Handle errors at both API and client levels:

```typescript
// Server
try {
  const result = await streamText({ ... })
  return result.toDataStreamResponse()
} catch (error) {
  // Log and return appropriate error
  return new Response(JSON.stringify({ error }), { status: 500 })
}

// Client
const chat = useChat({
  api: '/api/chat',
  onError: (error) => {
    // Show user-friendly error message
    console.error(error)
  },
})
```

### 3. Track Costs and Usage

Monitor token usage and costs:

```typescript
const costTracker = new CostTracker('gpt-4o')

const result = await streamText({
  model: openai('gpt-4o'),
  messages,
  onFinish: ({ usage }) => {
    costTracker.trackRequest({
      inputTokens: usage.promptTokens,
      outputTokens: usage.completionTokens,
    })
  },
})
```

### 4. Use Provider Caching

Enable provider-native caching for cost savings:

```typescript
import { anthropic } from '@ai-sdk/anthropic'

// Anthropic automatically caches system messages
const result = await streamText({
  model: anthropic('claude-3-5-sonnet-20241022'),
  system: 'You are a helpful assistant...', // This will be cached
  messages,
})
```

### 5. Optimize Token Usage

Compress prompts before sending:

```tsx
const { optimize } = useTokenOptimization({ preset: 'production' })

const handleSend = async (message: string) => {
  const { compressedPrompt, tokensSaved } = await optimize(message)
  await chat.append({ role: 'user', content: compressedPrompt })
  console.log(`Saved ${tokensSaved} tokens`)
}
```

## Troubleshooting

### Messages Not Appearing

Ensure your API route returns the correct response format:

```typescript
return result.toDataStreamResponse()
```

### Tools Not Working

Make sure tools are defined on both client and server:

```typescript
// Server: tool execution
const tools = { myTool: tool({ ... }) }

// Client: tool definitions
const adapter = createVercelAIAdapter({
  chat,
  tools: [{ name: 'myTool', ... }],
})
```

### Cost Tracking Inaccurate

Use the `onFinish` callback to get accurate token counts:

```typescript
const result = await streamText({
  model,
  messages,
  onFinish: ({ usage }) => {
    console.log('Actual tokens used:', usage.totalTokens)
  },
})
```
