## Overview

Remix provides a modern approach to building React applications with:

- **Server-Side Rendering (SSR)**: Fast initial page loads
- **Progressive Enhancement**: Works without JavaScript
- **Form Actions**: Server-side mutations with type safety
- **Streaming**: Real-time data with EventStream
- **Nested Routes**: Component-based routing
- **Error Boundaries**: Granular error handling

### Key Benefits

- SEO-friendly AI chat interfaces
- Fast initial load times
- Offline-first capabilities
- Progressive enhancement
- Type-safe data loading

## Form Actions

### Send Message Action

```typescript
// app/routes/chat.$id.tsx
import { json, type ActionFunctionArgs } from '@remix-run/node'
import { Form, useActionData, useNavigation } from '@remix-run/react'
import { z } from 'zod'

const messageSchema = z.object({
  content: z.string().min(1).max(4000),
  conversationId: z.string(),
})

// Server-side form action
export async function action({ request }: ActionFunctionArgs) {
  const formData = await request.formData()
  const data = Object.fromEntries(formData)

  // Validate input
  const result = messageSchema.safeParse(data)
  if (!result.success) {
    return json({ error: 'Invalid message' }, { status: 400 })
  }

  const { content, conversationId } = result.data

  try {
    // Save user message
    await db.message.create({
      data: {
        conversationId,
        role: 'user',
        content,
      },
    })

    // Generate AI response
    const messages = await db.message.findMany({
      where: { conversationId },
      orderBy: { createdAt: 'asc' },
    })

    const aiResponse = await generateAIResponse(messages)

    // Save AI response
    await db.message.create({
      data: {
        conversationId,
        role: 'assistant',
        content: aiResponse,
      },
    })

    return json({ success: true })
  } catch (error) {
    return json({ error: 'Failed to send message' }, { status: 500 })
  }
}

// Component with Form
export default function ChatRoute() {
  const { conversation } = useLoaderData<typeof loader>()
  const actionData = useActionData<typeof action>()
  const navigation = useNavigation()
  const isSubmitting = navigation.state === 'submitting'

  return (
    <div className="h-screen flex flex-col">
      <div className="flex-1 overflow-auto p-4">
        {conversation.messages.map((message) => (
          <div key={message.id} className="mb-4">
            <strong>{message.role}:</strong> {message.content}
          </div>
        ))}
      </div>

      <Form method="post" className="border-t p-4">
        <input type="hidden" name="conversationId" value={conversation.id} />
        <div className="flex gap-2">
          <input
            type="text"
            name="content"
            className="flex-1 px-4 py-2 border rounded-lg"
            placeholder="Type your message..."
            disabled={isSubmitting}
            required
          />
          <button
            type="submit"
            className="px-6 py-2 bg-brand-500 text-white rounded-lg disabled:opacity-50"
            disabled={isSubmitting}
          >
            {isSubmitting ? 'Sending...' : 'Send'}
          </button>
        </div>
        {actionData?.error && <div className="text-red-500 mt-2">{actionData.error}</div>}
      </Form>
    </div>
  )
}
```

## Streaming Responses

### Server-Sent Events (SSE)

```typescript
// app/routes/api.chat.stream.tsx
import { type LoaderFunctionArgs } from '@remix-run/node'

export async function loader({ request }: LoaderFunctionArgs) {
  const url = new URL(request.url)
  const conversationId = url.searchParams.get('conversationId')

  if (!conversationId) {
    throw new Response('Missing conversationId', { status: 400 })
  }

  // Create event stream
  const stream = new ReadableStream({
    async start(controller) {
      const encoder = new TextEncoder()

      try {
        // Get conversation messages
        const messages = await db.message.findMany({
          where: { conversationId },
          orderBy: { createdAt: 'asc' },
        })

        // Stream AI response
        const response = await streamAIResponse(messages)

        for await (const chunk of response) {
          controller.enqueue(encoder.encode(`data: ${JSON.stringify({ chunk })}\n\n`))
        }

        controller.enqueue(encoder.encode('data: [DONE]\n\n'))
        controller.close()
      } catch (error) {
        controller.enqueue(encoder.encode(`data: ${JSON.stringify({ error: 'Stream failed' })}\n\n`))
        controller.close()
      }
    },
  })

  return new Response(stream, {
    headers: {
      'Content-Type': 'text/event-stream',
      'Cache-Control': 'no-cache',
      Connection: 'keep-alive',
    },
  })
}
```

### Client-Side Stream Consumer

```typescript
// app/hooks/useStreamingChat.ts
import { useState, useEffect } from 'react'

export function useStreamingChat(conversationId: string) {
  const [streamedContent, setStreamedContent] = useState('')
  const [isStreaming, setIsStreaming] = useState(false)

  const startStreaming = () => {
    setIsStreaming(true)
    setStreamedContent('')

    const eventSource = new EventSource(`/api/chat/stream?conversationId=${conversationId}`)

    eventSource.onmessage = (event) => {
      if (event.data === '[DONE]') {
        eventSource.close()
        setIsStreaming(false)
        return
      }

      try {
        const data = JSON.parse(event.data)
        if (data.chunk) {
          setStreamedContent((prev) => prev + data.chunk)
        }
      } catch (error) {
        console.error('Failed to parse stream data:', error)
      }
    }

    eventSource.onerror = () => {
      eventSource.close()
      setIsStreaming(false)
    }

    return () => eventSource.close()
  }

  return { streamedContent, isStreaming, startStreaming }
}
```

## Optimistic UI

### Optimistic Message Updates

```typescript
// app/routes/chat.$id.tsx
import { useFetcher } from '@remix-run/react'
import { useState, useOptimistic } from 'react'

export default function ChatRoute() {
  const { conversation } = useLoaderData<typeof loader>()
  const fetcher = useFetcher()
  const [optimisticMessages, addOptimisticMessage] = useOptimistic(
    conversation.messages,
    (state, newMessage: Message) => [...state, newMessage]
  )

  const handleSendMessage = (content: string) => {
    // Add optimistic message
    const tempMessage: Message = {
      id: `temp-${Date.now()}`,
      role: 'user',
      content,
    }
    addOptimisticMessage(tempMessage)

    // Submit to server
    fetcher.submit(
      {
        content,
        conversationId: conversation.id,
      },
      { method: 'post' }
    )
  }

  return (
    <div className="h-screen flex flex-col">
      <div className="flex-1 overflow-auto p-4">
        {optimisticMessages.map((message) => (
          <div key={message.id} className="mb-4">
            <strong>{message.role}:</strong> {message.content}
          </div>
        ))}
      </div>

      <ChatInput onSend={handleSendMessage} isLoading={fetcher.state === 'submitting'} />
    </div>
  )
}
```

## Complete Example

```typescript
// app/routes/chat.$id.tsx - Complete production example
import { json, redirect, type LoaderFunctionArgs, type ActionFunctionArgs } from '@remix-run/node'
import {
  useLoaderData,
  useActionData,
  useFetcher,
  Link,
  isRouteErrorResponse,
  useRouteError,
} from '@remix-run/react'
import { ClarityChatApp } from '@clarity-chat/react'
import type { Message } from '@clarity-chat/react'
import { z } from 'zod'

// Validation schema
const messageSchema = z.object({
  content: z.string().min(1).max(4000),
  conversationId: z.string(),
})

// Server-side data loading
export async function loader({ params, request }: LoaderFunctionArgs) {
  const session = await getSession(request.headers.get('Cookie'))
  const userId = session.get('userId')

  if (!userId) {
    return redirect('/login')
  }

  const conversation = await db.conversation.findUnique({
    where: { id: params.id, userId },
    include: { messages: { orderBy: { createdAt: 'asc' } } },
  })

  if (!conversation) {
    throw new Response('Not Found', { status: 404 })
  }

  return json({ conversation })
}

// Server-side form action
export async function action({ request }: ActionFunctionArgs) {
  const formData = await request.formData()
  const result = messageSchema.safeParse(Object.fromEntries(formData))

  if (!result.success) {
    return json({ error: result.error.format() }, { status: 400 })
  }

  const { content, conversationId } = result.data

  try {
    await db.message.create({
      data: { conversationId, role: 'user', content },
    })

    const messages = await db.message.findMany({
      where: { conversationId },
      orderBy: { createdAt: 'asc' },
    })

    const aiResponse = await generateAIResponse(messages)

    await db.message.create({
      data: { conversationId, role: 'assistant', content: aiResponse },
    })

    return json({ success: true })
  } catch (error) {
    return json({ error: 'Failed to send message' }, { status: 500 })
  }
}

// Component
export default function ChatRoute() {
  const { conversation } = useLoaderData<typeof loader>()
  const fetcher = useFetcher()

  return (
    <div className="h-screen flex flex-col">
      <header className="border-b p-4 flex justify-between items-center">
        <h1 className="text-xl font-bold">{conversation.title}</h1>
        <Link to="/chat" className="text-brand-500 hover:underline">
          All Chats
        </Link>
      </header>

      <ClarityChatApp
        messages={conversation.messages as Message[]}
        onSendMessage={(content) => {
          fetcher.submit(
            { content, conversationId: conversation.id },
            { method: 'post' }
          )
        }}
        isLoading={fetcher.state === 'submitting'}
      />
    </div>
  )
}

// Error boundary
export function ErrorBoundary() {
  const error = useRouteError()

  if (isRouteErrorResponse(error) && error.status === 404) {
    return (
      <div className="flex items-center justify-center h-screen">
        <div className="text-center">
          <h2 className="text-2xl font-bold mb-4">Conversation Not Found</h2>
          <Link to="/chat" className="text-brand-500 hover:underline">
            Back to Chat
          </Link>
        </div>
      </div>
    )
  }

  return (
    <div className="flex items-center justify-center h-screen">
      <h2 className="text-2xl font-bold">Something went wrong</h2>
    </div>
  )
}
```
