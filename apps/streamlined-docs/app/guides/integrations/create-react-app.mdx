### Basic App Component

```typescript
// src/App.tsx
import React, { useState } from 'react'
import { ClarityChatApp } from '@clarity-chat/react'
import type { Message } from '@clarity-chat/react'
import './App.css'

function App() {
  const [messages, setMessages] = useState<Message[]>([
    {
      id: '1',
      role: 'assistant',
      content: 'Hello! How can I help you today?',
    },
  ])

  const handleSendMessage = async (content: string) => {
    // Add user message
    const userMessage: Message = {
      id: Date.now().toString(),
      role: 'user',
      content,
    }
    setMessages((prev) => [...prev, userMessage])

    // Simulate AI response
    const assistantMessage: Message = {
      id: (Date.now() + 1).toString(),
      role: 'assistant',
      content: `You said: ${content}`,
    }
    setMessages((prev) => [...prev, assistantMessage])
  }

  return (
    <div className="App min-h-screen bg-background">
      <ClarityChatApp
        messages={messages}
        onSendMessage={handleSendMessage}
        placeholder="Type your message..."
      />
    </div>
  )
}

export default App
```

## Styling Integration

### Dark Mode Setup

```typescript
// src/hooks/useDarkMode.ts
import { useEffect, useState } from 'react'

export function useDarkMode() {
  const [isDark, setIsDark] = useState(() => {
    const stored = localStorage.getItem('theme')
    return stored === 'dark' || (!stored && window.matchMedia('(prefers-color-scheme: dark)').matches)
  })

  useEffect(() => {
    const root = document.documentElement
    if (isDark) {
      root.classList.add('dark')
      localStorage.setItem('theme', 'dark')
    } else {
      root.classList.remove('dark')
      localStorage.setItem('theme', 'light')
    }
  }, [isDark])

  return { isDark, toggleDark: () => setIsDark(!isDark) }
}
```

### Theme Provider

```typescript
// src/providers/ThemeProvider.tsx
import React, { createContext, useContext, ReactNode } from 'react'
import { useDarkMode } from '../hooks/useDarkMode'

interface ThemeContextType {
  isDark: boolean
  toggleDark: () => void
}

const ThemeContext = createContext<ThemeContextType | undefined>(undefined)

export function ThemeProvider({ children }: { children: ReactNode }) {
  const { isDark, toggleDark } = useDarkMode()

  return <ThemeContext.Provider value={{ isDark, toggleDark }}>{children}</ThemeContext.Provider>
}

export function useTheme() {
  const context = useContext(ThemeContext)
  if (!context) {
    throw new Error('useTheme must be used within ThemeProvider')
  }
  return context
}
```

### App with Theme Provider

```typescript
// src/App.tsx
import React from 'react'
import { ThemeProvider, useTheme } from './providers/ThemeProvider'
import { ClarityChatApp } from '@clarity-chat/react'

function ChatApp() {
  const { isDark, toggleDark } = useTheme()

  return (
    <div className="min-h-screen">
      <header className="border-b p-4 flex justify-between items-center">
        <h1 className="text-2xl font-bold">AI Chat</h1>
        <button
          onClick={toggleDark}
          className="px-4 py-2 rounded-lg bg-neutral-200 dark:bg-neutral-800"
        >
          {isDark ? '☀️' : '🌙'}
        </button>
      </header>

      <ClarityChatApp
        messages={[]}
        onSendMessage={async (content) => {
          console.log('Message:', content)
        }}
      />
    </div>
  )
}

function App() {
  return (
    <ThemeProvider>
      <ChatApp />
    </ThemeProvider>
  )
}

export default App
```

### Security Best Practices


```javascript
module.exports = override(
```

## Troubleshooting

### Common Issues

#### 1. Module Not Found

```bash
# Clear cache and reinstall
rm -rf node_modules package-lock.json
npm install

# Clear CRA cache
rm -rf node_modules/.cache
```

#### 2. Tailwind Not Working

```bash
# Ensure Tailwind is properly configured
# Check that content paths include node_modules

# tailwind.config.js
content: [
  './src/**/*.{js,jsx,ts,tsx}',
  './node_modules/@clarity-chat/**/*.{js,jsx,ts,tsx}',
],
```

#### 3. TypeScript Errors

```bash
# Update TypeScript types
npm install -D @types/react @types/react-dom

# Check tsconfig.json includes src folder
"include": ["src"]
```

#### 4. Build Fails

```bash
# Increase Node memory limit
# Add to package.json scripts:
"build": "NODE_OPTIONS=--max_old_space_size=4096 react-scripts build"
```

### Performance Issues

#### Slow Development Server

```bash
# Disable source maps in development
# Create .env.development
GENERATE_SOURCEMAP=false
```

#### Large Bundle Size

```typescript
// Use dynamic imports for large dependencies
const HeavyComponent = lazy(() => import('./HeavyComponent'))

// Split vendor chunks
// config-overrides.js
optimization: {
  splitChunks: {
```

## Migration to Modern Tools

### Why Migrate?

Create React App is no longer actively maintained. Consider migrating to:

1. **Vite** - Fastest build tool, best DX
2. **Next.js** - Full-stack framework with SSR
3. **Remix** - Modern full-stack with better data loading

### Migration Path to Vite

```bash
# 1. Install Vite
npm install -D vite @vitejs/plugin-react

# 2. Create vite.config.ts
# See Vite Integration Guide

# 3. Update index.html
# Move to project root
# Change script imports

# 4. Update package.json scripts
"dev": "vite",
"build": "vite build",
"preview": "vite preview"

# 5. Test and migrate
npm run dev
```

### Migration Path to Next.js

```bash
# 1. Create new Next.js app
npx create-next-app@latest my-next-app

# 2. Copy components from src/
# Move to app/ or pages/

# 3. Update imports and API calls
# See Next.js Integration Guide

# 4. Test SSR compatibility
npm run dev
```

## Complete Example

### Production-Ready Setup

```typescript
// src/App.tsx
import React, { useState, useCallback } from 'react'
import { ClarityChatApp } from '@clarity-chat/react'
import { ThemeProvider } from './providers/ThemeProvider'
import { ErrorBoundary } from './components/ErrorBoundary'
import type { Message } from '@clarity-chat/react'

function ChatApp() {
  const [messages, setMessages] = useState<Message[]>([])
  const [isLoading, setIsLoading] = useState(false)

  const handleSendMessage = useCallback(async (content: string) => {
    const userMessage: Message = {
      id: Date.now().toString(),
      role: 'user',
      content,
    }
    setMessages((prev) => [...prev, userMessage])
    setIsLoading(true)

    try {
      const response = await fetch('/api/chat', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ messages: [...messages, userMessage] }),
      })

      if (!response.ok) throw new Error('API request failed')

      const data = await response.json()
      const assistantMessage: Message = {
        id: (Date.now() + 1).toString(),
        role: 'assistant',
        content: data.content,
      }
      setMessages((prev) => [...prev, assistantMessage])
    } catch (error) {
      console.error('Error sending message:', error)
    } finally {
      setIsLoading(false)
    }
  }, [messages])

  return (
    <div className="h-screen">
      <ClarityChatApp
        messages={messages}
        onSendMessage={handleSendMessage}
        isLoading={isLoading}
        placeholder="Ask me anything..."
      />
    </div>
  )
}

function App() {
  return (
    <ErrorBoundary>
      <ThemeProvider>
        <ChatApp />
      </ThemeProvider>
    </ErrorBoundary>
  )
}

export default App
```
