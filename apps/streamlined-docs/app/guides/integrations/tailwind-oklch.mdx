# TailwindCSS + OKLCH Color System Integration

OKLCH (Lightness, Chroma, Hue) is a perceptually uniform color space that provides superior color consistency, predictable lightness, and better accessibility compared to traditional HSL or RGB. This guide covers complete integration with TailwindCSS.

## Why OKLCH?

### Perceptual Uniformity

Unlike HSL where `hsl(0, 50%, 50%)` and `hsl(240, 50%, 50%)` have vastly different perceived brightness, OKLCH ensures equal lightness values appear equally bright to the human eye.

```css
/* HSL - Inconsistent perceived brightness */
--red-hsl: hsl(0, 70%, 50%);     /* Appears very bright */
--blue-hsl: hsl(240, 70%, 50%);  /* Appears much darker */

/* OKLCH - Consistent perceived brightness */
--red-oklch: oklch(60% 0.20 30);   /* Perceptually consistent */
--blue-oklch: oklch(60% 0.20 270); /* Same brightness as red */
```

### Superior Color Manipulation

OKLCH's three independent channels make color manipulation intuitive:

- **L (Lightness)**: 0% (black) to 100% (white)
- **C (Chroma)**: 0 (grayscale) to 0.4+ (vivid)
- **H (Hue)**: 0° to 360° (red → yellow → green → blue → magenta → red)

### Built-in Accessibility

Achieving WCAG contrast ratios is straightforward in OKLCH:

```css
/* Easy contrast control via lightness channel */
--background: oklch(98% 0.01 240);  /* Light background */
--foreground: oklch(20% 0.02 240);  /* Dark text - guaranteed 4.5:1+ contrast */
```

### Future-Proof

OKLCH is part of the CSS Color Module Level 4 specification and supported in all modern browsers (Safari 15.4+, Chrome 111+, Firefox 113+).

## Custom OKLCH Palette Generator

### JavaScript Utility

```typescript
// lib/oklch-palette.ts

export interface OKLCHColor {
  l: number // 0-100
  c: number // 0-0.4
  h: number // 0-360
}

export interface PaletteOptions {
  baseColor: OKLCHColor
  steps?: number
  contrastRatio?: number
}

/**
 * Generate a full color palette from a brand color
 */
export function generateOKLCHPalette(options: PaletteOptions) {
  const { baseColor, steps = 11, contrastRatio = 4.5 } = options
  const palette: Record<string, string> = {}

  // Generate lightness scale (50-950)
  for (let i = 0; i < steps; i++) {
    const step = (i + 1) * 100

    // Calculate lightness: 95% (50) -> 10% (950)
    const lightness = 95 - (i * 85 / (steps - 1))

    // Reduce chroma for very light/dark colors
    let chroma = baseColor.c
    if (lightness > 90) chroma *= 0.5
    if (lightness < 20) chroma *= 0.7

    palette[step] = `oklch(${lightness}% ${chroma} ${baseColor.h})`
  }

  return palette
}

/**
 * Calculate WCAG contrast ratio between two OKLCH colors
 */
export function calculateContrast(color1: OKLCHColor, color2: OKLCHColor): number {
  const l1 = color1.l / 100
  const l2 = color2.l / 100

  const lighter = Math.max(l1, l2)
  const darker = Math.min(l1, l2)

  return (lighter + 0.05) / (darker + 0.05)
}

/**
 * Find accessible foreground color for given background
 */
export function findAccessibleForeground(
  background: OKLCHColor,
  targetContrast: number = 4.5
): OKLCHColor {
  const isDark = background.l < 50
  const targetLightness = isDark ? 95 : 15

  return {
    l: targetLightness,
    c: background.c * 0.5,
    h: background.h,
  }
}

/**
 * Convert hex to OKLCH
 */
export function hexToOKLCH(hex: string): OKLCHColor {
  // Remove # if present
  hex = hex.replace('#', '')

  // Convert to RGB
  const r = parseInt(hex.substring(0, 2), 16) / 255
  const g = parseInt(hex.substring(2, 4), 16) / 255
  const b = parseInt(hex.substring(4, 6), 16) / 255

  // Convert RGB to Linear RGB
  const linearR = r <= 0.04045 ? r / 12.92 : Math.pow((r + 0.055) / 1.055, 2.4)
  const linearG = g <= 0.04045 ? g / 12.92 : Math.pow((g + 0.055) / 1.055, 2.4)
  const linearB = b <= 0.04045 ? b / 12.92 : Math.pow((b + 0.055) / 1.055, 2.4)

  // Convert Linear RGB to XYZ (D65 illuminant)
  const x = linearR * 0.4124564 + linearG * 0.3575761 + linearB * 0.1804375
  const y = linearR * 0.2126729 + linearG * 0.7151522 + linearB * 0.0721750
  const z = linearR * 0.0193339 + linearG * 0.1191920 + linearB * 0.9503041

  // Convert XYZ to OKLab
  const l_ = Math.cbrt(0.8189330101 * x + 0.3618667424 * y - 0.1288597137 * z)
  const m_ = Math.cbrt(0.0329845436 * x + 0.9293118715 * y + 0.0361456387 * z)
  const s_ = Math.cbrt(0.0482003018 * x + 0.2643662691 * y + 0.6338517070 * z)

  const l = 0.2104542553 * l_ + 0.7936177850 * m_ - 0.0040720468 * s_
  const a = 1.9779984951 * l_ - 2.4285922050 * m_ + 0.4505937099 * s_
  const b_ = 0.0259040371 * l_ + 0.7827717662 * m_ - 0.8086757660 * s_

  // Convert OKLab to OKLCH
  const L = l * 100
  const C = Math.sqrt(a * a + b_ * b_)
  let H = Math.atan2(b_, a) * 180 / Math.PI
  if (H < 0) H += 360

  return {
    l: Math.round(L * 100) / 100,
    c: Math.round(C * 1000) / 1000,
    h: Math.round(H * 100) / 100,
  }
}

/**
 * Generate CSS custom property string
 */
export function oklchToCSS(color: OKLCHColor): string {
  return `oklch(${color.l}% ${color.c} ${color.h})`
}

/**
 * Generate complete Tailwind palette from brand color
 */
export function generateBrandPalette(brandHex: string) {
  const brand = hexToOKLCH(brandHex)

  return {
    50: oklchToCSS({ l: 97, c: brand.c * 0.3, h: brand.h }),
    100: oklchToCSS({ l: 94, c: brand.c * 0.4, h: brand.h }),
    200: oklchToCSS({ l: 88, c: brand.c * 0.5, h: brand.h }),
    300: oklchToCSS({ l: 78, c: brand.c * 0.7, h: brand.h }),
    400: oklchToCSS({ l: 68, c: brand.c * 0.85, h: brand.h }),
    500: oklchToCSS({ l: 60, c: brand.c, h: brand.h }),
    600: oklchToCSS({ l: 52, c: brand.c * 0.95, h: brand.h }),
    700: oklchToCSS({ l: 44, c: brand.c * 0.85, h: brand.h }),
    800: oklchToCSS({ l: 32, c: brand.c * 0.7, h: brand.h }),
    900: oklchToCSS({ l: 22, c: brand.c * 0.5, h: brand.h }),
    950: oklchToCSS({ l: 12, c: brand.c * 0.3, h: brand.h }),
  }
}
```

```typescript
export interface ContrastResult {
  ratio: number
  AA: boolean      // 4.5:1 for normal text
  AALarge: boolean // 3:1 for large text
  AAA: boolean     // 7:1 for normal text
  AAALarge: boolean // 4.5:1 for large text

export function checkContrast(
  foreground: OKLCHColor,
  background: OKLCHColor
): ContrastResult {
  const ratio = calculateContrast(foreground, background)

  return {
    ratio: Math.round(ratio * 100) / 100,
    AA: ratio >= 4.5,
    AALarge: ratio >= 3,
    AAA: ratio >= 7,
    AAALarge: ratio >= 4.5,

}
```

## Migration from HSL/RGB

### Conversion Strategy

```typescript
// lib/color-migration.ts

/**
 * Convert HSL to OKLCH
 */
export function hslToOKLCH(h: number, s: number, l: number): OKLCHColor {
  // Convert HSL to RGB
  const c = (1 - Math.abs(2 * l / 100 - 1)) * s / 100
  const x = c * (1 - Math.abs((h / 60) % 2 - 1))
  const m = l / 100 - c / 2

  let r = 0, g = 0, b = 0

  if (h < 60) { r = c; g = x; b = 0 }
  else if (h < 120) { r = x; g = c; b = 0 }
  else if (h < 180) { r = 0; g = c; b = x }
  else if (h < 240) { r = 0; g = x; b = c }
  else if (h < 300) { r = x; g = 0; b = c }
  else { r = c; g = 0; b = x }

  r = (r + m) * 255
  g = (g + m) * 255
  b = (b + m) * 255

  // Convert RGB to hex then to OKLCH
  const hex = `#${Math.round(r).toString(16).padStart(2, '0')}${Math.round(g).toString(16).padStart(2, '0')}${Math.round(b).toString(16).padStart(2, '0')}`

  return hexToOKLCH(hex)
}

/**
 * Migrate existing color tokens
 */
export function migrateColorTokens(hslColors: Record<string, string>) {
  const oklchColors: Record<string, string> = {}

  for (const [name, hsl] of Object.entries(hslColors)) {
    // Parse HSL string: "221 83% 53%"
    const [h, s, l] = hsl.split(' ').map(v => parseFloat(v))
    const oklch = hslToOKLCH(h, s, l)
    oklchColors[name] = `${oklch.l}% ${oklch.c} ${oklch.h}`
  }

  return oklchColors
}
```

### Gradient Generation

```typescript
// Generate smooth OKLCH gradients
export function generateGradient(
  start: OKLCHColor,
  end: OKLCHColor,
  steps: number = 10
): string[] {
  const colors: string[] = []

  for (let i = 0; i < steps; i++) {
    const t = i / (steps - 1)

    const l = start.l + (end.l - start.l) * t
    const c = start.c + (end.c - start.c) * t
    const h = start.h + (end.h - start.h) * t

    colors.push(oklchToCSS({ l, c, h }))
  }

  return colors
}

// Usage
const gradient = generateGradient(
  { l: 60, c: 0.22, h: 270 }, // Blue
  { l: 70, c: 0.20, h: 330 }  // Pink
)

// Generate CSS gradient
const css = `linear-gradient(135deg, ${gradient.join(', ')})`
```

```css
  --primary: var(--primary-l) var(--primary-c) var(--primary-h);

  );
```

## Troubleshooting

### Browser DevTools Not Showing OKLCH

Some DevTools may convert OKLCH to RGB for display. Use `console.log(getComputedStyle(el).backgroundColor)` to verify actual format.

### Colors Look Different Than Expected

Ensure your monitor supports P3 color gamut. OKLCH can represent colors outside sRGB space.

### Fallback Not Working

Check PostCSS configuration and ensure `@csstools/postcss-oklab-function` is installed and configured.
