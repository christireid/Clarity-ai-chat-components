## Overview

Astro is a modern web framework that ships zero JavaScript by default, using "islands architecture" for interactive components.

### Key Benefits

- **Zero JavaScript**: Static by default, interactive on demand
- **Framework Agnostic**: Use React, Vue, Svelte, or all three
- **Islands Architecture**: Partial hydration for optimal performance
- **Content-First**: Perfect for documentation and content sites
- **Fast Builds**: Optimized for large sites

### When to Use Astro + Clarity Chat

- Documentation sites with AI assistant
- Marketing sites with chat support
- Content-heavy applications
- Sites requiring maximum performance
- Multi-page applications (MPA)

## Quick Start

```bash
# Create new Astro project
npm create astro@latest my-chat-app

# Choose options:
# - Template: Empty
# - TypeScript: Yes (strict)
# - Dependencies: Yes

cd my-chat-app

npx astro add react tailwind

# Install Clarity Chat

# Install peer dependencies
npm install framer-motion lucide-react clsx tailwind-merge

# Run development server
npm run dev
```

### React Chat Component

```tsx
// src/components/ChatIsland.tsx
import { useState } from 'react'
import { ClarityChatApp } from '@clarity-chat/react'
import type { Message } from '@clarity-chat/react'

export default function ChatIsland() {
  const [messages, setMessages] = useState<Message[]>([
    {
      id: '1',
      role: 'assistant',
      content: 'Hello! How can I help you today?',
    },
  ])

  const handleSendMessage = async (content: string) => {
    const userMessage: Message = {
      id: Date.now().toString(),
      role: 'user',
      content,
    }
    setMessages((prev) => [...prev, userMessage])

    try {
      const response = await fetch('/api/chat', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          messages: [...messages, userMessage],
        }),
      })

      const data = await response.json()
      const assistantMessage: Message = {
        id: (Date.now() + 1).toString(),
        role: 'assistant',
        content: data.content,
      }
      setMessages((prev) => [...prev, assistantMessage])
    } catch (error) {
      console.error('Error:', error)
    }
  }

  return (
    <div className="h-screen">
      <ClarityChatApp
        messages={messages}
        onSendMessage={handleSendMessage}
        placeholder="Ask anything..."
      />
    </div>
  )
}
```

## Client Directives

### Available Directives

```astro
<!-- Load immediately (blocks rendering) -->
<ChatIsland client:load />

<!-- Load when page is idle (after initial render) -->
<ChatIsland client:idle />

<!-- Load when visible in viewport -->
<ChatIsland client:visible />

<!-- Load on media query match -->
<ChatIsland client:media="(min-width: 768px)" />

<!-- Only render on client (skip SSR) -->
<ChatIsland client:only="react" />
```

### Choosing the Right Directive

```astro
---
// src/pages/docs.astro
import ChatAssistant from '../components/ChatAssistant'
import HeavyComponent from '../components/HeavyComponent'
---

<html>
  <body>
    <!-- Critical: Load immediately -->
    <ChatAssistant client:load />

    <!-- Below the fold: Load when visible -->
    <HeavyComponent client:visible />

    <!-- Mobile hidden: Load conditionally -->
    <DesktopChat client:media="(min-width: 1024px)" />
  </body>
</html>
```

## API Routes

### Chat API Endpoint

```typescript
// src/pages/api/chat.ts
import type { APIRoute } from 'astro'
import { openai } from '@ai-sdk/openai'
import { generateText } from 'ai'
import { z } from 'zod'

const requestSchema = z.object({
  messages: z.array(
    z.object({
      role: z.enum(['user', 'assistant', 'system']),
      content: z.string(),
    })
  ),
})

export const POST: APIRoute = async ({ request }) => {
  try {
    const body = await request.json()
    const { messages } = requestSchema.parse(body)

    const result = await generateText({
      model: openai('gpt-4o-mini'),
      messages,
      temperature: 0.7,
      maxTokens: 2000,
    })

    return new Response(
      JSON.stringify({
        content: result.text,
        usage: result.usage,
      }),
      {
        status: 200,
        headers: { 'Content-Type': 'application/json' },
      }
    )
  } catch (error) {
    console.error('Chat API error:', error)
    return new Response(JSON.stringify({ error: 'Internal server error' }), {
      status: 500,
      headers: { 'Content-Type': 'application/json' },
    })
  }
}
```

### Streaming API Endpoint

```typescript
// src/pages/api/chat/stream.ts
import type { APIRoute } from 'astro'
import { openai } from '@ai-sdk/openai'
import { streamText } from 'ai'

export const POST: APIRoute = async ({ request }) => {
  try {
    const { messages } = await request.json()

    const result = await streamText({
      model: openai('gpt-4o'),
      messages,
    })

    // Return streaming response
    return result.toAIStreamResponse()
  } catch (error) {
    console.error('Stream API error:', error)
    return new Response(JSON.stringify({ error: 'Stream failed' }), {
      status: 500,
      headers: { 'Content-Type': 'application/json' },
    })
  }
}
```

### RAG Search Endpoint

```typescript
// src/pages/api/docs-assistant.ts
import type { APIRoute } from 'astro'
import { getCollection } from 'astro:content'
import { openai } from '@ai-sdk/openai'
import { generateText, embed } from 'ai'

export const POST: APIRoute = async ({ request }) => {
  try {
    const { query, context, category } = await request.json()

    // Get all docs from collection
    const docs = await getCollection(category)

    // Generate query embedding
    const queryEmbedding = await embed({
      model: openai.embedding('text-embedding-3-small'),
      value: query,
    })

    // Find relevant docs (simplified - use vector DB in production)
    const relevantDocs = docs
      .filter((doc) => doc.data.title.toLowerCase().includes(query.toLowerCase()))
      .slice(0, 3)

    // Generate response with context
    const result = await generateText({
      model: openai('gpt-4o-mini'),
      messages: [
        {
          role: 'system',
          content: `You are a helpful documentation assistant. Answer questions based on the following context:\n\n${relevantDocs.map((doc) => `${doc.data.title}: ${doc.data.description}`).join('\n\n')}`,
        },
        {
          role: 'user',
          content: query,
        },
      ],
    })

    return new Response(
      JSON.stringify({
        content: result.text,
        sources: relevantDocs.map((doc) => ({
          title: doc.data.title,
          slug: doc.slug,
        })),
      }),
      {
        status: 200,
        headers: { 'Content-Type': 'application/json' },
      }
    )
  } catch (error) {
    console.error('Docs assistant error:', error)
    return new Response(JSON.stringify({ error: 'Failed to generate response' }), {
      status: 500,
      headers: { 'Content-Type': 'application/json' },
    })
  }
}
```

## Production Optimization

### Astro Configuration

```javascript
// astro.config.mjs
import { defineConfig } from 'astro/config'
import react from '@astrojs/react'
import tailwind from '@astrojs/tailwind'
import sitemap from '@astrojs/sitemap'

export default defineConfig({
  site: 'https://example.com',
  integrations: [
    react(),
    tailwind(),
    sitemap(),
  ],

  output: 'static', // or 'server' for SSR
  adapter: undefined, // Add adapter for SSR (vercel, netlify, node)

  vite: {
    build: {
      // Optimize chunks
      rollupOptions: {
        output: {
          manualChunks: {
            'react-vendor': ['react', 'react-dom'],
            'clarity-chat': ['@clarity-chat/react', '@clarity-chat/primitives'],
          },
        },
      },
    },
    optimizeDeps: {
      include: ['@clarity-chat/react', 'framer-motion'],
    },
  },

  // Build performance
  build: {
    inlineStylesheets: 'auto',
  },

  // Compress assets
  compressHTML: true,
})
```

1. **Zero JavaScript**: Static by default, interactive islands

3. **Framework Agnostic**: React, Vue, Svelte, or all three
4. **Content Collections**: Type-safe content management
5. **Fast Builds**: Optimized for large sites
6. **API Routes**: Built-in server endpoints

### Performance Benefits

- 90+ Lighthouse scores out of the box
- Minimal JavaScript shipped to client
- Optimal loading strategy with client directives
- Fast page loads with static generation
- Smart code splitting with islands
