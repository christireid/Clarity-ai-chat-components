## Overview

Next.js 15 App Router introduces powerful patterns for building AI chat applications:

- **Server Components**: Reduce JavaScript bundle size by rendering on the server
- **Streaming**: Progressive rendering with React Suspense
- **Route Handlers**: Type-safe API endpoints with validation
- **ISR & Caching**: Incremental Static Regeneration for optimal performance
- **Parallel Routes**: Complex layouts with independent loading states
- **Error Boundaries**: Granular error handling at multiple levels

## Server Component Strategy

### When to Use Server Components

**Default to Server Components for:**

- Static content and layouts
- Data fetching from databases or APIs
- Token-heavy operations (counting, validation)
- SEO-critical content
- Markdown rendering
- Initial page loads

```tsx
// app/chat/layout.tsx (Server Component by default)
import { Suspense } from 'react'
import { ChatSidebar } from '@/components/ChatSidebar'
import { getChatHistory } from '@/lib/db'

export default async function ChatLayout({ children }: { children: React.ReactNode }) {
  // Server-side data fetching
  const history = await getChatHistory()

  return (
    <div className="flex h-screen">
      {/* Server Component - No client JavaScript */}
      <Suspense fallback={<SidebarSkeleton />}>
        <ChatSidebar initialHistory={history} />
      </Suspense>

      <main className="flex-1 overflow-auto">{children}</main>
    </div>
  )
}
```

### When to Use Client Components

**Add 'use client' for:**

- Interactive UI (buttons, forms, inputs)
- Browser APIs (localStorage, geolocation)
- Event handlers (onClick, onChange)
- State management (useState, useReducer)
- Effects (useEffect, useLayoutEffect)
- Real-time features (WebSocket, SSE)
- Animation libraries (Framer Motion)

```tsx
'use client'

// app/components/ChatInput.tsx (Client Component)
import { useState } from 'react'
import { PillChatInput } from '@clarity-chat/react'
import { useChat } from 'ai/react'

export function ChatInput() {
  const [input, setInput] = useState('')
  const { append, isLoading } = useChat()

  const handleSubmit = async (value: string) => {
    await append({ role: 'user', content: value })
    setInput('')
  }

  return (
    <PillChatInput
      value={input}
      onChange={setInput}
      onSubmit={handleSubmit}
      isGenerating={isLoading}
      placeholder="Ask anything..."
    />
  )
}
```

### Component Boundary Pattern

Minimize client JavaScript by isolating interactive components:

```tsx
// app/chat/[id]/page.tsx (Server Component)
import { MessageRenderer } from '@/components/MessageRenderer' // Server Component
import { ChatInput } from '@/components/ChatInput' // Client Component
import { getConversation } from '@/lib/db'

export default async function ChatPage({ params }: { params: { id: string } }) {
  const conversation = await getConversation(params.id)

  return (
    <div className="flex flex-col h-full">
      {/* Server-rendered messages */}
      <div className="flex-1 overflow-auto p-4">
        {conversation.messages.map((message) => (
          <MessageRenderer key={message.id} message={message} />
        ))}
      </div>

      {/* Only this component is client-side */}
      <ChatInput />
    </div>
  )
}
```

### Server-Only Utilities

Use `server-only` package to prevent client imports:

```tsx
// lib/ai/server.ts
import 'server-only'
import { openai } from '@ai-sdk/openai'

export async function generateChatCompletion(messages: Message[]) {
  const response = await openai.chat.completions.create({
    model: 'gpt-4o',

  return response.choices[0].message
```

## Route Handlers

### Type-Safe Route Handler

```tsx
// app/api/chat/route.ts
import { NextRequest, NextResponse } from 'next/server'
import { z } from 'zod'
import { openai } from '@ai-sdk/openai'
import { generateText } from 'ai'

// Runtime validation schema
const messageSchema = z.object({
  messages: z.array(
    z.object({
      role: z.enum(['user', 'assistant', 'system']),
      content: z.string().min(1).max(4000),
    })
  ),
  model: z.enum(['gpt-4o', 'gpt-4o-mini']).optional().default('gpt-4o-mini'),
  temperature: z.number().min(0).max(2).optional().default(0.7),
})

export async function POST(request: NextRequest) {
  try {
    // Parse and validate request body
    const body = await request.json()
    const validation = messageSchema.safeParse(body)

    if (!validation.success) {
      return NextResponse.json(
        { error: 'Invalid request', details: validation.error.format() },
        { status: 400 }
      )
    }

    const { messages, model, temperature } = validation.data

    // Generate AI response
    const result = await generateText({
      model: openai(model),
      messages,
      temperature,
    })

    return NextResponse.json({
      content: result.text,
      usage: result.usage,
      finishReason: result.finishReason,
    })
  } catch (error) {
    console.error('Chat API error:', error)
    return NextResponse.json({ error: 'Internal server error' }, { status: 500 })
  }
}

// Enable Edge Runtime for lower latency
export const runtime = 'edge'
```

### Streaming Route Handler

```tsx
// app/api/chat/stream/route.ts
import { openai } from '@ai-sdk/openai'
import { streamText } from 'ai'
import { z } from 'zod'

const streamSchema = z.object({
  messages: z.array(
    z.object({
      role: z.enum(['user', 'assistant', 'system']),
      content: z.string(),
    })
  ),
})

export async function POST(req: Request) {
  const body = await req.json()
  const { messages } = streamSchema.parse(body)

  const result = await streamText({
    model: openai('gpt-4o'),
    messages,
    onFinish: async ({ text, usage }) => {
      // Log completion or save to database
      console.log('Stream finished:', { textLength: text.length, usage })
    },
  })

  return result.toAIStreamResponse()
}

export const runtime = 'edge'
```

### Route Handler with Caching

```tsx
// app/api/embeddings/route.ts
import { NextRequest, NextResponse } from 'next/server'
import { openai } from '@ai-sdk/openai'
import { embed } from 'ai'

export async function POST(request: NextRequest) {
  const { text } = await request.json()

  // Cache embeddings for 1 hour
  const cacheKey = `embedding:${Buffer.from(text).toString('base64')}`

  const result = await embed({
    model: openai.embedding('text-embedding-3-small'),
    value: text,
  })

  return NextResponse.json(
    { embedding: result.embedding },
    {
      headers: {
        'Cache-Control': 'public, s-maxage=3600, stale-while-revalidate=86400',
      },
    }
  )
}
```

## Caching Strategies

### ISR (Incremental Static Regeneration)

```tsx
// app/docs/[slug]/page.tsx
import { getDocumentation } from '@/lib/docs'

// Revalidate every hour
export const revalidate = 3600

export default async function DocumentationPage({ params }: { params: { slug: string } }) {
  const doc = await getDocumentation(params.slug)

  return (
    <article>
      <h1>{doc.title}</h1>
      <div dangerouslySetInnerHTML={{ __html: doc.content }} />
    </article>
  )
}

// Generate static paths at build time
export async function generateStaticParams() {
  const docs = await getAllDocumentation()
  return docs.map((doc) => ({ slug: doc.slug }))
}
```

### On-Demand Revalidation

```tsx
// app/api/revalidate/route.ts
import { NextRequest, NextResponse } from 'next/server'
import { revalidatePath, revalidateTag } from 'next/cache'

export async function POST(request: NextRequest) {
  const { path, tag } = await request.json()

  if (path) {
    revalidatePath(path)
    return NextResponse.json({ revalidated: true, path })
  }

  if (tag) {
    revalidateTag(tag)
    return NextResponse.json({ revalidated: true, tag })
  }

  return NextResponse.json({ error: 'Missing path or tag' }, { status: 400 })
}
```

Usage:

```tsx
// app/admin/docs/[id]/edit/actions.ts
'use server'

import { revalidatePath } from 'next/cache'

export async function updateDocumentation(id: string, data: UpdateData) {
  await db.documentation.update({ where: { id }, data })

  // Revalidate the documentation page
  revalidatePath(`/docs/${data.slug}`)
  revalidatePath('/docs')

  return { success: true }
}
```

### Tagged Caching

```tsx
// app/chat/[id]/page.tsx
import { unstable_cache } from 'next/cache'
import { getConversation } from '@/lib/db'

// Cache with tags for targeted revalidation
const getCachedConversation = unstable_cache(
  async (id: string) => getConversation(id),
  ['conversation'],
  {
    tags: (id) => [`conversation:${id}`],
    revalidate: 60, // 1 minute
  }
)

export default async function ChatPage({ params }: { params: { id: string } }) {
  const conversation = await getCachedConversation(params.id)
  return <ConversationView conversation={conversation} />
}
```

Revalidate specific conversation:

```tsx
// app/actions/conversation.ts
'use server'

import { revalidateTag } from 'next/cache'

export async function addMessage(conversationId: string, message: Message) {
  await db.message.create({ data: { conversationId, ...message } })

  // Revalidate only this conversation
  revalidateTag(`conversation:${conversationId}`)
}
```

### Request Memoization

Automatically deduplicates identical requests during a single render:

```tsx
// app/chat/[id]/layout.tsx
import { getConversation } from '@/lib/db'

export default async function ChatLayout({
  children,
  params,
}: {
  children: React.ReactNode
  params: { id: string }
}) {
  // First call
  const conversation = await getConversation(params.id)

  return (
    <div>
      <Header conversation={conversation} />
      {children}
    </div>
  )
}

// app/chat/[id]/page.tsx
import { getConversation } from '@/lib/db'

export default async function ChatPage({ params }: { params: { id: string } }) {
  // Second call - automatically memoized, no duplicate request
  const conversation = await getConversation(params.id)

  return <Messages messages={conversation.messages} />
}
```

```tsx
// app/chat/error.tsx

}
```

```tsx
import { getConversation } from '@/lib/db'

export async function generateMetadata({
  params,
}: {
  params: { id: string }
}): Promise<Metadata> {
  const conversation = await getConversation(params.id)

      title: 'Conversation Not Found',

    title: conversation.title || 'Chat Conversation',
    description: conversation.messages[0]?.content.slice(0, 160),

      description: conversation.messages[0]?.content.slice(0, 160),

    },
    twitter: {
      card: 'summary',
      title: conversation.title || 'Chat Conversation',
      description: conversation.messages[0]?.content.slice(0, 160),
    },
  }
}

export default async function ChatPage({ params }: { params: { id: string } }) {
  const conversation = await getConversation(params.id)
  return <ConversationView conversation={conversation} />
}
```

### Server Actions

```tsx
// app/actions/chat.ts
'use server'

import { revalidatePath, revalidateTag } from 'next/cache'
import { getServerSession } from 'next-auth'
import { db } from '@/lib/db'
import { generateAIResponse } from '@/lib/ai/client'
import { z } from 'zod'

const messageSchema = z.object({
  role: z.enum(['user', 'assistant', 'system']),
  content: z.string().min(1).max(4000),
})

export async function addMessage(
  conversationId: string,
  message: z.infer<typeof messageSchema>
) {
  const session = await getServerSession()

  if (!session) {
    throw new Error('Unauthorized')
  }

  // Validate input
  const validatedMessage = messageSchema.parse(message)

  // Save user message
  await db.message.create({
    data: {
      conversationId,
      ...validatedMessage,
    },
  })

  // Generate AI response
  const conversation = await db.conversation.findUnique({
    where: { id: conversationId },
    include: { messages: true },
  })

  const aiResponse = await generateAIResponse(conversation.messages)

  // Save AI response
  await db.message.create({
    data: {
      conversationId,
      role: 'assistant',
      content: aiResponse,
    },
  })

  // Revalidate the conversation page
  revalidatePath(`/chat/${conversationId}`)
  revalidateTag(`conversation:${conversationId}`)

  return { success: true }
}

export async function deleteConversation(conversationId: string) {
  const session = await getServerSession()

  if (!session) {
    throw new Error('Unauthorized')
  }

  await db.conversation.delete({
    where: { id: conversationId },
  })

  revalidatePath('/chat')
  return { success: true }
}
```

### API Route with Streaming

```tsx
// app/api/chat/route.ts
import { NextRequest } from 'next/server'
import { openai } from '@ai-sdk/openai'
import { streamText } from 'ai'
import { z } from 'zod'
import { getServerSession } from 'next-auth'

const chatSchema = z.object({
  messages: z.array(
    z.object({
      role: z.enum(['user', 'assistant', 'system']),
      content: z.string(),
    })
  ),
})

export async function POST(req: NextRequest) {
  const session = await getServerSession()

  if (!session) {
    return new Response('Unauthorized', { status: 401 })
  }

  try {
    const body = await req.json()
    const { messages } = chatSchema.parse(body)

    const result = await streamText({
      model: openai('gpt-4o'),
      messages,
      temperature: 0.7,
      maxTokens: 2000,
    })

    return result.toAIStreamResponse()
  } catch (error) {
    console.error('Chat API error:', error)
    return new Response('Internal Server Error', { status: 500 })
  }
}
```

### Middleware for Security

```tsx
// middleware.ts
import { NextRequest, NextResponse } from 'next/server'
import { getToken } from 'next-auth/jwt'

export async function middleware(request: NextRequest) {
  const token = await getToken({ req: request })
  const isAuthPage = request.nextUrl.pathname.startsWith('/login')
  const isApiRoute = request.nextUrl.pathname.startsWith('/api')

  // Redirect to login if not authenticated
  if (!token && !isAuthPage && !isApiRoute) {
    return NextResponse.redirect(new URL('/login', request.url))
  }

  // Redirect to chat if authenticated and on auth page
  if (token && isAuthPage) {
    return NextResponse.redirect(new URL('/chat', request.url))
  }

  const response = NextResponse.next()

  // Security headers
  response.headers.set('X-Content-Type-Options', 'nosniff')
  response.headers.set('X-Frame-Options', 'DENY')
  response.headers.set('X-XSS-Protection', '1; mode=block')
  response.headers.set('Referrer-Policy', 'strict-origin-when-cross-origin')

  return response
}

export const config = {
  matcher: ['/((?!_next/static|_next/image|favicon.ico).*)'],
}
```
