## Overview

Command palette search combines multiple algorithms to provide fast, accurate results even with typos or partial queries. The Clarity implementation uses a tiered matching strategy:

1. **Exact Match** - Highest priority (100 points)
2. **Prefix Match** - Starts with query (80+ points)
3. **Word Boundary Match** - Matches at word boundaries (60+ points)
4. **Substring Match** - Contains query anywhere (40+ points)
5. **Fuzzy Character Match** - Characters in order (20+ points)
6. **Phonetic Match** - Sounds similar (fallback)

<Callout type="info" title="Why Multiple Algorithms?">
Using multiple algorithms ensures users find what they need even with:
- **Typos**: "windwo" → "window" (Levenshtein distance)
- **Partial input**: "cw" → "ChatWindow" (fuzzy matching)
- **Case variations**: "CHAT" → "chat" (case-insensitive)
- **Word fragments**: "Window" in "ChatWindow" (word boundary detection)
</Callout>

## Fuzzy Matching Algorithms

### 1. Damerau-Levenshtein Distance

The **Damerau-Levenshtein** algorithm calculates the minimum number of edits (insertions, deletions, substitutions, and **transpositions**) needed to transform one string into another. This is more accurate than standard Levenshtein for typo detection because it accounts for common typing errors where adjacent characters are swapped.

<CodeBlock language="typescript" showLineNumbers title="lib/fuzzy-search.ts">
{`/**
 * Calculate Damerau-Levenshtein distance (includes transpositions)
 * More accurate than standard Levenshtein for typo detection
 *
 * Example:
 * - damerauLevenshtein('chat', 'caht') = 1 (transposition)
 * - damerauLevenshtein('chat', 'chta') = 2 (two operations needed)
 */
function damerauLevenshtein(a: string, b: string): number {
  if (a.length === 0) return b.length
  if (b.length === 0) return a.length
  if (a === b) return 0

  const aLen = a.length
  const bLen = b.length

  // Optimization: Use smaller array for short strings (< 32 chars)
  if (aLen <= 32 && bLen <= 32) {
    const matrix: number[][] = []

    // Initialize first row and column
    for (let i = 0; i <= bLen; i++) {
      matrix[i] = [i]
    }
    for (let j = 0; j <= aLen; j++) {
      matrix[0]![j] = j
    }

    // Fill matrix
    for (let i = 1; i <= bLen; i++) {
      for (let j = 1; j <= aLen; j++) {
        const cost = a.charAt(j - 1) === b.charAt(i - 1) ? 0 : 1

        matrix[i]![j] = Math.min(
          matrix[i - 1]![j]! + 1,      // deletion
          matrix[i]![j - 1]! + 1,      // insertion
          matrix[i - 1]![j - 1]! + cost // substitution
        )

        // TRANSPOSITION: Check if swapping adjacent chars helps
        if (
          i > 1 &&
          j > 1 &&
          a.charAt(j - 1) === b.charAt(i - 2) &&
          a.charAt(j - 2) === b.charAt(i - 1)
        ) {
          matrix[i]![j] = Math.min(
            matrix[i]![j]!,
            matrix[i - 2]![j - 2]! + cost
          )
        }
      }
    }

    return matrix[bLen]![aLen]!
  }

  // For longer strings (>32 chars), use space-optimized version
  let prevPrev: number[] = []
  let prev: number[] = []
  let curr: number[] = []

  for (let j = 0; j <= aLen; j++) {
    prev[j] = j
  }

  for (let i = 1; i <= bLen; i++) {
    curr[0] = i

    for (let j = 1; j <= aLen; j++) {
      const cost = a.charAt(j - 1) === b.charAt(i - 1) ? 0 : 1
      curr[j] = Math.min(
        prev[j]! + 1,
        curr[j - 1]! + 1,
        prev[j - 1]! + cost
      )
    }

    prevPrev = prev
    prev = curr
    curr = []
  }

  return prev[aLen]!
}`}
</CodeBlock>

#### Algorithm Complexity

| Metric | Complexity | Notes |
|--------|-----------|-------|
| **Time** | O(m × n) | Where m, n are string lengths |
| **Space** | O(min(m, n)) | Optimized with rolling arrays |
| **Best Case** | O(1) | Identical strings, early termination |

#### Example Usage

<CodeBlock language="typescript" showLineNumbers>
{`// Detect typos

damerauLevenshtein('search', 'serach')  // 1 (transposition)
damerauLevenshtein('command', 'comand') // 1 (deletion)

// Distance threshold determines match tolerance

  // Match!
}

// Reject: 'xyz' → 'search' (distance = 6)

  // No match
}`}

### 2. Jaro-Winkler Similarity

**Jaro-Winkler** is particularly effective for short strings and gives higher scores to strings that match from the beginning. It's commonly used in record linkage and duplicate detection.

<CodeBlock language="typescript" showLineNumbers title="lib/fuzzy-search.ts">
{`/**
 * Calculate Jaro-Winkler similarity (0-1, higher is better)
 *
 * Jaro-Winkler is optimized for:
 * - Short strings (names, identifiers)
 * - Prefix-heavy matching
 * - Record linkage tasks
 *
 * Example:
 * - jaroWinkler('martha', 'marhta') = 0.961 (high similarity)
 * - jaroWinkler('dixon', 'dicksonx') = 0.813 (moderate similarity)
 */
function jaroWinkler(s1: string, s2: string): number {
  // Handle edge cases
  if (!s1 || !s2) return 0
  if (s1 === s2) return 1

  const s1Len = s1.length
  const s2Len = s2.length

  // Calculate match window (half of max length - 1)
  const matchWindow = Math.floor(Math.max(s1Len, s2Len) / 2) - 1

  const s1Matches = new Array(s1Len).fill(false)
  const s2Matches = new Array(s2Len).fill(false)

  let matches = 0
  let transpositions = 0

  // Identify matches within window
  for (let i = 0; i < s1Len; i++) {
    const start = Math.max(0, i - matchWindow)
    const end = Math.min(i + matchWindow + 1, s2Len)

    for (let j = start; j < end; j++) {
      if (s2Matches[j] || s1[i] !== s2[j]) continue
      s1Matches[i] = true
      s2Matches[j] = true
      matches++
      break
    }
  }

  if (matches === 0) return 0

  // Count transpositions
  let k = 0
  for (let i = 0; i < s1Len; i++) {
    if (!s1Matches[i]) continue
    while (!s2Matches[k]) k++
    if (s1[i] !== s2[k]) transpositions++
    k++
  }

  // Calculate Jaro similarity
  const jaro = (
    (matches / s1Len) +
    (matches / s2Len) +
    ((matches - transpositions / 2) / matches)
  ) / 3

  // Apply Winkler prefix bonus (up to 4 chars)
  let prefix = 0
  for (let i = 0; i < Math.min(4, s1Len, s2Len); i++) {
    if (s1[i] === s2[i]) prefix++
    else break
  }

  const prefixScale = 0.1
  return jaro + (prefix * prefixScale * (1 - jaro))
}

/**
 * Use Jaro-Winkler for command matching
 */
function matchWithJaroWinkler(
  query: string,
  text: string,
  threshold = 0.8
): boolean {
  const similarity = jaroWinkler(
    query.toLowerCase(),
    text.toLowerCase()
  )
  return similarity >= threshold
}`}
</CodeBlock>

#### When to Use Jaro-Winkler

| Scenario | Recommended | Reason |
|----------|-------------|--------|
| **Short identifiers** | ✅ Yes | Optimized for names, codes |
| **Prefix-heavy matching** | ✅ Yes | Rewards matching prefixes |
| **Long text** | ❌ No | Levenshtein performs better |
| **Typo tolerance** | ⚠️ Moderate | Use Damerau-Levenshtein instead |

## Ranking and Scoring System

The scoring system uses a **tiered approach** where match quality determines the base score, and additional factors provide bonuses.

### Score Tiers

<CodeBlock language="typescript" showLineNumbers title="lib/fuzzy-search.ts">
{`export interface ScoreWeights {
  exact: number        // 100 - Perfect match
  startsWith: number   // 80  - Prefix match
  wordBoundary: number // 60  - Word boundary match
  contains: number     // 40  - Substring match
  fuzzy: number        // 20  - Fuzzy character match
}

const defaultWeights: ScoreWeights = {
  exact: 100,
  startsWith: 80,
  wordBoundary: 60,
  contains: 40,
  fuzzy: 20,
}

/**
 * Calculate match score with bonuses
 */
export function fuzzyScore(
  query: string,
  text: string,
  options?: FuzzySearchOptions
): {
  score: number
  matchType: 'exact' | 'startsWith' | 'wordBoundary' | 'contains' | 'fuzzy'
  indices: [number, number][]
} {
  const weights = { ...defaultWeights, ...options?.weights }
  const queryLower = query.toLowerCase()
  const textLower = text.toLowerCase()

  // 1. EXACT MATCH (100)
  if (textLower === queryLower) {
    return {
      score: weights.exact,
      matchType: 'exact',
      indices: [[0, text.length]],
    }
  }

  // 2. PREFIX MATCH (80 + length bonus)
  if (textLower.startsWith(queryLower)) {
    const lengthBonus = (queryLower.length / textLower.length) * 10
    return {
      score: weights.startsWith + lengthBonus,
      matchType: 'startsWith',
      indices: [[0, query.length]],
    }
  }

  // 3. WORD BOUNDARY MATCH (60 + position bonus)
  const boundaries = findWordBoundaries(text)
  for (const boundary of boundaries) {
    const fromBoundary = textLower.slice(boundary)
    if (fromBoundary.startsWith(queryLower)) {
      // Earlier boundaries score higher
      const positionBonus = (1 - boundary / text.length) * 10
      return {
        score: weights.wordBoundary + positionBonus,
        matchType: 'wordBoundary',
        indices: [[boundary, boundary + query.length]],
      }
    }
  }

  // 4. CONTAINS MATCH (40 + position bonus)
  const containsIndex = textLower.indexOf(queryLower)
  if (containsIndex !== -1) {
    // Earlier positions score higher
    const positionBonus = Math.max(0, 10 - containsIndex / 10)
    return {
      score: weights.contains + positionBonus,
      matchType: 'contains',
      indices: [[containsIndex, containsIndex + query.length]],
    }
  }

  // 5. FUZZY CHARACTER MATCH (20 + sequence bonus)
  let queryIndex = 0
  let matchedChars = 0
  let consecutiveBonus = 0
  let lastMatchIndex = -1
  const indices: [number, number][] = []

  for (let i = 0; i < textLower.length && queryIndex < queryLower.length; i++) {
    if (textLower[i] === queryLower[queryIndex]) {
      matchedChars++
      indices.push([i, i + 1])

      // Bonus for consecutive character matches
      if (lastMatchIndex === i - 1) {
        consecutiveBonus += 2
      }

      lastMatchIndex = i
      queryIndex++
    }
  }

  if (queryIndex === queryLower.length) {
    const matchRatio = matchedChars / queryLower.length
    const gapPenalty = indices.length > 1
      ? (indices[indices.length - 1]![0] - indices[0]![0] - queryLower.length) * 0.5
      : 0

    return {
      score: Math.max(1, weights.fuzzy + matchRatio * 10 + consecutiveBonus - gapPenalty),
      matchType: 'fuzzy',
      indices,
    }
  }

  // No match
  return { score: 0, matchType: 'fuzzy', indices: [] }
}`}
</CodeBlock>

### Scoring Examples

| Query | Text | Match Type | Base Score | Bonus | Final Score |
|-------|------|-----------|------------|-------|-------------|
| "chat" | "chat" | exact | 100 | 0 | **100** |
| "chat" | "ChatWindow" | startsWith | 80 | +4 (length) | **84** |

### Recency and Frequency Scoring

Beyond text matching, command palettes benefit from **usage-based scoring** that adapts to user behavior.

<CodeBlock language="typescript" showLineNumbers title="lib/command-search.ts">
{`interface CommandUsageStats {
  commandId: string
  lastUsed: Date
  useCount: number
  avgTimeSpent: number // milliseconds
}

/**
 * Apply recency and frequency boosts to search scores
 */
function applyUsageBoost(
  score: number,
  stats: CommandUsageStats | undefined,
  options: {
    recencyWeight?: number    // Default: 10
    frequencyWeight?: number  // Default: 5
    recencyDecay?: number     // Default: 7 days
  } = {}
): number {
  if (!stats) return score

  const recencyWeight = options.recencyWeight ?? 10
  const frequencyWeight = options.frequencyWeight ?? 5
  const recencyDecay = options.recencyDecay ?? 7 * 24 * 60 * 60 * 1000

  // RECENCY BOOST: Commands used recently rank higher
  const timeSinceUse = Date.now() - stats.lastUsed.getTime()
  const recencyFactor = Math.exp(-timeSinceUse / recencyDecay)
  const recencyBoost = recencyFactor * recencyWeight

  // FREQUENCY BOOST: Commands used often rank higher
  // Logarithmic scaling prevents over-boosting popular commands
  const frequencyBoost = Math.log10(stats.useCount + 1) * frequencyWeight

  return score + recencyBoost + frequencyBoost
}

/**
 * Example: Search with usage-based ranking
 */
function searchCommands(
  query: string,
  commands: Command[],
  usageStats: Map<string, CommandUsageStats>
): Command[] {
  return fuzzySearch(query, commands, (cmd) => [
    { field: 'label', value: cmd.label, weight: 2 },
    { field: 'description', value: cmd.description || '', weight: 1 },
  ])
    .map((result) => {
      const stats = usageStats.get(result.id)
      return {
        ...result,
        _score: applyUsageBoost(result._score, stats),
      }
    })
    .sort((a, b) => b._score - a._score)
}`}
</CodeBlock>

#### Recency Decay Curve

The recency boost follows an exponential decay:

```
Boost = e^(-t / decay) × weight

Where:
- t = time since last use (milliseconds)
- decay = half-life parameter (default: 7 days)
- weight = maximum boost value (default: 10)
```

| Time Since Use | Recency Boost (weight=10) |
|----------------|---------------------------|
| 1 hour | 9.98 |
| 1 day | 8.57 |
| 3 days | 5.98 |
| 7 days | 3.68 |
| 14 days | 1.35 |
| 30 days | 0.18 |

## Search Indexing for Performance

For large command lists (>500 items), building a search index dramatically improves performance.

### Inverted Index Implementation

<CodeBlock language="typescript" showLineNumbers title="lib/search-index.ts">
{`/**
 * Inverted index for fast command lookup
 * Maps tokens → command IDs
 */
class CommandSearchIndex {
  private index: Map<string, Set<string>> = new Map()
  private commands: Map<string, Command> = new Map()
  private tokenizer: (text: string) => string[]

  constructor(
    commands: Command[],
    tokenizer: (text: string) => string[] = defaultTokenizer
  ) {
    this.tokenizer = tokenizer
    this.buildIndex(commands)
  }

  /**
   * Build inverted index from commands
   * Time: O(n × m) where n = commands, m = avg tokens per command
   */
  private buildIndex(commands: Command[]): void {
    for (const command of commands) {
      this.commands.set(command.id, command)

      // Tokenize label and description
      const tokens = [
        ...this.tokenizer(command.label),
        ...this.tokenizer(command.description || ''),
        ...this.tokenizer(command.category || ''),
      ]

      // Add to inverted index
      for (const token of tokens) {
        if (!this.index.has(token)) {
          this.index.set(token, new Set())
        }
        this.index.get(token)!.add(command.id)
      }
    }
  }

  /**
   * Search using inverted index
   * Time: O(k × log(n)) where k = query tokens, n = matching commands
   */
  search(query: string, options?: FuzzySearchOptions): Command[] {
    const queryTokens = this.tokenizer(query)
    const candidateIds = new Set<string>()

    // Get candidate commands from index
    for (const token of queryTokens) {
      // Exact token match
      if (this.index.has(token)) {
        this.index.get(token)!.forEach((id) => candidateIds.add(id))
      }

      // Prefix match for partial tokens
      for (const [indexToken, commandIds] of this.index.entries()) {
        if (indexToken.startsWith(token)) {
          commandIds.forEach((id) => candidateIds.add(id))
        }
      }
    }

    // Convert candidate IDs to commands
    const candidates = Array.from(candidateIds)
      .map((id) => this.commands.get(id)!)
      .filter(Boolean)

    // Score and rank candidates
    return fuzzySearch(query, candidates, (cmd) => [
      { field: 'label', value: cmd.label, weight: 2 },
      { field: 'description', value: cmd.description || '', weight: 1 },
    ], options)
  }

  /**
   * Update index when commands change
   */
  updateCommands(commands: Command[]): void {
    this.index.clear()
    this.commands.clear()
    this.buildIndex(commands)
  }
}

/**
 * Default tokenizer: split on word boundaries, lowercase
 */
function defaultTokenizer(text: string): string[] {
  return text
    .toLowerCase()
    .split(/[\s\-_]+|(?=[A-Z])/)
    .filter((token) => token.length > 0)
}

/**
 * Usage example
 */
const searchIndex = new CommandSearchIndex(commands)

// Fast indexed search (100x faster for 1000+ commands)
const results = searchIndex.search('new chat', { threshold: 20 })
`}
</CodeBlock>

## Keyboard Shortcut Conflicts

Command palettes must handle keyboard shortcuts without conflicts. The system uses a **priority-based resolution** strategy.

### Conflict Detection

<CodeBlock language="typescript" showLineNumbers title="lib/keyboard-shortcuts.ts">
{`interface KeyboardShortcut {
  keys: string[]        // ['Cmd', 'K'] or ['Ctrl', 'Shift', 'P']
  commandId: string
  scope?: 'global' | 'editor' | 'panel'
  priority?: number     // Higher = takes precedence
}

/**
 * Detect keyboard shortcut conflicts
 */
function detectConflicts(
  shortcuts: KeyboardShortcut[]
): Map<string, KeyboardShortcut[]> {
  const conflicts = new Map<string, KeyboardShortcut[]>()

  // Group by normalized key combination
  for (const shortcut of shortcuts) {
    const key = normalizeShortcut(shortcut.keys)

    if (!conflicts.has(key)) {
      conflicts.set(key, [])
    }
    conflicts.get(key)!.push(shortcut)
  }

  // Filter to only actual conflicts (>1 command per key)
  return new Map(
    Array.from(conflicts.entries())
      .filter(([_, cmds]) => cmds.length > 1)
  )
}

/**
 * Normalize shortcut to canonical form
 * - Sort modifier keys: Ctrl < Alt < Shift < Meta
 * - Lowercase letters
 * - Platform-aware: Cmd → Meta on macOS
 */
function normalizeShortcut(keys: string[]): string {
  const modifierOrder = ['Ctrl', 'Alt', 'Shift', 'Meta', 'Cmd']

  const normalized = keys
    .map((key) => {
      // Platform normalization
      if (key === 'Cmd') return 'Meta'
      if (key === '⌘') return 'Meta'
      if (key === '⌥') return 'Alt'
      if (key === '⇧') return 'Shift'
      if (key === '⌃') return 'Ctrl'

      // Lowercase letters
      return key.length === 1 ? key.toLowerCase() : key
    })
    .sort((a, b) => {
      const aIdx = modifierOrder.indexOf(a)
      const bIdx = modifierOrder.indexOf(b)

      if (aIdx !== -1 && bIdx !== -1) return aIdx - bIdx
      if (aIdx !== -1) return -1
      if (bIdx !== -1) return 1
      return a.localeCompare(b)
    })

  return normalized.join('+')
}

/**
 * Resolve conflicts by priority and scope
 */
function resolveShortcut(
  event: KeyboardEvent,
  shortcuts: KeyboardShortcut[],
  activeScope: string = 'global'
): KeyboardShortcut | null {
  const pressedKeys = getPressedKeys(event)
  const normalizedPressed = normalizeShortcut(pressedKeys)

  // Find matching shortcuts
  const matches = shortcuts.filter((shortcut) => {
    const normalized = normalizeShortcut(shortcut.keys)
    return normalized === normalizedPressed
  })

  if (matches.length === 0) return null
  if (matches.length === 1) return matches[0]

  // Multiple matches - resolve by priority and scope
  const scopeMatches = matches.filter(
    (m) => !m.scope || m.scope === activeScope
  )

  const candidates = scopeMatches.length > 0 ? scopeMatches : matches

  // Sort by priority (descending)
  candidates.sort((a, b) => (b.priority ?? 0) - (a.priority ?? 0))

  return candidates[0]
}

/**
 * Get currently pressed keys from event
 */
function getPressedKeys(event: KeyboardEvent): string[] {
  const keys: string[] = []

  if (event.ctrlKey) keys.push('Ctrl')
  if (event.altKey) keys.push('Alt')
  if (event.shiftKey) keys.push('Shift')
  if (event.metaKey) keys.push('Meta')

  // Add the main key
  if (event.key && event.key !== 'Control' && event.key !== 'Alt' &&
      event.key !== 'Shift' && event.key !== 'Meta') {
    keys.push(event.key)
  }

  return keys
}`}
</CodeBlock>

## Text Highlighting

Highlighting matched portions of text helps users understand why a result was found.

### Basic Highlighting

<CodeBlock language="typescript" showLineNumbers title="lib/highlight.ts">
{`/**
 * Highlight matched portions of text with <mark> tags
 */
export function highlightMatches(
  query: string,
  text: string,
  className?: string
): string {
  if (!query || !text) return text

  const { indices } = fuzzyScore(query, text)

  if (indices.length === 0) return text

  // Merge overlapping indices
  const merged = mergeIndices(indices)

  // Build highlighted string
  const markClass = className ? \` class="\${className}"\` : ''
  let result = ''
  let lastEnd = 0

  for (const [start, end] of merged) {
    result += text.slice(lastEnd, start)
    result += \`<mark\${markClass}>\${text.slice(start, end)}</mark>\`
    lastEnd = end
  }

  result += text.slice(lastEnd)
  return result
}

/**
 * Merge overlapping or adjacent indices
 */
function mergeIndices(indices: [number, number][]): [number, number][] {
  if (indices.length === 0) return []

  const sorted = [...indices].sort((a, b) => a[0] - b[0])
  const merged: [number, number][] = [sorted[0]]

  for (let i = 1; i < sorted.length; i++) {
    const [start, end] = sorted[i]
    const last = merged[merged.length - 1]

    // Overlapping or adjacent
    if (start <= last[1]) {
      last[1] = Math.max(last[1], end)
    } else {
      merged.push([start, end])
    }
  }

  return merged
}`}
</CodeBlock>

### React Component Highlighting

For React applications, render highlighted text as components:

<CodeBlock language="typescript" showLineNumbers title="components/HighlightedText.tsx">
{`import React from 'react'

interface HighlightedTextProps {
  query: string
  text: string
  className?: string
  highlightClassName?: string
}

/**
 * React component for highlighted text
 */
export function HighlightedText({
  query,
  text,
  className,
  highlightClassName = 'bg-yellow-200 dark:bg-yellow-900',
}: HighlightedTextProps) {
  if (!query || !text) {
    return <span className={className}>{text}</span>
  }

  const { indices } = fuzzyScore(query, text)

  if (indices.length === 0) {
    return <span className={className}>{text}</span>
  }

  const merged = mergeIndices(indices)
  const parts: React.ReactNode[] = []
  let lastEnd = 0

  merged.forEach(([start, end], index) => {
    // Non-highlighted text before match
    if (start > lastEnd) {
      parts.push(
        <React.Fragment key={\`text-\${index}\`}>
          {text.slice(lastEnd, start)}
        </React.Fragment>
      )
    }

    // Highlighted match
    parts.push(
      <mark
        key={\`mark-\${index}\`}
        className={highlightClassName}
      >
        {text.slice(start, end)}
      </mark>
    )

    lastEnd = end
  })

  // Remaining text
  if (lastEnd < text.length) {
    parts.push(
      <React.Fragment key="text-end">
        {text.slice(lastEnd)}
      </React.Fragment>
    )
  }

  return <span className={className}>{parts}</span>
}

/**
 * Usage in command palette
 */
function CommandItem({ command, query }: { command: Command; query: string }) {
  return (
    <button className="command-item">
      <HighlightedText
        query={query}
        text={command.label}
        className="font-medium"
        highlightClassName="bg-brand-200 text-brand-900 dark:bg-brand-800 dark:text-brand-100"
      />
      {command.description && (
        <HighlightedText
          query={query}
          text={command.description}
          className="text-sm text-muted-foreground"
          highlightClassName="bg-yellow-200 dark:bg-yellow-900"
        />
      )}
    </button>
  )
}`}

function generateCommands(count: number): Command[] {
  const categories = ['Actions', 'Navigation', 'AI', 'Tools', 'Settings']

  const nouns = ['Chat', 'Window', 'Message', 'File', 'Panel', 'View', 'Model']

    label: \`\${verbs[i % verbs.length]} \${nouns[i % nouns.length]}\`,

    category: categories[i % categories.length],
  }))
}

/**
 * Benchmark search performance
 */
function benchmarkSearch(
  name: string,
  searchFn: (query: string) => any[],
  queries: string[],
  iterations = 1000
) {
  const times: number[] = []

  for (let i = 0; i < iterations; i++) {
    const query = queries[i % queries.length]
    const start = performance.now()
    searchFn(query)
    const end = performance.now()
    times.push(end - start)
  }

  const avg = times.reduce((a, b) => a + b, 0) / times.length
  const p50 = times.sort((a, b) => a - b)[Math.floor(times.length * 0.5)]
  const p95 = times.sort((a, b) => a - b)[Math.floor(times.length * 0.95)]
  const p99 = times.sort((a, b) => a - b)[Math.floor(times.length * 0.99)]

  console.log(\`\${name}:\`)
  console.log(\`  Average: \${avg.toFixed(3)} ms\`)
  console.log(\`  P50: \${p50.toFixed(3)} ms\`)
  console.log(\`  P95: \${p95.toFixed(3)} ms\`)
  console.log(\`  P99: \${p99.toFixed(3)} ms\`)
}

const queries = [
  'new chat',
  'window',
  'search',

## Real-World Implementation

### Complete Command Palette with Advanced Search

<CodeBlock language="typescript" showLineNumbers title="components/CommandPalette.tsx">
{`'use client'

import { useState, useMemo, useEffect } from 'react'
import { useDebounce } from '@/hooks/use-debounce'
import { fuzzySearch, highlightMatchesAdvanced } from '@/lib/fuzzy-search'
import { CommandSearchIndex } from '@/lib/search-index'

interface Command {
  id: string
  label: string
  description?: string
  category?: string
  shortcut?: string[]
  onSelect: () => void
}

export function AdvancedCommandPalette({
  commands,
  open,
  onClose,
}: {
  commands: Command[]
  open: boolean
  onClose: () => void
}) {
  const [search, setSearch] = useState('')
  const [selectedIndex, setSelectedIndex] = useState(0)

  // Debounce search for better performance
  const debouncedSearch = useDebounce(search, 150)

  // Build search index for large command sets
  const searchIndex = useMemo(
    () => new CommandSearchIndex(commands),
    [commands]
  )

  // Filter commands with fuzzy search
  const filteredCommands = useMemo(() => {
    if (!debouncedSearch) return commands

    // Use indexed search for performance
    return searchIndex.search(debouncedSearch, {
      threshold: 20,
      limit: 50,
    })
  }, [commands, debouncedSearch, searchIndex])

  // Reset selection when results change
  useEffect(() => {
    setSelectedIndex(0)
  }, [filteredCommands])

  // Keyboard navigation
  useEffect(() => {
    if (!open) return

    const handleKeyDown = (e: KeyboardEvent) => {
      switch (e.key) {
        case 'ArrowDown':
          e.preventDefault()
          setSelectedIndex((i) =>
            (i + 1) % filteredCommands.length
          )
          break
        case 'ArrowUp':
          e.preventDefault()
          setSelectedIndex((i) =>
            (i - 1 + filteredCommands.length) % filteredCommands.length
          )
          break
        case 'Enter':
          e.preventDefault()
          filteredCommands[selectedIndex]?.onSelect()
          onClose()
          break
        case 'Escape':
          e.preventDefault()
          onClose()
          break
      }
    }

    window.addEventListener('keydown', handleKeyDown)
    return () => window.removeEventListener('keydown', handleKeyDown)
  }, [open, filteredCommands, selectedIndex, onClose])

  return (
    <div className="command-palette">
      <input
        type="text"
        value={search}
        onChange={(e) => setSearch(e.target.value)}
        placeholder="Search commands..."
        className="search-input"
      />

      <div className="results">
        {filteredCommands.map((command, index) => (
          <CommandResult
            key={command.id}
            command={command}
            query={debouncedSearch}
            selected={index === selectedIndex}
            onClick={() => {
              command.onSelect()
              onClose()
            }}
          />
        ))}
      </div>

      <div className="footer">
        {filteredCommands.length} {filteredCommands.length === 1 ? 'command' : 'commands'}
      </div>
    </div>
  )
}

/**
 * Individual command result with highlighting
 */
function CommandResult({
  command,
  query,
  selected,
  onClick,
}: {
  command: Command
  query: string
  selected: boolean
  onClick: () => void
}) {
  const highlightedLabel = useMemo(
    () =>
      highlightMatchesAdvanced(
        query,
        command.label,
        (text, isMatch, index) =>
          isMatch ? (
            <mark key={index} className="bg-yellow-200 dark:bg-yellow-900">
              {text}
            </mark>
          ) : (
            <span key={index}>{text}</span>
          )
      ),
    [query, command.label]
  )

  return (
    <button
      onClick={onClick}
      className={\`command-result \${selected ? 'selected' : ''}\`}
    >
      <div className="label">{highlightedLabel}</div>
      {command.description && (
        <div className="description">{command.description}</div>
      )}
    </button>
  )
}`}
</CodeBlock>
