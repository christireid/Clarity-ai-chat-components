## Testing User Flows

### Example 1: Send Message Flow

```typescript
// tests/e2e/message-send.spec.ts
import { test, expect } from '@playwright/test'

test.describe('Send Message Flow', () => {
  test('user can send a text message', async ({ page }) => {
    // Navigate to chat
    await page.goto('/chat')

    // Wait for chat to be ready
    await expect(page.locator('[data-testid="chat-input"]')).toBeVisible()

    // Type message
    const input = page.locator('[data-testid="chat-input"]')
    await input.fill('Hello, how can you help me today?')

    // Send message
    await page.click('[data-testid="send-button"]')

    // Verify message appears in chat
    await expect(
      page.locator('[data-testid="message"]').last()
    ).toContainText('Hello, how can you help me today?')

    // Verify input is cleared
    await expect(input).toHaveValue('')

    // Verify send button is disabled (waiting for response)
    await expect(page.locator('[data-testid="send-button"]')).toBeDisabled()
  })

  test('user can send message with Enter key', async ({ page }) => {
    await page.goto('/chat')

    const input = page.locator('[data-testid="chat-input"]')
    await input.fill('Test message')

    // Press Enter to send
    await input.press('Enter')

    // Verify message was sent
    await expect(
      page.locator('[data-testid="message"]').last()
    ).toContainText('Test message')
  })

  test('Shift+Enter creates new line instead of sending', async ({ page }) => {
    await page.goto('/chat')

    const input = page.locator('[data-testid="chat-input"]')
    await input.fill('Line 1')
    await input.press('Shift+Enter')
    await input.type('Line 2')

    // Verify input contains both lines
    const value = await input.inputValue()
    expect(value).toContain('Line 1')
    expect(value).toContain('Line 2')

    // Verify message was not sent
    await expect(page.locator('[data-testid="message"]')).toHaveCount(0)
  })

  test('cannot send empty message', async ({ page }) => {
    await page.goto('/chat')

    const sendButton = page.locator('[data-testid="send-button"]')

    // Send button should be disabled when empty
    await expect(sendButton).toBeDisabled()

    // Type and clear
    const input = page.locator('[data-testid="chat-input"]')
    await input.fill('Test')
    await input.clear()

    // Still disabled
    await expect(sendButton).toBeDisabled()
  })
})
```

### Example 4: Page Object Model Pattern

```typescript
// tests/page-objects/ChatPage.ts
import { type Page, type Locator, expect } from '@playwright/test'

export class ChatPage {
  readonly page: Page
  readonly messageInput: Locator
  readonly sendButton: Locator
  readonly messages: Locator
  readonly typingIndicator: Locator
  readonly errorMessage: Locator
  readonly retryButton: Locator

  constructor(page: Page) {
    this.page = page
    this.messageInput = page.locator('[data-testid="chat-input"]')
    this.sendButton = page.locator('[data-testid="send-button"]')
    this.messages = page.locator('[data-testid="message"]')
    this.typingIndicator = page.locator('[data-testid="typing-indicator"]')
    this.errorMessage = page.locator('[data-testid="error-message"]')
    this.retryButton = page.locator('[data-testid="retry-button"]')
  }

  async goto() {
    await this.page.goto('/chat')
    await this.waitForReady()
  }

  async waitForReady() {
    await expect(this.messageInput).toBeVisible()
    await expect(this.sendButton).toBeVisible()
  }

  async sendMessage(text: string) {
    await this.messageInput.fill(text)
    await this.sendButton.click()
  }

  async sendMessageWithEnter(text: string) {
    await this.messageInput.fill(text)
    await this.messageInput.press('Enter')
  }

  async waitForResponse() {
    await expect(this.typingIndicator).toBeVisible()
    await expect(this.typingIndicator).not.toBeVisible({ timeout: 30000 })
  }

  async getLastMessage() {
    return this.messages.last()
  }

  async getMessageCount() {
    return await this.messages.count()
  }

  async clearConversation() {
    await this.page.click('[data-testid="clear-button"]')
    await this.page.click('[data-testid="confirm-clear"]')
  }

  async verifyMessageAppears(text: string) {
    await expect(this.getLastMessage()).toContainText(text)
  }

  async retryLastMessage() {
    await this.retryButton.click()
  }
}

// Usage in tests
test('complex user flow with Page Object', async ({ page }) => {
  const chatPage = new ChatPage(page)
  await chatPage.goto()

  await chatPage.sendMessage('Hello')
  await chatPage.waitForResponse()
  await chatPage.verifyMessageAppears('Hi there!')

  await chatPage.sendMessage('Tell me a joke')
  await chatPage.waitForResponse()

  expect(await chatPage.getMessageCount()).toBe(4) // 2 user, 2 assistant
})
```

### Example 5: Multi-Step Workflow Test

```typescript
// tests/e2e/complete-workflow.spec.ts
import { test, expect } from '../fixtures/test-helpers'
import { ChatPage } from '../page-objects/ChatPage'

test.describe('Complete Chat Workflow', () => {
  test('full conversation with multiple features', async ({ page, mockAPI }) => {
    const chatPage = new ChatPage(page)
    await chatPage.goto()

    // Step 1: Initial greeting
    await mockAPI.mockChatResponse({
      content: 'Hello! How can I help you today?'
    })
    await chatPage.sendMessage('Hi')
    await chatPage.waitForResponse()
    await chatPage.verifyMessageAppears('Hello! How can I help you today?')

    // Step 2: Ask question that triggers tool use
    await mockAPI.mockToolUse({
      toolName: 'web_search',
      toolInput: { query: 'React hooks' },
      toolResult: { results: ['useState', 'useEffect', 'useContext'] },
      finalResponse: 'React hooks include useState, useEffect, and useContext.',
    })
    await chatPage.sendMessage('What are React hooks?')

    // Verify tool execution
    await expect(page.locator('[data-testid="tool-card"]')).toBeVisible()
    await expect(
      page.locator('[data-testid="tool-name"]')
    ).toContainText('web_search')

    // Wait for final response
    await chatPage.waitForResponse()
    await chatPage.verifyMessageAppears('React hooks include')

    // Step 3: Follow-up question
    await mockAPI.mockChatResponse({
      content: 'useState is used for managing component state.',
    })
    await chatPage.sendMessage('Tell me more about useState')
    await chatPage.waitForResponse()
    await chatPage.verifyMessageAppears('managing component state')

    // Step 4: Verify message history
    const messageCount = await chatPage.getMessageCount()
    expect(messageCount).toBe(6) // 3 user messages + 3 assistant responses

    // Step 5: Clear conversation
    await chatPage.clearConversation()
    expect(await chatPage.getMessageCount()).toBe(0)
  })
})
```

### Accessibility Test Examples

```typescript
// tests/e2e/accessibility.spec.ts
import { test, expect } from '@playwright/test'
import { injectAxe, checkA11y, getViolations } from 'axe-playwright'

test.describe('Accessibility Tests', () => {
  test('chat interface has no accessibility violations', async ({ page }) => {
    await page.goto('/chat')

    // Inject axe-core
    await injectAxe(page)

    // Run accessibility check
    await checkA11y(page, undefined, {
      detailedReport: true,
      detailedReportOptions: {
        html: true,
      },
    })
  })

  test('keyboard navigation works correctly', async ({ page }) => {
    await page.goto('/chat')

    // Tab through interactive elements
    await page.keyboard.press('Tab')

    // Verify focus is on input
    await expect(page.locator('[data-testid="chat-input"]')).toBeFocused()

    // Tab to send button
    await page.keyboard.press('Tab')
    await expect(page.locator('[data-testid="send-button"]')).toBeFocused()

    // Tab to settings
    await page.keyboard.press('Tab')
    await expect(page.locator('[data-testid="settings-button"]')).toBeFocused()
  })

  test('screen reader announcements are present', async ({ page }) => {
    await page.goto('/chat')

    // Check for ARIA live regions
    const liveRegion = page.locator('[aria-live="polite"]')
    await expect(liveRegion).toBeAttached()

    // Send message
    await page.fill('[data-testid="chat-input"]', 'Test')
    await page.click('[data-testid="send-button"]')

    // Verify status announcement
    await expect(liveRegion).toContainText(/sending|sent/i)
  })

  test('all interactive elements have accessible names', async ({ page }) => {
    await page.goto('/chat')

    // Check buttons
    const sendButton = page.locator('[data-testid="send-button"]')
    await expect(sendButton).toHaveAttribute('aria-label', /.+/)

    const clearButton = page.locator('[data-testid="clear-button"]')
    await expect(clearButton).toHaveAttribute('aria-label', /.+/)

    // Check input
    const input = page.locator('[data-testid="chat-input"]')
    await expect(input).toHaveAttribute('aria-label', /.+/)
  })

  test('color contrast meets WCAG AA standards', async ({ page }) => {
    await page.goto('/chat')
    await injectAxe(page)

    // Check specifically for color contrast violations
    const violations = await getViolations(page, undefined, {
      rules: {
        'color-contrast': { enabled: true },
      },
    })

    expect(violations).toHaveLength(0)
  })

  test('focus indicators are visible', async ({ page }) => {
    await page.goto('/chat')

    // Tab to input
    await page.keyboard.press('Tab')

    // Get computed styles
    const input = page.locator('[data-testid="chat-input"]')
    const outline = await input.evaluate((el) => {
      const styles = window.getComputedStyle(el, ':focus-visible')
      return styles.outline
    })

    // Verify outline exists
    expect(outline).not.toBe('none')
    expect(outline).not.toBe('')
  })

  test('error messages are announced', async ({ page, mockAPI }) => {
    await page.goto('/chat')

    // Mock error
    await mockAPI.mockChatError(500, 'Server error')

    // Send message
    await page.fill('[data-testid="chat-input"]', 'Test')
    await page.click('[data-testid="send-button"]')

    // Check error is in live region
    const errorRegion = page.locator('[role="alert"]')
    await expect(errorRegion).toBeVisible()
    await expect(errorRegion).toContainText('Failed to send message')
  })
})
```

## Best Practices

### 1. Use Data Test IDs

```tsx
// ✅ Good: Stable selectors
<button data-testid="send-button">Send</button>

// ❌ Bad: Fragile selectors
<button className="btn btn-primary">Send</button>
```

### 2. Wait for Conditions, Not Timeouts

```typescript
// ✅ Good: Wait for condition
await expect(page.locator('[data-testid="message"]')).toBeVisible()

// ❌ Bad: Arbitrary timeout
await page.waitForTimeout(3000)
```

### 3. Use Page Object Model

```typescript
// ✅ Good: Reusable page objects
const chatPage = new ChatPage(page)
await chatPage.sendMessage('Hello')

// ❌ Bad: Duplicate selectors
await page.fill('[data-testid="chat-input"]', 'Hello')
await page.click('[data-testid="send-button"]')
```

### 4. Test User Flows, Not Implementation

```typescript
// ✅ Good: User-centric test
test('user can send and receive messages', async ({ page }) => {
  await chatPage.goto()
  await chatPage.sendMessage('Hello')
  await chatPage.verifyMessageAppears('Hello')
  await chatPage.waitForResponse()
  await chatPage.verifyMessageAppears('Hi there!')
})

// ❌ Bad: Testing implementation details
test('message state updates correctly', async ({ page }) => {
  await page.evaluate(() => {
    window.__store__.dispatch({ type: 'ADD_MESSAGE', payload: {} })
  })
})
```

### 5. Keep Tests Independent

```typescript
// ✅ Good: Independent tests
test('test 1', async ({ page }) => {
  const chatPage = new ChatPage(page)
  await chatPage.goto()
  // Test logic
})

test('test 2', async ({ page }) => {
  const chatPage = new ChatPage(page)
  await chatPage.goto()
  // Test logic - doesn't depend on test 1
})

// ❌ Bad: Tests depend on order
test('setup', async ({ page }) => {
  // Creates data
})

test('use data', async ({ page }) => {
  // Depends on previous test
})
```

### 6. Use Fixtures for Common Setup

```typescript
// ✅ Good: Reusable fixtures
const test = base.extend<{ authenticatedPage: Page }>({
  authenticatedPage: async ({ page }, use) => {
    await authenticate(page)
    await use(page)
  },
})

test('protected feature', async ({ authenticatedPage }) => {
  // Already authenticated
})

// ❌ Bad: Duplicate setup
test('test 1', async ({ page }) => {
  await authenticate(page)
  // Test
})

test('test 2', async ({ page }) => {
  await authenticate(page)
  // Test
})
```

### 7. Handle Flaky Tests

```typescript
// ✅ Good: Robust waiting
await expect(async () => {
  const text = await page.locator('[data-testid="status"]').textContent()
  expect(text).toBe('Complete')
}).toPass({
  intervals: [1000, 2000, 5000],
  timeout: 10000,
})

// ❌ Bad: Fixed retry without understanding
test.describe(() => {
  test.beforeEach(async ({ page }) => {
    // Just retry 3 times
    test.setTimeout(90000)
  })
})
```

### 8. Clean Up Resources

```typescript
// ✅ Good: Proper cleanup
test('file upload', async ({ page }) => {
  const filePath = await createTempFile()

  try {
    await page.setInputFiles('[data-testid="file-input"]', filePath)
    // Test logic
  } finally {
    await deleteTempFile(filePath)
  }
})

// ❌ Bad: No cleanup
test('file upload', async ({ page }) => {
  const filePath = await createTempFile()
  await page.setInputFiles('[data-testid="file-input"]', filePath)
  // File never deleted
})
```

## Troubleshooting

### Common Issues

1. **Flaky tests**: Add proper waits, use `toPass()` for retries
2. **Timeout errors**: Increase timeout for slow operations
3. **Element not found**: Check if element is in viewport, use better selectors
4. **Visual diff failures**: Update snapshots or fix styling issues
