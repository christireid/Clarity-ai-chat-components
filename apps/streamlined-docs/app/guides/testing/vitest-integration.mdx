```typescript
describe('AudioRecorder - Browser API Mocking', () => {
  // Mock MediaRecorder
  class MockMediaRecorder {
    state: 'inactive' | 'recording' | 'paused' = 'inactive'

      })

      return ['audio/webm', 'audio/wav'].includes(mimeType)

  beforeEach(() => {
```

### When to Use Snapshots

**Good Use Cases:**
- Testing complex rendered output structures
- Detecting unintended UI changes
- Regression testing for styled components
- API response structures

**Avoid Snapshots For:**
- Dynamic content (timestamps, IDs)
- Frequently changing components
- Simple boolean logic
- Testing behavior (use assertions instead)

### Snapshot Best Practices

```typescript
// Use serializers for consistent snapshots
expect.addSnapshotSerializer({
  test: (val) => val instanceof Date,
  serialize: (val) => `Date(${val.toISOString()})`,
})

// Exclude dynamic values

  const { container } = render(
    <UserProfile
      id="123"
      createdAt={new Date()}
      name="John"
    />
  )

  // Remove dynamic attributes before snapshot
  const element = container.firstChild as HTMLElement
  element.removeAttribute('data-testid')
  element.querySelector('[data-timestamp]')?.removeAttribute('data-timestamp')

  expect(element).toMatchSnapshot()
})
```

## Testing Async Components & Suspense

### Testing Async Data Loading

```typescript
import { Suspense } from 'react'

function AsyncUserProfile({ userId }: { userId: string }) {
  const user = use(fetchUser(userId)) // React 19 'use' hook
  return <div>{user.name}</div>
}

describe('AsyncUserProfile', () => {
  it('shows loading state then data', async () => {
    render(

    // Wait for data to load
    await waitFor(() => {
      expect(screen.getByText('John Doe')).toBeInTheDocument()
    })

    expect(screen.queryByText('Loading...')).not.toBeInTheDocument()

    server.use(

        return res(ctx.status(500))
      })

    render(
      <ErrorBoundary fallback={<div>Error loading user</div>}>
        <Suspense fallback={<div>Loading...</div>}>
          <AsyncUserProfile userId="123" />
        </Suspense>
      </ErrorBoundary>
    )

    await waitFor(() => {
      expect(screen.getByText('Error loading user')).toBeInTheDocument()
    })
  })
})
```

### Coverage Threshold Script

```javascript
// scripts/check-coverage.js
const fs = require('fs')
const path = require('path')

const coveragePath = path.join(__dirname, '../coverage/coverage-summary.json')
const coverage = JSON.parse(fs.readFileSync(coveragePath, 'utf-8'))

const thresholds = {
  lines: 85,
  functions: 85,
  branches: 80,
  statements: 85,
}

const total = coverage.total

let failed = false

for (const [metric, threshold] of Object.entries(thresholds)) {
  const actual = total[metric].pct

  if (actual < threshold) {
    console.error(`❌ ${metric} coverage ${actual}% is below threshold ${threshold}%`)
    failed = true
  } else {
    console.log(`✅ ${metric} coverage ${actual}% meets threshold ${threshold}%`)
  }
}

if (failed) {
  process.exit(1)
}

console.log('✅ All coverage thresholds met!')
```

```typescript
  class MockMediaStream {

    id = 'mock-stream'
    private tracks = [{
      stop: vi.fn(),
      kind: 'audio',
      enabled: true,

    ondataavailable: ((event: any) => void) | null = null
    onstop: ((event: any) => void) | null = null

      this.state = 'recording'
      queueMicrotask(() => {
        if (this.onstart) this.onstart(new Event('start'))
      })
    }

    stop() {
      this.state = 'inactive'
      if (this.ondataavailable) {
        const blob = new Blob(['audio'], { type: 'audio/webm' })
        this.ondataavailable({ data: blob })
      }
      queueMicrotask(() => {
        if (this.onstop) this.onstop(new Event('stop'))
      })
    }
  }

    global.MediaRecorder = MockMediaRecorder as any
    global.navigator.mediaDevices = {
      getUserMedia: vi.fn().mockResolvedValue(new MockMediaStream()),
    } as any
  })
```

## Best Practices

### General Testing Principles

1. **Test behavior, not implementation** - Focus on what the component does, not how it does it
2. **Write deterministic tests** - Tests should always produce the same result
3. **Keep tests focused** - One test should verify one thing
4. **Use meaningful test descriptions** - Test names should clearly state what is being tested
5. **Avoid testing internal state** - Test public API and rendered output
6. **Mock external dependencies** - Keep tests fast and isolated

### Performance Tips

```typescript
// Use userEvent.setup({ delay: null }) for faster tests
const user = userEvent.setup({ delay: null })

// Use queueMicrotask instead of setTimeout in mocks
queueMicrotask(() => callback())

// Disable animations in tests
vi.mock('framer-motion', ...)

// Use happy-dom instead of jsdom for faster tests
environment: 'happy-dom'
```

### Common Pitfalls

```typescript
// ❌ Bad: Testing implementation details
expect(component.state.count).toBe(1)

// ✅ Good: Testing rendered output
expect(screen.getByText('Count: 1')).toBeInTheDocument()

// ❌ Bad: Not waiting for async updates
user.click(button)
expect(onSubmit).toHaveBeenCalled() // May fail

// ✅ Good: Wait for async operations
await user.click(button)
await waitFor(() => {
  expect(onSubmit).toHaveBeenCalled()
})

// ❌ Bad: Not cleaning up side effects
it('test', () => {
  const spy = vi.spyOn(console, 'log')
  // ... test code
  // Spy is never restored
})

// ✅ Good: Restore mocks after test
it('test', () => {
  const spy = vi.spyOn(console, 'log')
  // ... test code
  spy.mockRestore()
})
```
