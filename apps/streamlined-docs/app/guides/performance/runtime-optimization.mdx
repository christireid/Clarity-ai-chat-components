## Performance Patterns

### Pattern 1: Component Memoization with React.memo

**Problem**: Parent re-renders cause all child components to re-render, even when their props haven't changed.

**Solution**: Use `React.memo` to prevent unnecessary re-renders of expensive components.

<CodeBlock language="tsx" title="Before: Unnecessary Re-renders">
```tsx
// Without memoization - re-renders on every parent update
function MessageItem({ message, onEdit, onDelete }) {
  console.log('MessageItem rendered:', message.id)

  return (
    <div className="message-item">
      <div className="message-content">{message.content}</div>
      <div className="message-actions">
        <button onClick={() => onEdit(message.id)}>Edit</button>
        <button onClick={() => onDelete(message.id)}>Delete</button>
      </div>
    </div>
  )
}

function MessageList({ messages }) {
  const [count, setCount] = useState(0)

  return (
    <div>
      <button onClick={() => setCount(c => c + 1)}>Count: {count}</button>
      {messages.map(msg => (
        <MessageItem
          key={msg.id}
          message={msg}
          onEdit={(id) => console.log('Edit', id)}
          onDelete={(id) => console.log('Delete', id)}
        />
      ))}
    </div>
  )
}
```
</CodeBlock>

<CodeBlock language="tsx" title="After: Memoized Components">
```tsx
// With memoization - only re-renders when props change
const MessageItem = React.memo(function MessageItem({
  message,
  onEdit,
  onDelete
}) {
  console.log('MessageItem rendered:', message.id)

  return (
    <div className="message-item">
      <div className="message-content">{message.content}</div>
      <div className="message-actions">
        <button onClick={() => onEdit(message.id)}>Edit</button>
        <button onClick={() => onDelete(message.id)}>Delete</button>
      </div>
    </div>
  )
})

function MessageList({ messages }) {
  const [count, setCount] = useState(0)

  // Memoize callbacks to maintain referential equality
  const handleEdit = useCallback((id: string) => {
    console.log('Edit', id)
  }, [])

  const handleDelete = useCallback((id: string) => {
    console.log('Delete', id)
  }, [])

  return (
    <div>
      <button onClick={() => setCount(c => c + 1)}>Count: {count}</button>
      {messages.map(msg => (
        <MessageItem
          key={msg.id}
          message={msg}
          onEdit={handleEdit}
          onDelete={handleDelete}
        />
      ))}
    </div>
  )
}
```
</CodeBlock>

### Pattern 2: Expensive Computation Memoization with useMemo

**Problem**: Heavy computations run on every render, blocking the UI thread.

**Solution**: Use `useMemo` to cache expensive computation results.

<CodeBlock language="tsx" title="Before: Repeated Computation">
```tsx
function TokenCounter({ messages }: { messages: Message[] }) {
  // Recalculates on every render - expensive!
  const totalTokens = messages.reduce((acc, msg) => {
    return acc + countTokens(msg.content) // CPU-intensive
  }, 0)

  const averageTokens = totalTokens / messages.length
  const maxTokens = Math.max(...messages.map(m => countTokens(m.content)))

  return (
    <div className="token-stats">
      <div>Total: {totalTokens}</div>
      <div>Average: {averageTokens.toFixed(0)}</div>
      <div>Max: {maxTokens}</div>
    </div>
  )
}
```
</CodeBlock>

<CodeBlock language="tsx" title="After: Memoized Computation">
```tsx
function TokenCounter({ messages }: { messages: Message[] }) {
  // Only recalculates when messages array changes
  const stats = useMemo(() => {
    const tokens = messages.map(msg => countTokens(msg.content))
    const total = tokens.reduce((acc, t) => acc + t, 0)

    return {
      total,
      average: total / messages.length,
      max: Math.max(...tokens)
    }
  }, [messages]) // Dependency: only recompute when messages change

  return (
    <div className="token-stats">
      <div>Total: {stats.total}</div>
      <div>Average: {stats.average.toFixed(0)}</div>
      <div>Max: {stats.max}</div>
    </div>
  )
}
```
</CodeBlock>

### Pattern 3: Callback Stability with useCallback

**Problem**: Inline function definitions create new references on every render, breaking React.memo optimization.

**Solution**: Use `useCallback` to maintain stable function references.

<CodeBlock language="tsx" title="Before: Unstable Callbacks">
```tsx
function ChatWindow({ messages }: { messages: Message[] }) {
  const [editingId, setEditingId] = useState<string | null>(null)

  return (
    <div>
      {messages.map(msg => (
        <MessageItem
          key={msg.id}
          message={msg}
          // New function created on every render - breaks memo
          onEdit={(id) => setEditingId(id)}
          // Another new function reference
          onDelete={(id) => deleteMessage(id)}
          // Yet another new reference
          isEditing={msg.id === editingId}
        />
      ))}
    </div>
  )
}
```
</CodeBlock>

<CodeBlock language="tsx" title="After: Stable Callbacks">
```tsx
function ChatWindow({ messages }: { messages: Message[] }) {
  const [editingId, setEditingId] = useState<string | null>(null)

  // Stable callback - same reference across renders
  const handleEdit = useCallback((id: string) => {
    setEditingId(id)
  }, []) // No dependencies - always stable

  // Stable callback with proper dependencies
  const handleDelete = useCallback((id: string) => {
    deleteMessage(id)
  }, []) // deleteMessage is stable (from module scope or props)

  const handleCancelEdit = useCallback(() => {
    setEditingId(null)
  }, [])

  return (
    <div>
      {messages.map(msg => (
        <MessageItem
          key={msg.id}
          message={msg}
          onEdit={handleEdit}
          onDelete={handleDelete}
          onCancelEdit={handleCancelEdit}
          isEditing={msg.id === editingId}
        />
      ))}
    </div>
  )
}
```
</CodeBlock>

### Pattern 4: Virtual Scrolling for Long Lists

**Problem**: Rendering 1000+ messages causes slow initial load and janky scrolling.

**Solution**: Use virtualization to only render visible items.

<CodeBlock language="tsx" title="Before: Render All Items">
```tsx
function MessageList({ messages }: { messages: Message[] }) {
  return (
    <div className="message-list" style={{ height: '600px', overflow: 'auto' }}>
      {messages.map((message, index) => (
        <MessageItem
          key={message.id}
          message={message}
          index={index}
        />
      ))}
    </div>
  )
}

// Performance issues:
// - Initial render: 1000+ DOM nodes created
// - Memory usage: High (all messages in DOM)
// - Scroll performance: Poor (browser renders all)
```
</CodeBlock>

<CodeBlock language="tsx" title="After: Virtualized List">
```tsx
import { VirtualScroller } from '@clarity-chat/react'

function MessageList({ messages }: { messages: Message[] }) {
  const scrollerRef = useRef<VirtualScrollerRef>(null)

  const renderMessage = useCallback((message: Message, index: number) => (
    <MessageItem message={message} index={index} />
  ), [])

  return (
    <VirtualScroller
      ref={scrollerRef}
      items={messages}
      itemHeight={150} // Estimated height
      height={600}
      renderItem={renderMessage}
      overscan={3} // Render 3 items above/below viewport
      gap={8}
    />
  )
}

// Performance benefits:
// - Initial render: ~20 DOM nodes (only visible + overscan)
// - Memory usage: Low (minimal DOM)
// - Scroll performance: Excellent (constant time)
```
</CodeBlock>

**Real-World Implementation**:

<CodeBlock language="tsx" title="packages/react/src/components/ai/VirtualScroller.tsx (Simplified)">
```tsx
export function VirtualScroller<T>({
  items,
  itemHeight,
  renderItem,
  height = 400,
  overscan = 3,
  gap = 0,
}: VirtualScrollerProps<T>) {
  const [scrollPosition, setScrollPosition] = useState(0)

  // Calculate item offsets and total size
  const { itemOffsets, totalSize } = useMemo(() => {
    const offsets: number[] = []
    let total = 0

    for (let i = 0; i < items.length; i++) {
      offsets.push(total)
      total += getItemSize(itemHeight, items[i], i) +
               (i < items.length - 1 ? gap : 0)
    }

    return { itemOffsets: offsets, totalSize: total }
  }, [items, itemHeight, gap])

  // Get visible range based on scroll position
  const getVisibleRange = useCallback(() => {
    const containerSize = typeof height === 'number' ? height : 400

    // Binary search for efficient range calculation
    let startIndex = binarySearch(itemOffsets, scrollPosition)
    let endIndex = binarySearch(itemOffsets, scrollPosition + containerSize)

    // Apply overscan
    startIndex = Math.max(0, startIndex - overscan)
    endIndex = Math.min(items.length - 1, endIndex + overscan)

    return { startIndex, endIndex }
  }, [scrollPosition, height, items.length, itemOffsets, overscan])

  const visibleRange = getVisibleRange()

  // Only render visible items
  const visibleItems = useMemo(() => {
    const result = []
    for (let i = visibleRange.startIndex; i <= visibleRange.endIndex; i++) {
      if (i < items.length) {
        result.push({
          item: items[i],
          index: i,
          offset: itemOffsets[i],
          size: getItemSize(itemHeight, items[i], i),
        })
      }
    }
    return result
  }, [visibleRange, items, itemOffsets, itemHeight])

  return (
    <div
      style={{ height, overflow: 'auto' }}
      onScroll={(e) => setScrollPosition(e.currentTarget.scrollTop)}
    >
      <div style={{ height: totalSize, position: 'relative' }}>
        {visibleItems.map(({ item, index, offset, size }) => (
          <div
            key={index}
            style={{
              position: 'absolute',
              top: offset,
              height: size,
              width: '100%',
            }}
          >
            {renderItem(item, index, {})}
          </div>
        ))}
      </div>
    </div>
  )
}
```
</CodeBlock>

### Pattern 5: Debouncing and Throttling Event Handlers

**Problem**: High-frequency events (typing, scrolling) trigger expensive operations repeatedly.

**Solution**: Use debouncing for delayed execution and throttling for rate-limiting.

<CodeBlock language="tsx" title="Pattern: Debouncing (Delay Until Idle)">
```tsx
import { useDebounce, useDebouncedCallback } from '@clarity-chat/react'

// Approach 1: Debounce the value
function SearchInput() {
  const [searchTerm, setSearchTerm] = useState('')
  const debouncedSearch = useDebounce(searchTerm, 300)

  useEffect(() => {
    // Only runs 300ms after user stops typing
    if (debouncedSearch) {
      searchAPI(debouncedSearch)
    }
  }, [debouncedSearch])

  return (
    <input
      value={searchTerm}
      onChange={(e) => setSearchTerm(e.target.value)}
      placeholder="Search..."
    />
  )
}

// Approach 2: Debounce the callback
function AutoSaveForm() {
  const [formData, setFormData] = useState({})

  const debouncedSave = useDebouncedCallback(
    (data) => {
      saveToAPI(data)
    },
    1000 // Wait 1s after last change
  )

  const handleChange = (field: string, value: any) => {
    const newData = { ...formData, [field]: value }
    setFormData(newData)
    debouncedSave(newData) // Auto-save after 1s of inactivity
  }

  return (
    <form>
      <input onChange={(e) => handleChange('name', e.target.value)} />
      <textarea onChange={(e) => handleChange('description', e.target.value)} />
    </form>
  )
}
```
</CodeBlock>

<CodeBlock language="tsx" title="Pattern: Throttling (Rate Limiting)">
```tsx
import { useThrottle, useThrottledCallback } from '@clarity-chat/react'

// Approach 1: Throttle the value
function ScrollTracker() {
  const [scrollY, setScrollY] = useState(0)
  const throttledScrollY = useThrottle(scrollY, 100) // Max once per 100ms

  useEffect(() => {
    const handleScroll = () => setScrollY(window.scrollY)
    window.addEventListener('scroll', handleScroll)
    return () => window.removeEventListener('scroll', handleScroll)
  }, [])

  useEffect(() => {
    // Only updates max once per 100ms
    updateScrollIndicator(throttledScrollY)
  }, [throttledScrollY])

  return <div>Scroll position: {throttledScrollY}px</div>
}

// Approach 2: Throttle the callback
function WindowResizeHandler() {
  const throttledResize = useThrottledCallback(
    () => {
      console.log('Window resized:', window.innerWidth)
      updateLayout()
    },
    200 // Max once per 200ms
  )

  useEffect(() => {
    window.addEventListener('resize', throttledResize)
    return () => window.removeEventListener('resize', throttledResize)
  }, [throttledResize])

  return <div>Resize the window</div>
}
```
</CodeBlock>

### Pattern 6: State Management Optimization

**Problem**: Inefficient state updates cause cascading re-renders across component tree.

**Solution**: Optimize state structure, use atomic updates, and implement state colocation.

<CodeBlock language="tsx" title="Before: Inefficient State Structure">
```tsx
function ChatApp() {
  // Single state object - any change re-renders entire component
  const [state, setState] = useState({
    messages: [],
    user: null,
    settings: {},
    ui: {
      sidebarOpen: true,
      theme: 'light',
      fontSize: 14,
    }
  })

  // Changing sidebar state re-renders messages unnecessarily
  const toggleSidebar = () => {
    setState(prev => ({
      ...prev,
      ui: { ...prev.ui, sidebarOpen: !prev.ui.sidebarOpen }
    }))
  }

  return (
    <div>
      <Sidebar open={state.ui.sidebarOpen} />
      <MessageList messages={state.messages} /> {/* Re-renders on any state change! */}
      <Settings settings={state.settings} />
    </div>
  )
}
```
</CodeBlock>

<CodeBlock language="tsx" title="After: Optimized State Structure">
```tsx
// Pattern 1: State Colocation - Keep state close to where it's used
function Sidebar() {
  // Sidebar state only affects Sidebar component
  const [open, setOpen] = useState(true)

  return (
    <div className={open ? 'sidebar-open' : 'sidebar-closed'}>
      <button onClick={() => setOpen(!open)}>Toggle</button>
      {/* Sidebar content */}
    </div>
  )
}

// Pattern 2: Split State by Concern
function ChatApp() {
  // Separate state - changes don't affect each other
  const [messages, setMessages] = useState<Message[]>([])
  const [user, setUser] = useState<User | null>(null)
  const [settings, setSettings] = useState<Settings>({})

  // Or use a state management library for complex state
  const { messages, user, settings } = useChatStore()

  return (
    <div>
      <Sidebar /> {/* Has its own state */}
      <MessageList messages={messages} /> {/* Only re-renders on messages change */}
      <Settings settings={settings} /> {/* Only re-renders on settings change */}
    </div>
  )
}

// Pattern 3: Atomic Updates with useReducer
type Action =
  | { type: 'ADD_MESSAGE'; message: Message }
  | { type: 'UPDATE_MESSAGE'; id: string; content: string }
  | { type: 'DELETE_MESSAGE'; id: string }

function messagesReducer(state: Message[], action: Action): Message[] {
  switch (action.type) {
    case 'ADD_MESSAGE':
      return [...state, action.message]
    case 'UPDATE_MESSAGE':
      return state.map(msg =>
        msg.id === action.id ? { ...msg, content: action.content } : msg
      )
    case 'DELETE_MESSAGE':
      return state.filter(msg => msg.id !== action.id)
    default:
      return state
  }
}

function OptimizedChatApp() {
  const [messages, dispatch] = useReducer(messagesReducer, [])

  // Atomic updates - predictable state transitions
  const addMessage = (message: Message) => {
    dispatch({ type: 'ADD_MESSAGE', message })
  }

  const updateMessage = (id: string, content: string) => {
    dispatch({ type: 'UPDATE_MESSAGE', id, content })
  }

  return (
    <MessageList
      messages={messages}
      onUpdate={updateMessage}
    />
  )
}
```
</CodeBlock>

**Pattern 4: Prevent Unnecessary Object Creation**:

<CodeBlock language="tsx" title="Common Pitfall: New Object References">
```tsx
// BAD: Creates new object on every render
function MessageList({ messages }) {
  return (
    <div>
      {messages.map(msg => (
        <MessageItem
          key={msg.id}
          message={msg}
          style={{ padding: 10, margin: 5 }} // New object every render!
          config={{ showTimestamp: true }}    // New object every render!
        />
      ))}
    </div>
  )
}

// GOOD: Stable references
const MESSAGE_STYLE = { padding: 10, margin: 5 } // Outside component
const MESSAGE_CONFIG = { showTimestamp: true }

function MessageList({ messages }) {
  return (
    <div>
      {messages.map(msg => (
        <MessageItem
          key={msg.id}
          message={msg}
          style={MESSAGE_STYLE}    // Same reference every render
          config={MESSAGE_CONFIG}  // Same reference every render
        />
      ))}
    </div>
  )
}

// OR: Use useMemo for dynamic values
function MessageList({ messages, showTimestamps }) {
  const config = useMemo(() => ({
    showTimestamp: showTimestamps,
    showAvatar: true,
  }), [showTimestamps])

  return (
    <div>
      {messages.map(msg => (
        <MessageItem
          key={msg.id}
          message={msg}
          config={config} // Stable unless showTimestamps changes
        />
      ))}
    </div>
  )
}
```
</CodeBlock>

```
   ├─ MessageItem (8ms) × 5 = 40ms
   └─ LoadingSpinner (2ms)

Why MessageList rendered:
- Props changed: messages (100 items → 101 items)

Why MessageItem[0] rendered:
- Parent rendered (could be optimized with React.memo)

Why MessageItem[1-4] rendered:
- Parent rendered (could be optimized with React.memo)
```
</CodeBlock>

**What to Look For**:

1. **Yellow/red bars** in flamegraph = Slow components (>12ms)
2. **"Parent rendered"** = Opportunity for React.memo
3. **Frequent renders** = Opportunity for useMemo/useCallback
4. **Large commit times** = Bundle size or virtualization issue

## Common Anti-Patterns to Avoid

<Callout type="warning">
### 1. Premature Optimization
Don't optimize everything upfront. Profile first, then optimize bottlenecks.
</Callout>

<CodeBlock language="tsx" title="Anti-Pattern: Over-Optimization">
```tsx
// BAD: Unnecessary memoization adds complexity
const SimpleButton = React.memo(({ label, onClick }) => (
  <button onClick={onClick}>{label}</button>
))

const MemoizedLabel = useMemo(() => props.label, [props.label])
const MemoizedOnClick = useCallback(props.onClick, [props.onClick])

// GOOD: Only optimize when profiling shows it's needed
function SimpleButton({ label, onClick }) {
  return <button onClick={onClick}>{label}</button>
}
```
</CodeBlock>

<Callout type="warning">
### 2. Incorrect useMemo Dependencies
Missing or incorrect dependencies lead to stale values and bugs.
</Callout>

<CodeBlock language="tsx" title="Anti-Pattern: Wrong Dependencies">
```tsx
// BAD: Missing dependency - uses stale value
function FilteredList({ items, filterText, category }) {
  const filtered = useMemo(() => {
    return items.filter(item =>
      item.name.includes(filterText) &&
      item.category === category
    )
  }, [items]) // Missing filterText and category!

  // ...
}

// GOOD: Complete dependencies
function FilteredList({ items, filterText, category }) {
  const filtered = useMemo(() => {
    return items.filter(item =>
      item.name.includes(filterText) &&
      item.category === category
    )
  }, [items, filterText, category]) // All dependencies included

  // ...
}
```
</CodeBlock>

<Callout type="warning">
### 3. Mutating State Directly
Direct mutations break React's change detection.
</Callout>

<CodeBlock language="tsx" title="Anti-Pattern: State Mutation">
```tsx
// BAD: Mutating array directly
function TodoList() {
  const [todos, setTodos] = useState([])

  const addTodo = (text) => {
    todos.push({ id: Date.now(), text }) // Mutation!
    setTodos(todos) // React won't detect change
  }

  // ...
}

// GOOD: Immutable updates
function TodoList() {
  const [todos, setTodos] = useState([])

  const addTodo = (text) => {
    setTodos(prev => [...prev, { id: Date.now(), text }])
  }

  // ...
}
```
</CodeBlock>
