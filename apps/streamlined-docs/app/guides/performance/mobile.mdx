## Touch Optimization

### No Hover States on Touch Devices

Remove hover effects on touch devices to prevent sticky hover states and improve performance.

#### Detection Pattern

```tsx
import { useState, useEffect } from 'react'

/**
 * Detect if the primary input is touch
 */
export function useTouchDevice(): boolean {
  const [isTouchDevice, setIsTouchDevice] = useState(false)

  useEffect(() => {
    // Check for touch capability
    const hasTouch =
      'ontouchstart' in window ||
      navigator.maxTouchPoints > 0 ||
      (navigator as any).msMaxTouchPoints > 0

    setIsTouchDevice(hasTouch)
  }, [])

  return isTouchDevice
}

/**
 * Detect primary input type with media query
 */
export function usePrimaryPointer(): 'touch' | 'mouse' | 'none' {
  const [pointerType, setPointerType] = useState<'touch' | 'mouse' | 'none'>('none')

  useEffect(() => {
    // Use pointer media query (most reliable)
    const coarse = window.matchMedia('(pointer: coarse)').matches
    const fine = window.matchMedia('(pointer: fine)').matches

    if (coarse) {
      setPointerType('touch')
    } else if (fine) {
      setPointerType('mouse')
    } else {
      setPointerType('none')
    }
  }, [])

  return pointerType
}
```

#### Conditional Hover Styles

```tsx
import { usePrimaryPointer } from '@/hooks/usePrimaryPointer'
import { cn } from '@clarity-chat/primitives'

export function MessageCard({ message }: Props) {

      )}

      {message.content}
    </div>
  )
}
```

## Reduced Motion Preferences

Respect user's motion preferences for accessibility and battery savings.

### useReducedMotion Hook

```tsx
import { useEffect, useState } from 'react'

/**
 * Detect if user prefers reduced motion
 * Respects system-level accessibility settings
 */
export function useReducedMotion(): boolean {
  const [prefersReducedMotion, setPrefersReducedMotion] = useState(false)

  useEffect(() => {
    const mediaQuery = window.matchMedia('(prefers-reduced-motion: reduce)')

    // Set initial value
    setPrefersReducedMotion(mediaQuery.matches)

    // Listen for changes
    const handler = (event: MediaQueryListEvent) => {
      setPrefersReducedMotion(event.matches)
    }

    mediaQuery.addEventListener('change', handler)
    return () => mediaQuery.removeEventListener('change', handler)
  }, [])

  return prefersReducedMotion
}
```

### Conditional Animations

```tsx
import { motion } from 'framer-motion'

  )
}
```

## Battery Efficiency

Optimize for battery life on mobile devices.

### Throttle Animations

Reduce animation frequency when battery is low:

```tsx
import { useState, useEffect } from 'react'
import { useReducedMotion } from '@clarity-chat/react/utils/animations'

/**
 * Detect battery status and adjust performance
 */
export function useBatteryOptimization() {
  const [isLowBattery, setIsLowBattery] = useState(false)
  const [isBatteryCharging, setIsBatteryCharging] = useState(false)
  const prefersReducedMotion = useReducedMotion()

  useEffect(() => {
    const checkBattery = async () => {
      if ('getBattery' in navigator) {
        const battery = await (navigator as any).getBattery()

        const updateBatteryStatus = () => {
          setIsLowBattery(battery.level < 0.2) // Below 20%
          setIsBatteryCharging(battery.charging)
        }

        updateBatteryStatus()

        battery.addEventListener('levelchange', updateBatteryStatus)
        battery.addEventListener('chargingchange', updateBatteryStatus)

        return () => {
          battery.removeEventListener('levelchange', updateBatteryStatus)
          battery.removeEventListener('chargingchange', updateBatteryStatus)
        }
      }
    }

    checkBattery()
  }, [])

  // Determine if we should reduce animations
  const shouldReduceAnimations = prefersReducedMotion || (isLowBattery && !isBatteryCharging)

  return {
    isLowBattery,
    isBatteryCharging,
    shouldReduceAnimations,
  }
}
```

### Optimize Scroll Performance

Use passive event listeners and throttle scroll handlers:

```tsx
import { useEffect, useRef } from 'react'
import { throttle } from '@clarity-chat/react/utils'

export function MessageList({ messages }: Props) {
  const containerRef = useRef<HTMLDivElement>(null)

  useEffect(() => {
    const container = containerRef.current
    if (!container) return

    // Throttle scroll handler to 60fps (16.67ms)
    const handleScroll = throttle(() => {
      // Handle scroll position
      const scrollPosition = container.scrollTop
      // Update UI based on scroll
    }, 16)

    // Use passive listener for better scroll performance
    container.addEventListener('scroll', handleScroll, { passive: true })

    return () => {
      container.removeEventListener('scroll', handleScroll)
    }
  }, [])

  return (
    <div ref={containerRef} className="overflow-y-auto scrollbar-hide h-full">
      {messages.map((message) => (
        <Message key={message.id} message={message} />
      ))}
    </div>
  )
}
```

### Handle Mobile Keyboard

Adjust UI when keyboard appears:

```tsx
import { useEffect, useState } from 'react'

/**
 * Detect when mobile keyboard is visible
 */
export function useKeyboardVisible(): boolean {
  const [isKeyboardVisible, setIsKeyboardVisible] = useState(false)

  useEffect(() => {
    const handleResize = () => {
      // On mobile, visualViewport changes when keyboard appears
      if (window.visualViewport) {
        const viewportHeight = window.visualViewport.height
        const windowHeight = window.innerHeight

        // Keyboard is visible if viewport is significantly smaller
        setIsKeyboardVisible(viewportHeight < windowHeight * 0.75)
      }
    }

    if (window.visualViewport) {
      window.visualViewport.addEventListener('resize', handleResize)
      return () => window.visualViewport.removeEventListener('resize', handleResize)
    }
  }, [])

  return isKeyboardVisible
}

export function ChatInput() {
  const isKeyboardVisible = useKeyboardVisible()

  return (
    <div
      className={cn(
        'transition-all duration-300',
        // Reduce padding when keyboard is visible
        isKeyboardVisible ? 'py-2' : 'py-4'
      )}
    >
      <textarea placeholder="Type a message..." className="w-full" />
    </div>
  )
}
```

```tsx
import { useEffect } from 'react'

/**
 * Log performance metrics in development
 */
export function usePerformanceMonitor(componentName: string) {
  useEffect(() => {
    if (process.env.NODE_ENV !== 'development') return

    const mark = `${componentName}-render`
    performance.mark(mark)

    return () => {
      performance.measure(`${componentName} render`, mark)

      const measurements = performance.getEntriesByName(`${componentName} render`)
      if (measurements.length > 0) {
        const duration = measurements[0].duration
        if (duration > 16.67) {
          // Slower than 60fps
          console.warn(`⚠️ ${componentName} render took ${duration.toFixed(2)}ms`)
        }
      }

      performance.clearMarks(mark)
      performance.clearMeasures(`${componentName} render`)
    }
  })
}

  return (
    <div>
      {messages.map((message) => (
        <Message key={message.id} message={message} />
```

```tsx
/**
 * Simulate slow network in development

export function withNetworkThrottling<T extends (...args: any[]) => Promise<any>>(
  fn: T,
  delayMs: number = 1000
): T {
  if (process.env.NODE_ENV !== 'development') {
    return fn
  }

    await new Promise((resolve) => setTimeout(resolve, delayMs))
    return fn(...args)
```

## Performance Monitoring

Track mobile performance metrics in production.

### Core Web Vitals

Monitor key performance metrics:

```tsx
import { useEffect } from 'react'

/**
 * Report Core Web Vitals
 */
export function useWebVitals() {
  useEffect(() => {
    if (typeof window === 'undefined') return

    // LCP - Largest Contentful Paint
    const observeLCP = () => {
      const observer = new PerformanceObserver((list) => {
        const entries = list.getEntries()
        const lastEntry = entries[entries.length - 1] as any

        console.log('LCP:', lastEntry.renderTime || lastEntry.loadTime)
        // Send to analytics
        analytics.track('web_vital_lcp', {
          value: lastEntry.renderTime || lastEntry.loadTime,
        })
      })

      observer.observe({ entryTypes: ['largest-contentful-paint'] })
      return observer
    }

    // FID - First Input Delay
    const observeFID = () => {
      const observer = new PerformanceObserver((list) => {
        const entries = list.getEntries()
        entries.forEach((entry: any) => {
          console.log('FID:', entry.processingStart - entry.startTime)
          analytics.track('web_vital_fid', {
            value: entry.processingStart - entry.startTime,
          })
        })
      })

      observer.observe({ entryTypes: ['first-input'] })
      return observer
    }

    // CLS - Cumulative Layout Shift
    const observeCLS = () => {
      let clsScore = 0
      const observer = new PerformanceObserver((list) => {
        for (const entry of list.getEntries()) {
          if (!(entry as any).hadRecentInput) {
            clsScore += (entry as any).value
          }
        }

        console.log('CLS:', clsScore)
        analytics.track('web_vital_cls', { value: clsScore })
      })

      observer.observe({ entryTypes: ['layout-shift'] })
      return observer
    }

    const lcpObserver = observeLCP()
    const fidObserver = observeFID()
    const clsObserver = observeCLS()

    return () => {
      lcpObserver?.disconnect()
      fidObserver?.disconnect()
      clsObserver?.disconnect()
    }
  }, [])
}
```

### Frame Rate Monitoring

Track animation performance:

```tsx
export function useFrameRate() {
  const [fps, setFps] = useState(60)

  useEffect(() => {
    let lastTime = performance.now()
    let frames = 0
    let rafId: number

    const measureFPS = (currentTime: number) => {
      frames++

      if (currentTime >= lastTime + 1000) {
        const currentFps = Math.round((frames * 1000) / (currentTime - lastTime))
        setFps(currentFps)

        if (currentFps < 30) {
          console.warn('⚠️ Low frame rate detected:', currentFps, 'fps')
        }

        frames = 0
        lastTime = currentTime
      }

      rafId = requestAnimationFrame(measureFPS)
    }

    rafId = requestAnimationFrame(measureFPS)

    return () => cancelAnimationFrame(rafId)
  }, [])

  return fps
}
```

```tsx
        console.log(`Memory: ${usedMB.toFixed(2)}MB / ${totalMB.toFixed(2)}MB`)

        if (usedMB > totalMB * 0.9) {
          console.warn('⚠️ High memory usage detected')
        }
      }
    }, 5000) // Check every 5 seconds

    return () => clearInterval(interval)
  }, [])
}
```

## Mobile-Specific Patterns

Common patterns for mobile optimization.

### Pull-to-Refresh

Implement native-like pull-to-refresh:

```tsx
import { useRef, useState } from 'react'

export function MessageList({ onRefresh }: Props) {
  const [isRefreshing, setIsRefreshing] = useState(false)
  const startY = useRef(0)
  const currentY = useRef(0)

  const handleTouchStart = (e: React.TouchEvent) => {
    startY.current = e.touches[0].clientY
  }

  const handleTouchMove = (e: React.TouchEvent) => {
    currentY.current = e.touches[0].clientY

    // Check if scrolled to top and pulling down
    const element = e.currentTarget
    const isPullingDown = currentY.current > startY.current
    const isAtTop = element.scrollTop === 0

    if (isAtTop && isPullingDown && !isRefreshing) {
      const pullDistance = currentY.current - startY.current

      if (pullDistance > 80) {
        setIsRefreshing(true)
        onRefresh().finally(() => setIsRefreshing(false))
      }
    }
  }

  return (
    <div
      onTouchStart={handleTouchStart}
      onTouchMove={handleTouchMove}
      className="overflow-y-auto scrollbar-hide h-full"
    >
      {isRefreshing && (
        <div className="flex items-center justify-center py-4">
          <div className="animate-spin h-6 w-6 border-2 border-primary rounded-full border-t-transparent" />
        </div>
      )}

      <MessageListContent />
    </div>
  )
}
```

### Bottom Sheet

Mobile-optimized bottom sheet:

```tsx
import { motion, PanInfo } from 'framer-motion'
import { useState } from 'react'

export function BottomSheet({ children, isOpen, onClose }: Props) {
  const [isDragging, setIsDragging] = useState(false)

  const handleDragEnd = (event: any, info: PanInfo) => {
    setIsDragging(false)

    // Close if dragged down more than 50%
    if (info.offset.y > 150) {
      onClose()
    }
  }

  return (
    <motion.div
      initial={{ y: '100%' }}
      animate={{ y: isOpen ? 0 : '100%' }}
      drag="y"
      dragConstraints={{ top: 0, bottom: 0 }}
      dragElastic={0.2}
      onDragStart={() => setIsDragging(true)}
      onDragEnd={handleDragEnd}
      className={cn(
        'fixed bottom-0 left-0 right-0',
        'bg-card rounded-t-2xl shadow-2xl',
        'max-h-[90vh] overflow-hidden',
        'z-50'
      )}
    >
      {/* Drag handle */}
      <div className="flex justify-center py-3">
        <div className="w-10 h-1 bg-muted-foreground/30 rounded-full" />
      </div>

      {/* Content */}
      <div className="overflow-y-auto scrollbar-hide px-4 pb-safe">{children}</div>
    </motion.div>
  )
}
```

### Infinite Scroll

Efficient infinite scrolling for mobile:

```tsx
import { useEffect, useRef } from 'react'

export function InfiniteMessageList({ messages, loadMore, hasMore }: Props) {
  const observerTarget = useRef<HTMLDivElement>(null)

  useEffect(() => {
    const observer = new IntersectionObserver(
      (entries) => {
        if (entries[0].isIntersecting && hasMore) {
          loadMore()
        }
      },
      {
        threshold: 0.5,
        // Add rootMargin to trigger slightly early
        rootMargin: '100px',
      }
    )

    const target = observerTarget.current
    if (target) {
      observer.observe(target)
    }

    return () => {
      if (target) {
        observer.unobserve(target)
      }
    }
  }, [loadMore, hasMore])

  return (
    <div className="overflow-y-auto scrollbar-hide h-full">
      {messages.map((message) => (
        <Message key={message.id} message={message} />
      ))}

      {/* Sentinel element to trigger loading */}
      <div ref={observerTarget} className="h-4" />

      {hasMore && (
        <div className="flex justify-center py-4">
          <div className="animate-spin h-6 w-6 border-2 border-primary rounded-full border-t-transparent" />
        </div>
      )}
    </div>
  )
}
```
