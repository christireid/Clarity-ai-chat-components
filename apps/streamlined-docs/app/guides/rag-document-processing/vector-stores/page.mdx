# Vector Store Setup Guide

Configure and optimize vector stores for production RAG systems. This guide covers setup, configuration, performance tuning, and scaling for all supported vector databases.

## Quick Start

Choose your vector store provider and follow the setup instructions:

- [Pinecone](#pinecone-setup) - Managed, serverless vector database
- [Qdrant](#qdrant-setup) - Open-source vector search engine
- [Weaviate](#weaviate-setup) - Open-source vector database with modules
- [Chroma](#chroma-setup) - Lightweight embedding database

## Provider Comparison

### At a Glance

| Feature | Pinecone | Qdrant | Weaviate | Chroma |
|---------|----------|--------|----------|--------|
| **Hosting** | Managed cloud | Self-hosted or cloud | Self-hosted or cloud | Self-hosted |
| **Pricing** | Usage-based | Free (self-hosted) | Free (self-hosted) | Free |
| **Setup Time** | 5 minutes | 15 minutes | 20 minutes | 10 minutes |
| **Scaling** | Automatic | Manual | Manual | Manual |
| **Max Vectors** | Billions | Billions | Billions | Millions |
| **Performance** | Excellent | Excellent | Very Good | Good |
| **Best For** | Production apps | High performance | Feature-rich apps | Development/prototyping |

### 2. Document Indexing Pipeline

```typescript
import { createVectorStore } from '@clarity-chat/react/vector-stores'
import { OpenAIEmbeddings } from '@clarity-chat/react/embeddings'
import { RecursiveCharacterTextSplitter } from '@clarity-chat/react/text-chunking'

async function indexDocuments(documents: Document[]) {
  // Initialize components
  const embeddings = new OpenAIEmbeddings({
    apiKey: process.env.OPENAI_API_KEY!,
    model: 'text-embedding-3-large'
  })

  const splitter = new RecursiveCharacterTextSplitter({
    chunkSize: 1000,
    chunkOverlap: 200
  })

  const vectorStore = createVectorStore({
    provider: 'pinecone',
    apiKey: process.env.PINECONE_API_KEY!,
    environment: 'us-east1-gcp',
    indexName: 'documents'
  })

  await vectorStore.initialize()

  // Process documents
  for (const doc of documents) {
    // Split into chunks
    const chunks = await splitter.split(doc.content)

    // Generate embeddings
    const vectors = await Promise.all(
      chunks.map(async (chunk, i) => ({
        id: `${doc.id}-chunk-${i}`,
        values: await embeddings.embed(chunk.content),
        metadata: {
          documentId: doc.id,
          title: doc.title,
          chunkIndex: i,
          totalChunks: chunks.length,
          content: chunk.content,
          timestamp: Date.now()
        }
      }))
    )

    // Upsert in batches
    await vectorStore.upsert(vectors, {
      namespace: 'production',
      batchSize: 100
    })

    console.log(`Indexed ${doc.id}: ${chunks.length} chunks`)
  }
}
```

### 3. Semantic Search with Reranking

```typescript
import { createVectorStore } from '@clarity-chat/react/vector-stores'
import { CohereReranker } from '@clarity-chat/react/reranking'

async function semanticSearch(query: string) {
  const embeddings = new OpenAIEmbeddings({
    apiKey: process.env.OPENAI_API_KEY!
  })

  const vectorStore = createVectorStore({
    provider: 'qdrant',
    endpoint: process.env.QDRANT_ENDPOINT!,
    indexName: 'documents'
  })

  // Initial retrieval (over-fetch for reranking)
  const queryVector = await embeddings.embed(query)
  const initialResults = await vectorStore.query({
    vector: queryVector,
    topK: 50,  // Retrieve more for reranking
    filter: {
      status: { $eq: 'published' }
    }
  })

  // Rerank for precision
  const reranker = new CohereReranker({
    apiKey: process.env.COHERE_API_KEY!,
    model: 'rerank-english-v3.0'
  })

  const reranked = await reranker.rerank({
    query,
    documents: initialResults.map(r => r.metadata!.content as string),
    topK: 10
  })

  // Map back to original results
  return reranked.map(r => initialResults[r.index])
}
```

### 4. Multi-Tenant Isolation

```typescript
import { createVectorStore } from '@clarity-chat/react/vector-stores'

class MultiTenantVectorStore {
  private store: VectorStore

  constructor(config: VectorStoreConfig) {
    this.store = createVectorStore(config)
  }

  async initialize() {
    await this.store.initialize()
  }

  // Tenant-scoped upsert
  async upsertForTenant(tenantId: string, vectors: Vector[]) {
    return this.store.upsert(vectors, {
      namespace: `tenant-${tenantId}`
    })
  }

  // Tenant-scoped query
  async queryForTenant(tenantId: string, query: VectorQuery) {
    return this.store.query({
      ...query,
      namespace: `tenant-${tenantId}`
    })
  }

  // Tenant deletion (GDPR compliance)
  async deleteTenant(tenantId: string) {
    await this.store.deleteNamespace(`tenant-${tenantId}`)
  }
}

// Usage
const multiTenantStore = new MultiTenantVectorStore({
  provider: 'pinecone',
  apiKey: process.env.PINECONE_API_KEY!,
  environment: 'us-east1-gcp',
  indexName: 'multi-tenant-docs'
})

await multiTenantStore.initialize()

// Tenant A
await multiTenantStore.upsertForTenant('tenant-a', vectorsA)
const resultsA = await multiTenantStore.queryForTenant('tenant-a', { vector, topK: 10 })

// Tenant B (isolated)
await multiTenantStore.upsertForTenant('tenant-b', vectorsB)
const resultsB = await multiTenantStore.queryForTenant('tenant-b', { vector, topK: 10 })
```

## Query Performance Tips

### 1. Use Approximate Search

```typescript
// Trade accuracy for speed
const fastResults = await vectorStore.query({
  vector: queryVector,
  topK: 10,
  minScore: 0.7  // Skip low-quality matches early

```

### 2. Cache Frequent Queries

```typescript
import { ExactCache } from '@clarity-chat/token-optimization'

const queryCache = new ExactCache<VectorMatch[]>({
  maxSize: 1000,
  ttl: 3600000  // 1 hour
})

async function cachedQuery(query: string) {
  // Check cache
  const cached = queryCache.get(query)
  if (cached.hit) return cached.value!

  // Query vector store
  const queryVector = await embeddings.embed(query)
  const results = await vectorStore.query({
    vector: queryVector,
    topK: 10
  })

  // Cache results
  queryCache.set(query, results)
  return results
}
```

### 3. Pre-filter Before Search

```typescript
// ✅ Good: Filter reduces search space
const results = await vectorStore.query({
  vector: queryVector,
  topK: 10,
  filter: {
    category: { $eq: 'documentation' },
    timestamp: { $gte: Date.now() - 86400000 }
  }
})

// ❌ Bad: Post-filter wastes computation
const allResults = await vectorStore.query({
  vector: queryVector,
  topK: 100
})
const filtered = allResults.filter(r =>
  r.metadata?.category === 'documentation' &&
  r.metadata?.timestamp >= Date.now() - 86400000
).slice(0, 10)
```

// Export vectors for backup
async function backupVectorStore() {
  const allIds: string[] = []
  let nextToken: string | undefined

  // Paginate through all vectors
  do {
    const page = await vectorStore.list(undefined, 1000, nextToken)
    allIds.push(...page.ids)
    nextToken = page.nextToken
  } while (nextToken)

  // Fetch and save vectors
  const batchSize = 100
  for (let i = 0; i < allIds.length; i += batchSize) {
    const batch = allIds.slice(i, i + batchSize)
    const vectors = await vectorStore.fetch(batch)

    // Save to backup storage
    await fs.writeFile(
      `backup-${i / batchSize}.json`,
      JSON.stringify(vectors)
    )
  }

  console.log(`Backed up ${allIds.length} vectors`)
}

// Migrate from Pinecone to Qdrant
async function migrateVectorStore() {
  const source = createVectorStore({
    provider: 'pinecone',
    apiKey: process.env.PINECONE_API_KEY!,
    environment: 'us-east1-gcp',
    indexName: 'old-index'
  })

  const target = createVectorStore({
    provider: 'qdrant',
    endpoint: process.env.QDRANT_ENDPOINT!,
    apiKey: process.env.QDRANT_API_KEY,
    indexName: 'new-collection'
  })

  await source.initialize()
  await target.initialize()

  // Paginate and migrate
  let nextToken: string | undefined
  let totalMigrated = 0

  do {
    const { ids, nextToken: token } = await source.list(undefined, 1000, nextToken)
    const vectors = await source.fetch(ids)

    await target.upsert(vectors, { batchSize: 100 })

    totalMigrated += vectors.length
    nextToken = token

    console.log(`Migrated ${totalMigrated} vectors...`)
  } while (nextToken)

  console.log(`Migration complete: ${totalMigrated} vectors`)
}

### 3. Monitor Performance

```typescript
class MonitoredVectorStore {
  private store: VectorStore
  private metrics = {
    upserts: 0,
    queries: 0,
    errors: 0,
    avgUpsertLatency: 0,
    avgQueryLatency: 0
  }

  constructor(store: VectorStore) {
    this.store = store
  }

  async upsert(vectors: Vector[], options?: VectorUpsertOptions) {
    const start = Date.now()
    try {
      await this.store.upsert(vectors, options)

      const latency = Date.now() - start
      this.metrics.upserts++
      this.metrics.avgUpsertLatency =
        (this.metrics.avgUpsertLatency * (this.metrics.upserts - 1) + latency) /
        this.metrics.upserts
    } catch (error) {
      this.metrics.errors++
      throw error
    }
  }

  async query(query: VectorQuery) {
    const start = Date.now()
    try {
      const results = await this.store.query(query)

      const latency = Date.now() - start
      this.metrics.queries++
      this.metrics.avgQueryLatency =
        (this.metrics.avgQueryLatency * (this.metrics.queries - 1) + latency) /
        this.metrics.queries

      return results
    } catch (error) {
      this.metrics.errors++
      throw error
    }
  }

  getMetrics() {
    return { ...this.metrics }
  }
}
```
