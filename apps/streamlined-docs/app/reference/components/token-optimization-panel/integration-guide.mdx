### 1. Analytics API Endpoints

Create a REST API for token optimization analytics.

```typescript
// app/api/analytics/token-optimization/route.ts
import { NextRequest, NextResponse } from 'next/server'
import { z } from 'zod'

// Request validation schema
const analyticsQuerySchema = z.object({
  timeRange: z.enum(['1h', '24h', '7d', '30d', '90d', 'custom']),
  startDate: z.string().datetime().optional(),
  endDate: z.string().datetime().optional(),
  aggregation: z.enum(['hour', 'day', 'week', 'month']).default('day'),
  userId: z.string().optional(),
  teamId: z.string().optional(),
  models: z.array(z.string()).optional(),
})

export async function GET(request: NextRequest) {
  try {
    const { searchParams } = new URL(request.url)
    const query = {
      timeRange: searchParams.get('timeRange') || '7d',
      startDate: searchParams.get('startDate'),
      endDate: searchParams.get('endDate'),
      aggregation: searchParams.get('aggregation') || 'day',
      userId: searchParams.get('userId'),
      teamId: searchParams.get('teamId'),
      models: searchParams.getAll('models'),
    }

    // Validate request
    const validated = analyticsQuerySchema.parse(query)

    // Fetch data from your database
    const data = await fetchTokenOptimizationMetrics(validated)

    return NextResponse.json(data)
  } catch (error) {
    if (error instanceof z.ZodError) {
      return NextResponse.json(
        { error: 'Invalid request parameters', details: error.errors },
        { status: 400 }
      )
    }

    console.error('Analytics fetch error:', error)
    return NextResponse.json(
      { error: 'Failed to fetch analytics data' },
      { status: 500 }
    )
  }
}

// Database query function
async function fetchTokenOptimizationMetrics(query: z.infer<typeof analyticsQuerySchema>) {
  const { timeRange, startDate, endDate, aggregation, userId, teamId, models } = query

  // Calculate date range
  const dateRange = calculateDateRange(timeRange, startDate, endDate)

  // Query your database (example with PostgreSQL)
  const metrics = await db.query(`
    SELECT
      DATE_TRUNC($1, created_at) as timestamp,
      COUNT(*) as total_optimizations,
      SUM(tokens_saved) as tokens_saved,
      SUM(cost_saved) as cost_saved,
      AVG(optimization_percent) as avg_optimization_percent,
      jsonb_object_agg(optimization_method, method_count) as method_breakdown,
      AVG(cache_hit_rate) as cache_hit_rate
    FROM token_optimization_events
    WHERE
      created_at BETWEEN $2 AND $3
      ${userId ? 'AND user_id = $4' : ''}
      ${teamId ? 'AND team_id = $5' : ''}
      ${models?.length ? 'AND model_id = ANY($6)' : ''}
    GROUP BY timestamp
    ORDER BY timestamp DESC
  `, [
    aggregation,
    dateRange.start,
    dateRange.end,
    userId,
    teamId,
    models,
  ])

  // Calculate summary statistics
  const summary = {
    totalOptimizations: metrics.reduce((sum, m) => sum + m.total_optimizations, 0),
    totalTokensSaved: metrics.reduce((sum, m) => sum + m.tokens_saved, 0),
    totalCostSaved: metrics.reduce((sum, m) => sum + m.cost_saved, 0),
    averageSavings: metrics.reduce((sum, m) => sum + m.avg_optimization_percent, 0) / metrics.length,
    averageCacheHitRate: metrics.reduce((sum, m) => sum + m.cache_hit_rate, 0) / metrics.length,
  }

  return {
    summary,
    timeseries: metrics,
    dateRange,
    aggregation,
  }
}

function calculateDateRange(timeRange: string, startDate?: string, endDate?: string) {
  if (timeRange === 'custom' && startDate && endDate) {
    return {
      start: new Date(startDate),
      end: new Date(endDate),
    }
  }

  const end = new Date()
  const start = new Date()

  switch (timeRange) {
    case '1h':
      start.setHours(start.getHours() - 1)
      break
    case '24h':
      start.setHours(start.getHours() - 24)
      break
    case '7d':
      start.setDate(start.getDate() - 7)
      break
    case '30d':
      start.setDate(start.getDate() - 30)
      break
    case '90d':
      start.setDate(start.getDate() - 90)
      break
  }

  return { start, end }
}
```

### 2. Event Tracking Integration

Track token optimization events in real-time.

```typescript
// lib/analytics/token-optimization-tracker.ts
export interface TokenOptimizationEvent {
  userId: string
  teamId?: string
  conversationId: string
  timestamp: Date
  originalTokens: number
  optimizedTokens: number
  tokensSaved: number
  optimizationPercent: number
  optimizationMethod: 'toon' | 'llmlingua' | 'semantic' | 'truncation' | 'caching'
  modelId: string
  costBefore: number
  costAfter: number
  costSaved: number
  cacheHit: boolean
  metadata?: Record<string, unknown>
}

export class TokenOptimizationTracker {
  private apiEndpoint: string
  private batchSize: number
  private batchInterval: number
  private eventQueue: TokenOptimizationEvent[] = []
  private flushInterval?: NodeJS.Timeout

  constructor(config: {
    apiEndpoint: string
    batchSize?: number
    batchInterval?: number
  }) {
    this.apiEndpoint = config.apiEndpoint
    this.batchSize = config.batchSize || 50
    this.batchInterval = config.batchInterval || 5000 // 5 seconds

    this.startBatchFlush()
  }

  /**
   * Track a token optimization event
   */
  async trackOptimization(event: TokenOptimizationEvent): Promise<void> {
    this.eventQueue.push(event)

    // Flush immediately if batch size reached
    if (this.eventQueue.length >= this.batchSize) {
      await this.flush()
    }
  }

  /**
   * Flush events to the backend
   */
  private async flush(): Promise<void> {
    if (this.eventQueue.length === 0) return

    const events = [...this.eventQueue]
    this.eventQueue = []

    try {
      await fetch(this.apiEndpoint, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
        },
        body: JSON.stringify({ events }),
      })
    } catch (error) {
      console.error('Failed to flush optimization events:', error)
      // Re-queue events on failure
      this.eventQueue.unshift(...events)
    }
  }

  /**
   * Start periodic batch flush
   */
  private startBatchFlush(): void {
    this.flushInterval = setInterval(() => {
      this.flush()
    }, this.batchInterval)
  }

  /**
   * Stop tracking and flush remaining events
   */
  async stop(): Promise<void> {
    if (this.flushInterval) {
      clearInterval(this.flushInterval)
    }
    await this.flush()
  }
}

// Usage
export const optimizationTracker = new TokenOptimizationTracker({
  apiEndpoint: '/api/analytics/token-optimization/events',
})
```

```typescript
import { useState, useEffect, useCallback } from 'react'

  timeRange: '1h' | '24h' | '7d' | '30d' | '90d' | 'custom'
  startDate?: string
  endDate?: string
  aggregation?: 'hour' | 'day' | 'week' | 'month'

  const queryString = new URLSearchParams({
    timeRange: query.timeRange,
    ...(query.startDate && { startDate: query.startDate }),
    ...(query.endDate && { endDate: query.endDate }),
    ...(query.aggregation && { aggregation: query.aggregation }),
    ...(query.userId && { userId: query.userId }),
    ...(query.teamId && { teamId: query.teamId }),
  }).toString()

  const { data, error, mutate, isLoading } = useSWR(
    `/api/analytics/token-optimization?${queryString}`,
    async (url) => {
      const response = await fetch(url)
      if (!response.ok) {
        throw new Error('Failed to fetch analytics')
      }
      return response.json()
    },
    {
      refreshInterval: 30000, // Refresh every 30 seconds
      revalidateOnFocus: true,
    }
  )

  const refresh = useCallback(() => {
    mutate()
  }, [mutate])

  return {
    data,
    error,
    isLoading,
    refresh,
  }
}
```

```typescript
// components/SavingsChart.tsx
'use client'

import { useMemo } from 'react'
import { motion } from 'framer-motion'
import { TrendingUp, DollarSign } from 'lucide-react'

interface DataPoint {
  timestamp: string
  tokensSaved: number
  costSaved: number
}

interface SavingsChartProps {
  data: DataPoint[]
  height?: number
  showCost?: boolean
  showTokens?: boolean
  className?: string
}

export function SavingsChart({
  data,
  height = 300,
  showCost = true,
  showTokens = true,
  className,
}: SavingsChartProps) {
  // Calculate scales
  const { maxTokens, maxCost, points } = useMemo(() => {
    if (!data.length) return { maxTokens: 0, maxCost: 0, points: [] }

    const maxTokens = Math.max(...data.map((d) => d.tokensSaved))

      x: (i / (data.length - 1)) * 100,

      currency: 'USD',

          <div className="rounded-lg border border-border bg-card p-4">

        )}

                key={y}
                x1="0"

                x2="100%"
                y2={`${y}%`}

                strokeWidth="1"
```

```typescript
// components/MethodBreakdownChart.tsx
'use client'

import { useMemo } from 'react'
import { motion } from 'framer-motion'

interface MethodData {
  method: string
  count: number
  color: string
}

interface MethodBreakdownChartProps {
  data: Record<string, number>
  size?: number
  className?: string
}

const METHOD_COLORS: Record<string, string> = {
  toon: '#3b82f6', // blue
  llmlingua: '#10b981', // green
  semantic: '#f59e0b', // amber
  truncation: '#ef4444', // red
  caching: '#8b5cf6', // purple
}

export function MethodBreakdownChart({
  data,
  size = 200,
  className,
}: MethodBreakdownChartProps) {
  const { total, segments } = useMemo(() => {
    const total = Object.values(data).reduce((sum, count) => sum + count, 0)
    let currentAngle = 0

    const segments = Object.entries(data).map(([method, count]) => {
      const percentage = (count / total) * 100
      const angle = (count / total) * 360
      const startAngle = currentAngle
      const endAngle = currentAngle + angle

      currentAngle = endAngle

      return {
        method,
        count,
        percentage,
        startAngle,
        endAngle,
        color: METHOD_COLORS[method] || '#6b7280',
      }
    })

    return { total, segments }
  }, [data])

  const createArc = (startAngle: number, endAngle: number, radius: number) => {
    const startRadians = ((startAngle - 90) * Math.PI) / 180
    const endRadians = ((endAngle - 90) * Math.PI) / 180

    const x1 = radius + radius * Math.cos(startRadians)
    const y1 = radius + radius * Math.sin(startRadians)
    const x2 = radius + radius * Math.cos(endRadians)
    const y2 = radius + radius * Math.sin(endRadians)

    const largeArc = endAngle - startAngle > 180 ? 1 : 0

    return `M ${radius} ${radius} L ${x1} ${y1} A ${radius} ${radius} 0 ${largeArc} 1 ${x2} ${y2} Z`
  }

  if (total === 0) {
    return (
      <div
        className={`flex items-center justify-center rounded-xl border border-border bg-muted/30 ${className}`}
        style={{ width: size, height: size }}
      >
        <p className="text-sm text-muted-foreground">No data</p>
      </div>
    )
  }

  const radius = size / 2
  const innerRadius = radius * 0.6

  return (
    <div className={`space-y-4 ${className}`}>
      <div className="relative" style={{ width: size, height: size }}>
        <svg width={size} height={size}>
          {/* Segments */}
          {segments.map((segment, i) => (
            <motion.path
              key={segment.method}
              d={createArc(segment.startAngle, segment.endAngle, radius)}
              fill={segment.color}
              initial={{ opacity: 0, scale: 0.8 }}
              animate={{ opacity: 1, scale: 1 }}
              transition={{ duration: 0.5, delay: i * 0.1 }}
              className="cursor-pointer hover:opacity-80 transition-opacity"
            >
              <title>
                {segment.method}: {segment.count} ({segment.percentage.toFixed(1)}%)
              </title>
            </motion.path>
          ))}

          {/* Inner circle (donut hole) */}
          <circle
            cx={radius}
            cy={radius}
            r={innerRadius}
            fill="hsl(var(--card))"
          />

          {/* Center text */}
          <text
            x={radius}
            y={radius - 10}
            textAnchor="middle"
            className="text-2xl font-bold fill-foreground"
          >
            {total}
          </text>
          <text
            x={radius}
            y={radius + 15}
            textAnchor="middle"
            className="text-sm fill-muted-foreground"
          >
            Total
          </text>
        </svg>
      </div>

      {/* Legend */}
      <div className="space-y-2">
        {segments.map((segment) => (
          <div key={segment.method} className="flex items-center justify-between text-sm">
            <div className="flex items-center gap-2">
              <div
                className="w-3 h-3 rounded-full"
                style={{ backgroundColor: segment.color }}
              />
              <span className="font-medium capitalize">{segment.method}</span>
            </div>
            <div className="flex items-center gap-3">
              <span className="text-muted-foreground">{segment.count}</span>
              <span className="font-mono text-xs text-muted-foreground w-12 text-right">
                {segment.percentage.toFixed(1)}%
              </span>
            </div>
          </div>
        ))}
      </div>
    </div>
  )
}
```

## Alert System

### 1. Budget Threshold Alerts

```typescript
// components/BudgetAlertSystem.tsx
'use client'

import { useState, useEffect } from 'react'
import { motion, AnimatePresence } from 'framer-motion'
import { AlertTriangle, AlertCircle, CheckCircle, X, Bell } from 'lucide-react'

export interface BudgetThreshold {
  id: string
  name: string
  type: 'cost' | 'tokens' | 'usage'
  threshold: number
  period: 'daily' | 'weekly' | 'monthly'
  currentValue: number
  enabled: boolean
  notificationChannels: ('email' | 'slack' | 'webhook' | 'ui')[]
}

export interface BudgetAlert {
  id: string
  thresholdId: string
  severity: 'info' | 'warning' | 'critical'
  message: string
  currentValue: number
  threshold: number
  percentage: number
  timestamp: Date
  acknowledged: boolean
}

interface BudgetAlertSystemProps {
  thresholds: BudgetThreshold[]
  onThresholdUpdate?: (threshold: BudgetThreshold) => void
  onAlertAcknowledge?: (alertId: string) => void
  className?: string
}

export function BudgetAlertSystem({
  thresholds,
  onThresholdUpdate,
  onAlertAcknowledge,
  className,
}: BudgetAlertSystemProps) {
  const [alerts, setAlerts] = useState<BudgetAlert[]>([])

  // Check thresholds and generate alerts
  useEffect(() => {
    const newAlerts: BudgetAlert[] = []

    thresholds.forEach((threshold) => {
      if (!threshold.enabled) return

      const percentage = (threshold.currentValue / threshold.threshold) * 100

      if (percentage >= 95) {
        newAlerts.push({
          id: `${threshold.id}-critical`,
          thresholdId: threshold.id,
          severity: 'critical',
          message: `Critical: ${threshold.name} at ${percentage.toFixed(1)}% of ${threshold.period} budget`,
          currentValue: threshold.currentValue,
          threshold: threshold.threshold,
          percentage,
          timestamp: new Date(),
          acknowledged: false,
        })
      } else if (percentage >= 80) {
        newAlerts.push({
          id: `${threshold.id}-warning`,
          thresholdId: threshold.id,
          severity: 'warning',
          message: `Warning: ${threshold.name} at ${percentage.toFixed(1)}% of ${threshold.period} budget`,
          currentValue: threshold.currentValue,
          threshold: threshold.threshold,
          percentage,
          timestamp: new Date(),
          acknowledged: false,
        })
      } else if (percentage >= 60) {
        newAlerts.push({
          id: `${threshold.id}-info`,
          thresholdId: threshold.id,
          severity: 'info',
          message: `Info: ${threshold.name} at ${percentage.toFixed(1)}% of ${threshold.period} budget`,
          currentValue: threshold.currentValue,
          threshold: threshold.threshold,
          percentage,
          timestamp: new Date(),
          acknowledged: false,
        })
      }
    })

    setAlerts(newAlerts)
  }, [thresholds])

  const getSeverityStyles = (severity: BudgetAlert['severity']) => {
    switch (severity) {
      case 'critical':
        return {
          bg: 'bg-red-500/10',
          border: 'border-red-500/30',
          text: 'text-red-500',
          icon: AlertCircle,
        }
      case 'warning':
        return {
          bg: 'bg-amber-500/10',
          border: 'border-amber-500/30',
          text: 'text-amber-500',
          icon: AlertTriangle,
        }
      case 'info':
        return {
          bg: 'bg-blue-500/10',
          border: 'border-blue-500/30',
          text: 'text-blue-500',
          icon: Bell,
        }
    }
  }

  const handleAcknowledge = (alertId: string) => {
    setAlerts((prev) =>
      prev.map((alert) =>
        alert.id === alertId ? { ...alert, acknowledged: true } : alert
      )
    )
    onAlertAcknowledge?.(alertId)
  }

  const activeAlerts = alerts.filter((alert) => !alert.acknowledged)

  return (
    <div className={`space-y-4 ${className}`}>
      {/* Alert Summary */}
      <div className="rounded-xl border border-border bg-card p-4">
        <div className="flex items-center justify-between mb-4">

            </span>
            {activeAlerts.length > 0 && (
              <motion.div

          </div>

        {/* Alert List */}
        <AnimatePresence mode="popLayout">

              </div>
            </motion.div>
          ) : (
            <div className="space-y-3">
              {activeAlerts.map((alert) => {
                const styles = getSeverityStyles(alert.severity)
                const Icon = styles.icon

                return (
                  <motion.div
                    key={alert.id}
                    initial={{ opacity: 0, x: -20 }}

                                : 'bg-blue-500'
                          }`}
                          initial={{ width: 0 }}
```

```typescript
// lib/export/analytics-exporter.ts
export class AnalyticsExporter {
  /**
   * Export analytics data to CSV
   */
  static exportToCSV(data: any[], filename: string = 'analytics.csv'): void {
    if (!data.length) return

    // Generate CSV headers
    const headers = Object.keys(data[0])
    const csvContent = [
      headers.join(','),
      ...data.map((row) =>
        headers.map((header) => {
          const value = row[header]
          // Escape values containing commas or quotes
          if (typeof value === 'string' && (value.includes(',') || value.includes('"'))) {
            return `"${value.replace(/"/g, '""')}"`
          }
          return value
        }).join(',')
      ),
    ].join('\n')

    this.downloadFile(csvContent, filename, 'text/csv')
  }

  /**
   * Export analytics data to JSON
   */
  static exportToJSON(data: any, filename: string = 'analytics.json'): void {
    const jsonContent = JSON.stringify(data, null, 2)
    this.downloadFile(jsonContent, filename, 'application/json')
  }

  /**
   * Export analytics report to PDF (requires jsPDF)
   */
  static async exportToPDF(
    data: {
      title: string
      summary: any
      charts: Array<{ title: string; data: any }>
      tables: Array<{ title: string; headers: string[]; rows: any[][] }>
    },
    filename: string = 'analytics-report.pdf'
  ): Promise<void> {
    // This is a simplified example - you'd use a library like jsPDF in production
    const reportHTML = `
      <html>
        <head>
          <title>${data.title}</title>
          <style>
            body { font-family: Arial, sans-serif; margin: 20px; }
            h1 { color: #333; }
            table { width: 100%; border-collapse: collapse; margin-top: 20px; }
            th, td { border: 1px solid #ddd; padding: 8px; text-align: left; }
            th { background-color: #f2f2f2; }
          </style>
        </head>
        <body>
          <h1>${data.title}</h1>
          <div>
            <h2>Summary</h2>
            ${Object.entries(data.summary).map(([key, value]) => `
              <p><strong>${key}:</strong> ${value}</p>
            `).join('')}
          </div>
          ${data.tables.map((table) => `
            <div>
              <h2>${table.title}</h2>
              <table>
                <thead>
                  <tr>${table.headers.map((h) => `<th>${h}</th>`).join('')}</tr>
                </thead>
                <tbody>
                  ${table.rows.map((row) => `
                    <tr>${row.map((cell) => `<td>${cell}</td>`).join('')}</tr>
                  `).join('')}
                </tbody>
              </table>
            </div>
          `).join('')}
        </body>
      </html>
    `

    // Print the HTML (user can save as PDF)
    const printWindow = window.open('', '', 'height=600,width=800')
    if (printWindow) {
      printWindow.document.write(reportHTML)
      printWindow.document.close()
      printWindow.print()
    }
  }

  /**
   * Helper to download file
   */
  private static downloadFile(content: string, filename: string, mimeType: string): void {
    const blob = new Blob([content], { type: mimeType })
    const url = URL.createObjectURL(blob)
    const link = document.createElement('a')
    link.href = url
    link.download = filename
    document.body.appendChild(link)
    link.click()
    document.body.removeChild(link)
    URL.revokeObjectURL(url)
  }

  /**

  static generateEmailReport(data: {
```

## Best Practices

### 1. Performance Optimization

- **Batch API Requests**: Group analytics requests to reduce API calls
- **Cache Aggressively**: Use SWR or React Query for client-side caching
- **Lazy Load Charts**: Only render charts when visible using Intersection Observer
- **Debounce Updates**: Throttle real-time updates to avoid excessive re-renders

### 2. Security Considerations

- **Validate All Inputs**: Use Zod schemas for API request validation
- **Rate Limiting**: Implement rate limits on analytics endpoints
- **Authentication**: Require authentication for all analytics endpoints
- **RBAC**: Implement role-based access control for team dashboards

### 3. Scalability

- **Time-Series Database**: Use InfluxDB or TimescaleDB for efficient time-series storage
- **Aggregation**: Pre-aggregate data at multiple granularities (hour, day, week)
- **Pagination**: Implement pagination for large datasets
- **CDN Caching**: Cache static analytics reports on CDN

### 4. User Experience

- **Loading States**: Show skeleton loaders while data loads
- **Empty States**: Provide helpful messages when no data is available
- **Error Recovery**: Gracefully handle API failures with retry mechanisms
- **Responsive Design**: Ensure all charts and tables work on mobile devices
