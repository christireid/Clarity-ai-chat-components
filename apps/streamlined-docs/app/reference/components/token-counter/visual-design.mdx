## Accessibility

TokenCounter is built with WCAG 2.1 AA compliance as a core requirement.

### Color Contrast

All colors meet minimum contrast ratios:

```tsx
// Test color contrast programmatically
export function ContrastTester() {
  const testContrast = (color1: string, color2: string) => {
    // Use contrast-checker library
    const ratio = calculateContrastRatio(color1, color2)
    const meetsAA = ratio >= 4.5 // WCAG AA for normal text
    const meetsAAA = ratio >= 7.0 // WCAG AAA for normal text

    return { ratio, meetsAA, meetsAAA }
  }

  return (
    <div className="space-y-4">
      {/* Safe state - Green on background */}
      <div className="p-4 bg-background border rounded-lg">
        <div className="text-[hsl(var(--success))]">
          ✓ Safe: 3,250 / 4,096 tokens (contrast: 4.8:1)
        </div>
      </div>

      {/* Warning state - Amber on background */}
      <div className="p-4 bg-background border rounded-lg">
        <div className="text-[hsl(var(--warning))]">
          ⚠ Warning: 3,400 / 4,096 tokens (contrast: 5.2:1)
        </div>
      </div>

      {/* Critical state - Red on background */}
      <div className="p-4 bg-background border rounded-lg">
        <div className="text-destructive">
          ⚠ Critical: 3,900 / 4,096 tokens (contrast: 6.1:1)
        </div>
      </div>
    </div>
  )
}
```

### Screen Reader Announcements

Enhanced ARIA support for status updates:

```tsx
export function ScreenReaderOptimized() {
  const [currentTokens, setCurrentTokens] = useState(2500)
  const maxTokens = 4096
  const percentage = (currentTokens / maxTokens) * 100

  const getStatus = () => {
    if (percentage >= 95) return 'critical'
    if (percentage >= 80) return 'warning'
    return 'safe'
  }

  return (
    <div>
      {/* Enhanced ARIA labels */}
      <TokenCounter
        currentTokens={currentTokens}
        maxTokens={maxTokens}
        className="[&_[role='status']]:sr-only"
      />

      {/* Live region for dynamic updates */}
      <div
        role="status"
        aria-live="polite"
        aria-atomic="true"
        className="sr-only"
      >
        Token usage: {currentTokens} of {maxTokens} tokens.
        {percentage.toFixed(1)}% used.
        Status: {getStatus()}.
        {percentage >= 80 && 'Approaching token limit.'}
        {percentage >= 95 && 'Critical token limit reached.'}
      </div>
    </div>
  )
}
```

export function ARIAComplete() {
  const currentTokens = 3400
  const maxTokens = 4096
  const percentage = (currentTokens / maxTokens) * 100

  return (
    <div
      role="region"
      aria-labelledby="token-usage-heading"
      className="space-y-4"
    >
      <h3 id="token-usage-heading" className="font-semibold">
        Token Usage Monitor
      </h3>

      <div
        role="status"
        aria-label={`Token usage: ${currentTokens} of ${maxTokens} tokens (${percentage.toFixed(1)}%)`}
      >
        {/* Progress bar with full ARIA */}
        <div className="relative w-full bg-muted rounded-full overflow-hidden">
          <div
            className="h-2 bg-[hsl(var(--warning))] transition-all"
            style={{ width: `${percentage}%` }}
            role="progressbar"
            aria-valuenow={currentTokens}
            aria-valuemin={0}
            aria-valuemax={maxTokens}
            aria-label={`${percentage.toFixed(1)}% of token limit used`}
          />
        </div>

        {/* Alert with proper role */}
        {percentage >= 80 && (
          <div
            role="alert"
            aria-live="assertive"
            className="mt-3 p-3 bg-[hsl(var(--warning))]/10 border border-[hsl(var(--warning))]/20 rounded-lg"
          >
            <strong>Approaching Context Limit</strong>
            <p className="text-sm mt-1">
              You're using {percentage.toFixed(1)}% of the context window.
            </p>
          </div>
        )}

        {/* Visually hidden description for screen readers */}
        <p className="sr-only">
          Current token count: {currentTokens.toLocaleString()}.
          Maximum tokens: {maxTokens.toLocaleString()}.
          Percentage used: {percentage.toFixed(1)}%.
          {percentage >= 95 && 'Critical: Approaching maximum token limit.'}
          {percentage >= 80 && percentage < 95 && 'Warning: High token usage.'}
          {percentage < 80 && 'Normal token usage.'}
        </p>
      </div>
    </div>
  )
}
