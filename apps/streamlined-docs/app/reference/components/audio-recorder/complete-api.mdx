## TypeScript Interfaces

### AudioRecorderProps

Complete TypeScript interface with detailed property descriptions:

```typescript
interface AudioRecorderProps {
  // ============================================================================
  // Recording Settings
  // ============================================================================

  /**
   * Maximum recording duration in seconds
   * @default 300
   * @range 1-7200 (2 hours recommended maximum)
   * @remarks Recording automatically stops when this limit is reached
   */
  maxDuration?: number

  /**
   * Minimum recording duration in seconds before stop is allowed
   * @default 1
   * @range 0-60
   * @remarks Stop button is disabled until this duration is reached
   */
  minDuration?: number

  /**
   * Automatically start recording when component mounts
   * @default false
   * @requires User gesture for browser security policies
   * @remarks May be blocked by browser if not triggered by user action
   */
  autoStart?: boolean

  /**
   * Enable pause/resume functionality during recording
   * @default true
   * @remarks Shows pause/resume buttons in UI when enabled
   */
  pausable?: boolean

  /**
   * Show countdown timer before recording starts
   * @default 0
   * @range 0-10
   * @remarks Set to 0 to disable countdown
   */
  countdownDuration?: number

  // ============================================================================
  // Format Options
  // ============================================================================

  /**
   * Desired output audio format
   * @default 'webm'
   * @remarks Actual format depends on browser support. Falls back to webm if unsupported.
   *
   * Format recommendations:
   * - webm: Best cross-browser compatibility, good compression
   * - mp3: Safari preferred, smaller file sizes
   * - wav: Uncompressed, largest files, best quality
   * - ogg: Good compression, limited Safari support
   * - flac: Lossless compression, limited browser support
   */
  outputFormat?: 'mp3' | 'wav' | 'ogg' | 'webm' | 'flac'

  /**
   * Override MIME type for MediaRecorder
   * @default undefined (auto-detected)
   * @example 'audio/webm;codecs=opus'
   * @remarks Use MediaRecorder.isTypeSupported() to verify support
   */
  mimeType?: string

  /**
   * Audio bitrate in bits per second
   * @default 128000 (128 kbps)
   * @range 64000-320000
   * @remarks Higher bitrate = better quality but larger files
   *
   * Recommended values:
   * - 64000: Voice, minimal quality
   * - 128000: Voice, standard quality
   * - 192000: Music, good quality
   * - 256000: Music, high quality
   */
  bitrate?: number

  /**
   * Audio sample rate in Hz
   * @default undefined (device default, typically 48000)
   * @range 8000-96000
   * @remarks Higher sample rate = better quality. Common values:
   * - 8000: Telephone quality
   * - 16000: VoIP quality
   * - 44100: CD quality
   * - 48000: Professional audio
   * - 96000: High-resolution audio
   */
  sampleRate?: number

  /**
   * Number of audio channels
   * @default 1 (mono)
   * @remarks
   * - 1 (mono): Recommended for voice recording, smaller files
   * - 2 (stereo): Recommended for music recording
   */
  channels?: 1 | 2

  // ============================================================================
  // Audio Processing
  // ============================================================================

  /**
   * Enable Web Audio API noise cancellation
   * @default false
   * @remarks Reduces background noise for clearer recordings.
   * Uses browser's built-in noise suppression algorithms.
   */
  enableNoiseCancellation?: boolean

  /**
   * Enable echo cancellation for clearer recordings
   * @default false
   * @remarks Useful in rooms with echo or when using speakers
   * May reduce audio quality slightly
   */
  enableEchoCancellation?: boolean

  /**
   * Automatically adjust input gain to maintain consistent volume
   * @default false
   * @remarks Normalizes volume levels, useful for varying speaker volumes
   */
  enableAutoGainControl?: boolean

  /**
   * Enable noise suppression using MediaStream constraints
   * @default false
   * @remarks Alternative to enableNoiseCancellation with different algorithm
   * Can be used together for maximum noise reduction
   */
  noiseSuppression?: boolean

  /**
   * Automatically pause recording during silence
   * @default false
   * @remarks Saves space by not recording silent periods
   * Uses amplitude threshold detection (see silenceThreshold)
   */
  voiceActivityDetection?: boolean

  /**
   * Amplitude threshold for silence detection (0-1)
   * @default 0.01
   * @range 0-1
   * @remarks Used with voiceActivityDetection
   * - Lower values = more sensitive (pauses less often)
   * - Higher values = less sensitive (pauses more often)
   * - Typical range: 0.005-0.05
   */
  silenceThreshold?: number

  // ============================================================================
  // Callbacks
  // ============================================================================

  /**
   * Callback when recording starts successfully
   * @remarks Called after microphone permission is granted and recording begins
   */
  onStart?: () => void

  /**
   * Callback when recording stops
   * @param audioBlob - Complete audio recording as Blob
   * @param audioUrl - Object URL for immediate playback (blob:// URL)
   * @remarks Remember to revoke object URL when done: URL.revokeObjectURL(audioUrl)
   */
  onStop?: (audioBlob: Blob, audioUrl: string) => void

  /**
   * Callback when recording is paused
   * @remarks Only called if pausable is true
   */
  onPause?: () => void

  /**
   * Callback when recording is resumed after being paused
   * @remarks Only called if pausable is true
   */
  onResume?: () => void

  /**
   * Callback when audio data chunks become available during recording
   * @param data - Audio data chunk as Blob
   * @remarks Useful for real-time processing or incremental uploads
   * Called periodically during recording (typically every few seconds)
   */
  onDataAvailable?: (data: Blob) => void

  /**
   * Callback when an error occurs
   * @param error - Error object with message and stack
   * @remarks Common error scenarios:
   * - Permission denied: User denied microphone access
   * - Unsupported format: Browser doesn't support requested format
   * - MediaRecorder error: Recording device unavailable or busy
   * - Security error: Not using HTTPS or invalid context
   */
  onError?: (error: Error) => void

  /**
   * Callback fired every second with current recording duration
   * @param duration - Current duration in seconds
   * @remarks Useful for custom UI or analytics
   */
  onDurationChange?: (duration: number) => void

  /**
   * Callback with current audio amplitude (0-1)
   * @param amplitude - Normalized amplitude value (0 = silence, 1 = maximum)
   * @remarks Fired continuously during recording for custom waveform visualization
   * Uses RMS (Root Mean Square) calculation for accurate amplitude
   */
  onAmplitudeChange?: (amplitude: number) => void

  // ============================================================================
  // UI Configuration
  // ============================================================================

  /**
   * Display real-time waveform visualization
   * @default true
   * @remarks Shows 60-bar animated waveform display
   * Disable for better performance if not needed
   */
  showWaveform?: boolean

  /**
   * Display recording duration timer
   * @default true
   * @remarks Shows duration in MM:SS format
   */
  showDuration?: boolean

  /**
   * Show built-in recording controls
   * @default true
   * @remarks Includes start, stop, pause/resume buttons
   * Set to false for custom control implementation
   */
  showControls?: boolean

  /**
   * Display audio input level meter
   * @default true
   * @remarks Shows real-time amplitude with color gradient (green to red)
   */
  showAmplitudeMeter?: boolean

  /**
   * Additional CSS classes for the container element
   * @remarks Applied to the outermost div wrapper
   */
  className?: string

  /**
   * Color theme for the recorder UI
   * @default 'auto'
   * @remarks 'auto' follows system preference
   */
  theme?: 'light' | 'dark' | 'auto'

  /**
   * Disable recording functionality
   * @default false
   * @remarks Start button will be disabled when true
   */
  disabled?: boolean
}
```

## Edge Cases & Error Handling

### 1. Permission Denied

**Scenario**: User denies microphone access or browser blocks permission.

**Error**: `NotAllowedError: Permission denied`

**Handling**:

```tsx
import { AudioRecorder } from '@clarity-chat/react'
import { useState } from 'react'

function PermissionHandling() {
  const [permissionError, setPermissionError] = useState<string | null>(null)

  const handleError = (error: Error) => {
    if (error.name === 'NotAllowedError' || error.message.includes('Permission denied')) {
      setPermissionError(
        'Microphone access denied. Please allow microphone access in your browser settings.'
      )

      // Show browser-specific instructions
      const isChrome = /Chrome/.test(navigator.userAgent)
      const isFirefox = /Firefox/.test(navigator.userAgent)
      const isSafari = /Safari/.test(navigator.userAgent) && !/Chrome/.test(navigator.userAgent)

      if (isChrome) {
        console.log('Chrome: Click the camera icon in the address bar')
      } else if (isFirefox) {
        console.log('Firefox: Click the microphone icon in the address bar')
      } else if (isSafari) {
        console.log('Safari: Safari → Settings → Websites → Microphone')
      }
    }
  }

  return (
    <div>
      <AudioRecorder
        onStart={() => setPermissionError(null)}
        onError={handleError}
        onStop={(blob, url) => console.log('Recording saved')}
      />

      {permissionError && (
        <div className="mt-4 p-4 bg-red-50 border border-red-200 rounded-lg">
          <p className="text-sm text-red-700">{permissionError}</p>
          <button
            onClick={() => window.location.reload()}
            className="mt-2 text-sm text-red-600 underline"
          >
            Try Again
          </button>
        </div>
      )}
    </div>
  )
}
```

**Prevention**:

```tsx
// Check permission before rendering
async function checkMicrophonePermission(): Promise<PermissionState> {
  try {
    const result = await navigator.permissions.query({ name: 'microphone' as PermissionName })
    return result.state // 'granted', 'denied', or 'prompt'
  } catch (error) {
    return 'prompt' // Fallback if Permissions API not supported
  }
}

function RecorderWithPermissionCheck() {
  const [permissionState, setPermissionState] = useState<PermissionState>('prompt')

  useEffect(() => {
    checkMicrophonePermission().then(setPermissionState)
  }, [])

  if (permissionState === 'denied') {
    return (
      <div className="p-4 bg-yellow-50 border border-yellow-200 rounded-lg">
        <p>Microphone access is blocked. Please enable it in your browser settings.</p>
      </div>
    )
  }

  return <AudioRecorder onStop={handleStop} />
}
```

### 2. Unsupported Format

**Scenario**: Browser doesn't support the requested audio format.

**Error**: `NotSupportedError: The MIME type is not supported`

**Handling**:

```tsx
function FormatFallbackHandling() {
  const [selectedFormat, setSelectedFormat] = useState<'webm' | 'mp4' | 'wav'>('webm')

  // Detect best supported format
  useEffect(() => {
    const detectFormat = () => {
      if (MediaRecorder.isTypeSupported('audio/webm;codecs=opus')) {
        return 'webm'
      } else if (MediaRecorder.isTypeSupported('audio/mp4')) {
        return 'mp4'
      } else {
        return 'wav'
      }
    }

    setSelectedFormat(detectFormat())
  }, [])

  const handleError = (error: Error) => {
    if (error.name === 'NotSupportedError' || error.message.includes('MIME type')) {
      console.error('Format not supported, falling back to:', selectedFormat)

      // Try alternative formats
      const fallbackFormats: Array<'webm' | 'mp4' | 'wav'> = ['webm', 'mp4', 'wav']
      const nextFormat = fallbackFormats.find(fmt =>
        fmt !== selectedFormat &&
        MediaRecorder.isTypeSupported(`audio/${fmt}`)
      )

      if (nextFormat) {
        setSelectedFormat(nextFormat)
      }
    }
  }

  return (
    <div>
      <p className="text-sm text-gray-600 mb-2">
        Using format: {selectedFormat}
      </p>
      <AudioRecorder
        outputFormat={selectedFormat}
        onError={handleError}
        onStop={(blob) => {
          console.log(`Recording saved as ${blob.type}`)
        }}
      />
    </div>
  )
}
```

**Component Built-in Fallback**:

```tsx
// AudioRecorder automatically handles format fallback
<AudioRecorder
  outputFormat="flac"  // Requested format
  // If FLAC is not supported:
  // 1. Tries audio/webm;codecs=opus
  // 2. Falls back to audio/webm
  // No error thrown, recording still works
  onStop={(blob, url) => {
    // blob.type will reflect actual format used
    console.log('Actual format:', blob.type)
  }}
/>
```

### 3. Microphone Busy

**Scenario**: Another application or tab is using the microphone.

**Error**: `NotReadableError: Could not start audio source`

**Handling**:

```tsx
function MicrophoneBusyHandling() {
  const handleError = (error: Error) => {
    if (error.name === 'NotReadableError' || error.message.includes('audio source')) {
      // Microphone is busy
      alert(
        'Microphone is currently in use by another application. ' +
        'Please close other apps using the microphone and try again.'
      )

      // Check if mic is available
      navigator.mediaDevices.enumerateDevices().then(devices => {
        const microphones = devices.filter(d => d.kind === 'audioinput')
        console.log('Available microphones:', microphones)

        if (microphones.length === 0) {
          console.error('No microphone devices found')
        }
      })
    }
  }

  return (
    <AudioRecorder
      onError={handleError}
      onStop={(blob, url) => console.log('Recording saved')}
    />
  )
}
```

```tsx
      </div>

  return <AudioRecorder onStop={handleStop} />
}
```

### 5. Browser Not Supported

**Scenario**: Old browser without MediaRecorder API support.

**Handling**:

```tsx
function BrowserSupportCheck() {
  const [isSupported, setIsSupported] = useState(true)
  const [browserInfo, setBrowserInfo] = useState({ name: '', version: '' })

  useEffect(() => {
    const supported =
      typeof window !== 'undefined' &&
      'mediaDevices' in navigator &&
      'MediaRecorder' in window

    setIsSupported(supported)

    // Detect browser
    const ua = navigator.userAgent
    let name = 'Unknown'
    let version = ''

    if (ua.includes('Chrome')) {
      name = 'Chrome'
      version = ua.match(/Chrome\/(\d+)/)?.[1] || ''
    } else if (ua.includes('Firefox')) {
      name = 'Firefox'
      version = ua.match(/Firefox\/(\d+)/)?.[1] || ''
    } else if (ua.includes('Safari') && !ua.includes('Chrome')) {
      name = 'Safari'
      version = ua.match(/Version\/(\d+)/)?.[1] || ''
    }

    setBrowserInfo({ name, version })
  }, [])

  if (!isSupported) {
    return (
      <div className="p-4 bg-orange-50 border border-orange-200 rounded-lg">
        <h3 className="font-semibold text-orange-800">Browser Not Supported</h3>
        <p className="text-sm text-orange-700 mt-2">
          Your browser ({browserInfo.name} {browserInfo.version}) doesn't support audio recording.
        </p>
        <p className="text-sm text-orange-700 mt-2">
          Please update to a modern browser:
```

```tsx
  const [isTranscribing, setIsTranscribing] = useState(false)
  const [silenceDetected, setSilenceDetected] = useState(false)

  const handleDataAvailable = async (chunk: Blob) => {
    // Send chunk to transcription API in real-time
    try {
      const formData = new FormData()
      formData.append('audio', chunk)

      const response = await fetch('/api/transcribe-chunk', {
        method: 'POST',
        body: formData,
      })

      const { text } = await response.json()

      if (text.trim()) {
        setTranscript(prev => [...prev, text])
      }
    } catch (error) {
      console.error('Transcription error:', error)
    }
  }

  const handleAmplitudeChange = (amplitude: number) => {
    // Visual feedback for silence detection
    setSilenceDetected(amplitude < 0.02)
  }

  const handleStop = async (audioBlob: Blob, audioUrl: string) => {
    setIsTranscribing(true)

    // Final transcription pass
    const formData = new FormData()
    formData.append('audio', audioBlob)

    try {
      const response = await fetch('/api/transcribe-final', {
        method: 'POST',
        body: formData,
      })

      const { fullTranscript } = await response.json()
      setTranscript([fullTranscript])
    } catch (error) {
      console.error('Final transcription error:', error)
    } finally {
      setIsTranscribing(false)
    }
  }

  const copyTranscript = () => {
    const text = transcript.join(' ')
    navigator.clipboard.writeText(text)
  }

  return (
    <div className="max-w-3xl mx-auto p-6">
      <h2 className="text-2xl font-bold mb-6">Speech-to-Text Transcription</h2>

      <div className="mb-6">
        <AudioRecorder
          maxDuration={300}             // 5 minutes
          outputFormat="webm"
          bitrate={128000}
          channels={1}
          sampleRate={16000}            // Optimized for speech
          enableNoiseCancellation={true}
          enableEchoCancellation={true}
          voiceActivityDetection={true}
          silenceThreshold={0.02}
          showWaveform={true}
          showAmplitudeMeter={true}
          onDataAvailable={handleDataAvailable}
          onAmplitudeChange={handleAmplitudeChange}
          onStop={handleStop}
        />

        {silenceDetected && (
          <div className="mt-2 text-sm text-amber-600">
            🔇 Silence detected - recording paused
          </div>
        )}
      </div>

      {/* Transcript Display */}
      <div className="border rounded-lg p-4 bg-white shadow">
        <div className="flex items-center justify-between mb-4">
          <h3 className="text-lg font-semibold">Transcript</h3>
          <button
            onClick={copyTranscript}
            disabled={transcript.length === 0}
            className="px-3 py-1 text-sm bg-blue-500 text-white rounded hover:bg-blue-600 disabled:opacity-50"
          >
            Copy
          </button>
        </div>

        {isTranscribing && (
          <div className="text-center py-8">
            <div className="inline-block animate-spin rounded-full h-8 w-8 border-4 border-gray-200 border-t-blue-500"></div>
            <p className="mt-2 text-sm text-gray-600">Finalizing transcription...</p>
          </div>
        )}

        {!isTranscribing && transcript.length === 0 && (
          <p className="text-gray-500 italic">Start recording to see transcript...</p>
        )}

        {!isTranscribing && transcript.length > 0 && (
          <div className="space-y-2">
            {transcript.map((text, index) => (
              <p key={index} className="text-gray-800">
                {text}
              </p>
            ))}
          </div>
        )}
      </div>

      {/* Word Count */}
      {transcript.length > 0 && (
        <div className="mt-4 text-sm text-gray-600">
          Word count: {transcript.join(' ').split(/\s+/).length}
        </div>
      )}
    </div>
  )
}

export default TranscriptionRecorder
```

### Example 5: Accessibility-Enhanced Recorder

Fully accessible recorder with comprehensive ARIA support:

```tsx
import { AudioRecorder } from '@clarity-chat/react'
import { useState, useEffect } from 'react'

function AccessibleRecorder() {
  const [status, setStatus] = useState('Ready to record')
  const [duration, setDuration] = useState(0)
  const [error, setError] = useState<string | null>(null)
  const [recordings, setRecordings] = useState<Array<{
    id: string
    url: string
    duration: number
    timestamp: Date
  }>>([])

  // Screen reader announcements
  const announce = (message: string) => {
    setStatus(message)

    // Also log for debugging
    console.log('[A11Y]', message)
  }

  const formatDuration = (seconds: number) => {
    const mins = Math.floor(seconds / 60)
    const secs = seconds % 60
    return `${mins} minute${mins !== 1 ? 's' : ''} ${secs} second${secs !== 1 ? 's' : ''}`
  }

  const handleStart = () => {
    announce('Recording started')
    setError(null)
  }

  const handlePause = () => {
    announce('Recording paused')
  }

  const handleResume = () => {
    announce('Recording resumed')
  }

  const handleStop = (audioBlob: Blob, audioUrl: string) => {
    const recording = {
      id: Date.now().toString(),
      url: audioUrl,
      duration,
      timestamp: new Date(),
    }

    setRecordings(prev => [...prev, recording])
    announce(`Recording stopped. Duration: ${formatDuration(duration)}`)
    setDuration(0)
  }

  const handleError = (err: Error) => {
    let message = 'An error occurred. Please try again.'

    if (err.name === 'NotAllowedError') {
      message = 'Microphone access denied. Please allow microphone access in your browser settings.'
    } else if (err.name === 'NotFoundError') {
      message = 'No microphone found. Please connect a microphone and try again.'
    } else if (err.name === 'NotReadableError') {
      message = 'Microphone is busy. Please close other applications using the microphone.'
    }

    setError(message)
    announce(`Error: ${message}`)
  }

  const deleteRecording = (id: string) => {
    const recording = recordings.find(r => r.id === id)
    if (recording) {
      URL.revokeObjectURL(recording.url)
      setRecordings(prev => prev.filter(r => r.id !== id))
      announce('Recording deleted')
    }
  }

  return (
    <div className="max-w-2xl mx-auto p-6">
      <h1 className="text-2xl font-bold mb-6">Accessible Audio Recorder</h1>

      {/* Skip to recordings link */}
      <a
        href="#recordings-list"
        className="sr-only focus:not-sr-only focus:absolute focus:top-4 focus:left-4 focus:z-50 focus:bg-blue-500 focus:text-white focus:px-4 focus:py-2 focus:rounded"
      >
        Skip to recordings list
      </a>

      {/* Instructions */}
      <div
        id="recorder-instructions"
        className="mb-6 p-4 bg-blue-50 border-l-4 border-blue-500"
        role="region"
        aria-label="Recording instructions"
      >
        <h2 className="font-semibold text-blue-900 mb-2">Instructions</h2>
        <ul className="text-sm text-blue-800 space-y-1">
          <li>• Press Enter or Space to start recording</li>
          <li>• Use Tab to navigate between controls</li>
          <li>• Press Escape to stop recording</li>
          <li>• Status updates are announced to screen readers</li>
        </ul>
      </div>

      {/* Error Display */}
      {error && (
        <div
          role="alert"
          aria-live="assertive"
          className="mb-6 p-4 bg-red-50 border-l-4 border-red-500"
        >
          <h3 className="font-semibold text-red-800">Error</h3>
          <p className="text-sm text-red-700 mt-1">{error}</p>
        </div>
      )}

      {/* Recorder */}
      <div
        role="region"
        aria-label="Audio recorder"
        aria-describedby="recorder-instructions"
      >
        <AudioRecorder
          maxDuration={180}
          minDuration={1}
          outputFormat="webm"
          bitrate={128000}
          channels={1}
          enableNoiseCancellation={true}
          pausable={true}
          showWaveform={true}
          showDuration={true}
          showAmplitudeMeter={true}
          onStart={handleStart}
          onPause={handlePause}
          onResume={handleResume}
          onStop={handleStop}
          onError={handleError}
          onDurationChange={setDuration}
        />
      </div>

      {/* Live Region for Status */}
      <div
        role="status"
        aria-live="polite"
        aria-atomic="true"
        className="sr-only"
      >
        {status}
      </div>

      {/* Recordings List */}
      {recordings.length > 0 && (
        <div
          id="recordings-list"
          role="region"
          aria-label="Saved recordings"
          className="mt-8"
        >
          <h2 className="text-xl font-semibold mb-4">
            Saved Recordings ({recordings.length})
          </h2>

          <ul className="space-y-4">
            {recordings.map((recording, index) => (
              <li
                key={recording.id}
                className="p-4 bg-white border rounded-lg shadow-sm"
              >
                <div className="flex items-start justify-between mb-3">
                  <div>
                    <h3 className="font-medium">
                      Recording {index + 1}
                    </h3>
                    <p className="text-sm text-gray-600">
                      {recording.timestamp.toLocaleString()}
                    </p>
                    <p className="text-sm text-gray-600">
                      Duration: {formatDuration(recording.duration)}
                    </p>
                  </div>
                  <button
                    onClick={() => deleteRecording(recording.id)}
                    aria-label={`Delete recording ${index + 1}`}
                    className="px-3 py-1 text-sm text-red-600 hover:bg-red-50 rounded focus:outline-none focus-visible:ring-2 focus-visible:ring-red-500"
                  >
                    Delete
                  </button>
                </div>

                <audio
                  controls
                  src={recording.url}
                  aria-label={`Play recording ${index + 1}`}
                  className="w-full"
                />
              </li>
            ))}
          </ul>
        </div>
      )}

      {recordings.length === 0 && (
        <div className="mt-8 text-center text-gray-500">
          <p>No recordings yet. Start recording to create your first audio file.</p>
        </div>
      )}
    </div>
  )
}

export default AccessibleRecorder
```
