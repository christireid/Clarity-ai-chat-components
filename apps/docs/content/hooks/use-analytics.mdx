# useAnalytics

A React hook for component usage tracking, performance monitoring, and interaction analytics.

## Import

```typescript
import { useAnalytics } from '@clarity-chat/react'
```

## Usage

### Basic Component Tracking

```tsx
import { useAnalytics } from '@clarity-chat/react'

function MyComponent() {
  const { trackInteraction } = useAnalytics('MyComponent')

  return (
    <button onClick={() => trackInteraction('click', 'submit-button')}>
      Click me
    </button>
  )
}
```

### Performance Monitoring

```tsx
function DataTable({ data }) {
  const { trackPerformance } = useAnalytics('DataTable')

  React.useEffect(() => {
    const startTime = performance.now()
    // Process data...
    const duration = performance.now() - startTime
    trackPerformance('render', { duration, itemCount: data.length })
  }, [data])

  return <table>...</table>
}
```

### Custom Analytics Configuration

```tsx
import { AnalyticsProvider } from '@clarity-chat/react'

function App() {
  return (
    <AnalyticsProvider config={{
      enabled: true,
      sampleRate: 1.0, // Track 100% of events
      batchEvents: true,
      onEvent: (event) => {
        console.log('Analytics event:', event)
        // Send to your analytics service
      }
    }}>
      <MyApp />
    </AnalyticsProvider>
  )
}
```

## API Reference

### useAnalytics(componentName, trackUsage?)

Returns an object with analytics tracking methods.

#### Parameters

- `componentName` (string): Identifier for the component being tracked
- `trackUsage` (boolean, optional): Whether to track mount/unmount events (default: true)

#### Returns

```typescript
{
  trackInteraction: (interactionType: string, element?: string, metadata?: Record<string, any>) => void
  trackPerformance: (category: PerformanceEvent['category'], metadata: Partial<PerformanceEvent['metadata']>) => void
  renderCount: number
}
```

### AnalyticsProvider

Context provider for analytics configuration.

#### Props

```typescript
interface AnalyticsProviderProps {
  children: React.ReactNode
  config?: Partial<AnalyticsConfig>
}
```

### AnalyticsConfig

```typescript
interface AnalyticsConfig {
  enabled?: boolean           // Enable/disable analytics
  endpoint?: string          // Analytics endpoint URL
  batchEvents?: boolean      // Batch events before sending
  batchSize?: number         // Events per batch
  flushInterval?: number     // Auto-flush interval (ms)
  sampleRate?: number        // Sampling rate (0-1)
  anonymize?: boolean        // Anonymize user data
  trackPerformance?: boolean // Track performance metrics
  trackInteractions?: boolean // Track user interactions
  trackComponents?: boolean  // Track component usage
  onEvent?: (event: AnalyticsEvent) => void  // Custom event handler
  onError?: (error: Error) => void           // Error handler
}
```

## Event Types

### ComponentUsageEvent
```typescript
{
  type: 'component_usage'
  component: string
  action: 'mount' | 'unmount' | 'render' | 'interaction'
  metadata: {
    props?: Record<string, any>
    renderTime?: number
    interactionType?: string
    element?: string
  }
}
```

### PerformanceEvent
```typescript
{
  type: 'performance'
  component: string
  category: 'render' | 'interaction' | 'memory' | 'network'
  metadata: {
    duration?: number
    memoryUsage?: number
    renderCount?: number
    interactionLatency?: number
  }
}
```

### ErrorEvent
```typescript
{
  type: 'error'
  component?: string
  error: {
    name: string
    message: string
    stack?: string
  }
  context?: Record<string, any>
}
```

## Best Practices

### 1. Component Naming
Use consistent, descriptive names for components:

```tsx
// Good
const { trackInteraction } = useAnalytics('UserProfileCard')

// Avoid
const { trackInteraction } = useAnalytics('Component123')
```

### 2. Event Metadata
Include relevant context in event metadata:

```tsx
trackInteraction('click', 'save-button', {
  formId: 'user-settings',
  hasUnsavedChanges: true,
  userRole: 'admin'
})
```

### 3. Performance Tracking
Track meaningful performance metrics:

```tsx
trackPerformance('render', {
  duration: renderTime,
  itemCount: items.length,
  isInitialLoad: isFirstRender
})
```

### 4. Privacy Considerations
Respect user privacy by:
- Using appropriate sampling rates
- Anonymizing sensitive data
- Providing opt-out mechanisms
- Complying with privacy regulations
