## Custom Keyboard Shortcuts

The CommandPalette integrates with the `useCommandPalette` hook to provide flexible keyboard shortcut registration.

### Basic Shortcut Registration

```tsx
import { CommandPalette, useCommandPalette } from '@clarity-chat/react'

function MyApp() {
  // Default: Cmd+K (Mac) or Ctrl+K (Windows/Linux)
  const { isOpen, close, shortcutDisplay } = useCommandPalette()

  return (
    <>
      <button onClick={() => setIsOpen(true)}>
        Open Palette ({shortcutDisplay})
      </button>
      <CommandPalette
        items={commands}
        open={isOpen}
        onClose={close}
      />
    </>
  )
}
```

### Custom Shortcut Keys

```tsx
// Use Cmd+P instead of Cmd+K
const { isOpen, close } = useCommandPalette({
  shortcut: 'mod+p', // 'mod' = Cmd on Mac, Ctrl on Windows/Linux
})

// Use Ctrl+Shift+P (regardless of platform)
const { isOpen, close } = useCommandPalette({
  shortcut: 'ctrl+shift+p',
})

// Multiple shortcuts for the same action
// Note: useCommandPalette only supports one shortcut

const [isOpen, setIsOpen] = useState(false)

useKeyboardShortcuts([
  {
    key: 'mod+k',
    callback: () => setIsOpen(true),
    description: 'Open command palette',
  },
  {
    key: 'mod+p',
    callback: () => setIsOpen(true),
    description: 'Open command palette (alt)',
  },
])
```

### Advanced Ranking Algorithm

```tsx
interface ScoredCommand extends CommandItem {
  score: number
}

function calculateRelevanceScore(
  command: CommandItem,
  query: string,
  options: {
    recentCommands?: string[]
    favoriteCommands?: string[]
  }
): number {
  const lowerQuery = query.toLowerCase()
  const lowerLabel = command.label.toLowerCase()
  const lowerDesc = (command.description || '').toLowerCase()

  let score = 0

  // Exact match (highest priority)
  if (lowerLabel === lowerQuery) {
    score += 100
  }

  // Starts with query
  if (lowerLabel.startsWith(lowerQuery)) {
    score += 80
  }

  // Word boundary match
  if (new RegExp(`\\b${lowerQuery.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')}`).test(lowerLabel)) {
    score += 60
  }

  // Contains query
  if (lowerLabel.includes(lowerQuery)) {
    score += 40
  }

  // Description match (lower weight)
  if (lowerDesc.includes(lowerQuery)) {
    score += 20
  }

  // Position bonus (earlier = better)
  const position = lowerLabel.indexOf(lowerQuery)
  if (position !== -1) {
    score += Math.max(0, 20 - position)
  }

  // Recent command boost
  if (options.recentCommands?.includes(command.id)) {
    score += 15
  }

  // Favorite command boost
  if (options.favoriteCommands?.includes(command.id)) {
    score += 10
  }

  return score
}

function useRankedCommands(
  commands: CommandItem[],
  query: string,
  options: { recentCommands?: string[]; favoriteCommands?: string[] }
) {
  return useMemo(() => {
    if (!query) return commands

    const scored: ScoredCommand[] = commands
      .map(cmd => ({
        ...cmd,
        score: calculateRelevanceScore(cmd, query, options),
      }))
      .filter(cmd => cmd.score > 0)
      .sort((a, b) => b.score - a.score)

    return scored
  }, [commands, query, options])
}
```

### Fuzzy Match with Highlighting

```tsx
function highlightMatch(text: string, query: string): React.ReactNode {
  if (!query.trim()) return text

  const lowerText = text.toLowerCase()
  const lowerQuery = query.toLowerCase()
  const index = lowerText.indexOf(lowerQuery)

  if (index === -1) return text

  const before = text.slice(0, index)
  const match = text.slice(index, index + query.length)
  const after = text.slice(index + query.length)

  return (
    <>
      {before}
      <mark className="bg-yellow-200 dark:bg-yellow-800/40 text-inherit font-semibold rounded px-0.5">
        {match}
      </mark>
      {highlightMatch(after, query)}
    </>
  )
}

// Use in command items
const enhancedCommands = filteredCommands.map(cmd => ({
  ...cmd,
  label: highlightMatch(cmd.label, search),
  description: cmd.description && highlightMatch(cmd.description, search),
}))
```

```tsx
function CommandPaletteWithFilters() {
```

## Complete Examples

### Example 1: Full-Featured Documentation Search

Complete implementation of a documentation search command palette with fuzzy search, category filtering, and recent pages.

```tsx
'use client'

import { useState, useEffect, useMemo } from 'react'
import { useRouter, usePathname } from 'next/navigation'
import {
  CommandPalette,
  useCommandPalette,
  useDebounce,
} from '@clarity-chat/react'
import {
  FileText,
  Box,
  Zap,
  BookOpen,
  Code,
  Clock,
  Sparkles,
} from 'lucide-react'

interface Page {
  title: string
  url: string
  description?: string
  category: string
}

const DOCS_PAGES: Page[] = [
  {
    title: 'Getting Started',
    url: '/docs/getting-started',
    description: 'Installation and setup guide',
    category: 'Guides',
  },
  {
    title: 'Chat Window',
    url: '/docs/components/chat-window',
    description: 'Main chat interface component',
    category: 'Components',
  },
  {
    title: 'useClarityChat',
    url: '/docs/hooks/use-clarity-chat',
    description: 'Core chat functionality hook',
    category: 'Hooks',
  },
  // ... more pages
]

export function DocsCommandPalette() {
  const router = useRouter()
  const pathname = usePathname()
  const { isOpen, close } = useCommandPalette()

  const [search, setSearch] = useState('')
  const [recentPages, setRecentPages] = useState<string[]>([])

  // Debounce search input
  const debouncedSearch = useDebounce(search, 150)

  // Load recent pages from localStorage
  useEffect(() => {
    const stored = localStorage.getItem('docs-recent-pages')
    if (stored) {
      try {
        setRecentPages(JSON.parse(stored))
      } catch {}
    }
  }, [])

  // Track current page
  useEffect(() => {
    if (pathname && pathname !== '/') {
      setRecentPages(prev => {
        const updated = [pathname, ...prev.filter(p => p !== pathname)].slice(0, 5)
        localStorage.setItem('docs-recent-pages', JSON.stringify(updated))
        return updated
      })
    }
  }, [pathname])

  // Search and rank commands
  const commands = useMemo(() => {
    const items: CommandItem[] = []

    // Calculate relevance scores
    const scored = DOCS_PAGES.map(page => {
      const lowerQuery = debouncedSearch.toLowerCase()
      const lowerTitle = page.title.toLowerCase()
      const lowerDesc = (page.description || '').toLowerCase()

      let score = 0

      // Scoring logic
      if (lowerTitle === lowerQuery) score += 100
      else if (lowerTitle.startsWith(lowerQuery)) score += 80
      else if (lowerTitle.includes(lowerQuery)) score += 40

      if (lowerDesc.includes(lowerQuery)) score += 20

      // Recent page boost
      if (recentPages.includes(page.url)) score += 15

      return { page, score }
    })
      .filter(({ score }) => !debouncedSearch || score > 0)
      .sort((a, b) => b.score - a.score)

    // Convert to command items
    scored.forEach(({ page }) => {
      items.push({
        id: page.url,
        label: page.title,
        description: page.description,
        category: page.category,
        icon: getCategoryIcon(page.category),
        onSelect: () => {
          router.push(page.url)
          close()
        },
      })
    })

    // Add recent pages section if no search
    if (!debouncedSearch && recentPages.length > 0) {
      recentPages.forEach(url => {
        const page = DOCS_PAGES.find(p => p.url === url)
        if (page) {
          items.unshift({
            id: `recent-${url}`,
            label: page.title,
            description: page.description,
            category: 'Recent',
            icon: <Clock className="w-4 h-4" />,
            onSelect: () => {
              router.push(url)
              close()
            },
          })
        }
      })
    }

    return items
  }, [debouncedSearch, recentPages, router, close])

  return (
    <CommandPalette
      items={commands}
      open={isOpen}
      onClose={close}
      placeholder="Search documentation..."
      loading={search !== debouncedSearch}
    />
  )
}

function getCategoryIcon(category: string): React.ReactNode {
  const icons: Record<string, React.ReactNode> = {
    Components: <Box className="w-4 h-4" />,
    Hooks: <Zap className="w-4 h-4" />,
    Guides: <FileText className="w-4 h-4" />,
    Examples: <Sparkles className="w-4 h-4" />,
    Cookbook: <BookOpen className="w-4 h-4" />,
    Features: <Code className="w-4 h-4" />,
  }
  return icons[category] || <FileText className="w-4 h-4" />
}
```

## Best Practices

### Performance

1. **Debounce search input** to prevent excessive filtering on rapid typing
2. **Memoize filtered commands** with `useMemo` to avoid unnecessary recalculations
3. **Lazy load command icons** for large command sets
4. **Use virtual scrolling** for 100+ commands

### Accessibility

1. **Always provide keyboard shortcuts** for critical commands
2. **Use semantic HTML** with proper ARIA attributes
3. **Ensure high contrast** for selected items (4.5:1 minimum)
4. **Respect reduced motion** preferences

### User Experience

1. **Show recent/frequent commands** at the top when no search query
2. **Highlight matching text** in search results
3. **Group related commands** in categories
4. **Provide visual feedback** for command execution
5. **Close palette automatically** after command execution

### Code Organization

1. **Separate command definitions** from component logic
2. **Use custom hooks** for complex command generation
3. **Type command items strictly** with TypeScript
4. **Document command purposes** with clear labels and descriptions
