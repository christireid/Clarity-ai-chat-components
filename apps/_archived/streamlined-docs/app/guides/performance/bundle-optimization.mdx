# Bundle Size Optimization

Reduce JavaScript bundle sizes by 40-60% through strategic code splitting, lazy loading, and dependency optimization.

## Overview

Bundle size directly impacts load time, Time to Interactive (TTI), and user experience. This guide provides proven strategies to minimize bundle sizes while maintaining functionality.

## Code Splitting Strategies

### Route-Based Code Splitting

Split code at route boundaries for optimal initial load performance.

```typescript
// apps/examples/code-assistant/src/routes.tsx
import { lazy, Suspense } from 'react';
import { Route, Routes } from 'react-router-dom';
import LoadingSpinner from './components/LoadingSpinner';

// Lazy load route components
const Home = lazy(() => import('./pages/Home'));
const Chat = lazy(() => import('./pages/Chat'));
const Settings = lazy(() => import('./pages/Settings'));
const Analytics = lazy(() => import('./pages/Analytics'));

// Heavy dependencies loaded only when needed
const TokenOptimization = lazy(() =>
  import('./pages/TokenOptimization')
);

const AudioRecorder = lazy(() =>
  import('./pages/AudioRecorder')
);

export function AppRoutes() {
  return (
    <Suspense fallback={<LoadingSpinner />}>
      <Routes>
        <Route path="/" element={<Home />} />
        <Route path="/chat" element={<Chat />} />
        <Route path="/settings" element={<Settings />} />
        <Route path="/analytics" element={<Analytics />} />
        <Route path="/tokens" element={<TokenOptimization />} />
        <Route path="/audio" element={<AudioRecorder />} />
      </Routes>
    </Suspense>
  );
}
```

### Component-Level Code Splitting

Lazy load heavy or conditionally rendered components.

```typescript
// packages/react/src/components/ChatInterface.tsx
import { lazy, Suspense, useState } from 'react';

// Always loaded - critical path
import { MessageList } from './MessageList';
import { ChatInput } from './ChatInput';

// Lazy loaded - optional features
const AudioRecorder = lazy(() =>
  import('./input/AudioRecorder').then(m => ({ default: m.AudioRecorder }))
);

const CommandPalette = lazy(() =>
  import('./navigation/CommandPalette').then(m => ({ default: m.CommandPalette }))
);

const TokenOptimizer = lazy(() =>
  import('./token/TokenOptimizer').then(m => ({ default: m.TokenOptimizer }))
);

export function ChatInterface() {
  const [showAudio, setShowAudio] = useState(false);
  const [showPalette, setShowPalette] = useState(false);
  const [showTokens, setShowTokens] = useState(false);

  return (
    <div className="chat-interface">
      <MessageList />
      <ChatInput onVoiceClick={() => setShowAudio(true)} />

      {/* Only load when user activates feature */}
      {showAudio && (
        <Suspense fallback={<div>Loading audio...</div>}>
          <AudioRecorder />
        </Suspense>
      )}

      {showPalette && (
        <Suspense fallback={null}>
          <CommandPalette />
        </Suspense>
      )}

      {showTokens && (
        <Suspense fallback={<div>Loading optimizer...</div>}>
          <TokenOptimizer />
        </Suspense>
      )}
    </div>
  );
}
```

### Dynamic Import Patterns

Advanced lazy loading with preloading and prefetching.

```typescript
// packages/react/src/utils/lazy-loading.ts

/**
 * Lazy load with preload support
 */
export function lazyWithPreload<T extends React.ComponentType<any>>(
  factory: () => Promise<{ default: T }>
) {
  const LazyComponent = lazy(factory);

  // Add preload method
  (LazyComponent as any).preload = factory;

  return LazyComponent as typeof LazyComponent & { preload: typeof factory };
}

// Usage example
const HeavyChart = lazyWithPreload(() =>
  import('./components/HeavyChart')
);

function Dashboard() {
  // Preload on hover
  const handleMouseEnter = () => {
    HeavyChart.preload();
  };

  return (
    <div onMouseEnter={handleMouseEnter}>
      <Suspense fallback={<ChartSkeleton />}>
        <HeavyChart />
      </Suspense>
    </div>
  );
}
```

```typescript
// packages/react/src/utils/prefetch.ts

/**
 * Prefetch chunks during idle time
 */
export function prefetchRoutes(routes: string[]) {
  if ('requestIdleCallback' in window) {
    requestIdleCallback(() => {
      routes.forEach(route => {
        const link = document.createElement('link');
        link.rel = 'prefetch';
        link.href = route;
        document.head.appendChild(link);
      });
    });
  }
}

// Usage in app initialization
prefetchRoutes([
  '/chunks/chat.js',
  '/chunks/settings.js',
  '/chunks/analytics.js',
]);
```

## Removing Unused Dependencies

### Dependency Audit

```typescript
// scripts/analyze-dependencies.ts
import { execSync } from 'child_process';
import fs from 'fs';

interface DependencyUsage {
  name: string;
  version: string;
  size: number;
  used: boolean;
  files: string[];
}

async function analyzeDependencies() {
  const packageJson = JSON.parse(
    fs.readFileSync('package.json', 'utf-8')
  );

  const dependencies = {
    ...packageJson.dependencies,
    ...packageJson.devDependencies,
  };

  const usage: DependencyUsage[] = [];

  for (const [name] of Object.entries(dependencies)) {
    // Check if imported anywhere
    const searchResult = execSync(
      `grep -r "from '${name}'" src/`,
      { encoding: 'utf-8', stdio: 'pipe' }
    ).toString();

    const used = searchResult.trim().length > 0;
    const files = used ? searchResult.split('\n').map(l => l.split(':')[0]) : [];

    // Get package size
    const size = parseInt(
      execSync(`du -sk node_modules/${name}`, { encoding: 'utf-8' })
        .split('\t')[0]
    );

    usage.push({
      name,
      version: dependencies[name],
      size,
      used,
      files: [...new Set(files)],
    });
  }

  // Report unused
  const unused = usage.filter(d => !d.used);
  const unusedSize = unused.reduce((sum, d) => sum + d.size, 0);

  console.log('\nUnused Dependencies:');
  console.log('━'.repeat(50));
  unused.forEach(dep => {
    console.log(`${dep.name}@${dep.version} (${dep.size}KB)`);
  });
  console.log(`\nTotal unused size: ${unusedSize}KB`);

  return usage;
}

analyzeDependencies();
```

### Bundle Size Tracking

```typescript
// scripts/check-bundle-size.ts
import fs from 'fs';
import path from 'path';
import { gzipSync } from 'zlib';

interface BundleSize {
  file: string;
  raw: number;
  gzip: number;
  budget: number;
  status: 'pass' | 'warn' | 'fail';
}

const budgets = {
  'index.js': 200 * 1024,        // 200KB
  'vendor.js': 150 * 1024,       // 150KB
  'chat.chunk.js': 100 * 1024,   // 100KB
  'audio.chunk.js': 50 * 1024,   // 50KB
};

function checkBundleSizes(distDir: string): BundleSize[] {
  const results: BundleSize[] = [];

  Object.entries(budgets).forEach(([filename, budget]) => {
    const filepath = path.join(distDir, filename);

    if (!fs.existsSync(filepath)) {
      console.warn(`File not found: ${filename}`);
      return;
    }

    const content = fs.readFileSync(filepath);
    const raw = content.length;
    const gzip = gzipSync(content).length;

    let status: 'pass' | 'warn' | 'fail' = 'pass';
    if (gzip > budget) {
      status = 'fail';
    } else if (gzip > budget * 0.9) {
      status = 'warn';
    }

    results.push({
      file: filename,
      raw,
      gzip,
      budget,
      status,
    });
  });

  return results;
}

function formatSize(bytes: number): string {
  return `${(bytes / 1024).toFixed(2)}KB`;
}

function printReport(results: BundleSize[]) {
  console.log('\nBundle Size Report');
  console.log('━'.repeat(70));
  console.log('File                    Raw        Gzip      Budget    Status');
  console.log('━'.repeat(70));

  results.forEach(r => {
    const status = r.status === 'pass' ? '✓' :
                   r.status === 'warn' ? '⚠' : '✗';
    const percentage = ((r.gzip / r.budget) * 100).toFixed(1);

    console.log(
      `${r.file.padEnd(20)} ${formatSize(r.raw).padStart(10)} ` +
      `${formatSize(r.gzip).padStart(10)} ` +
      `${formatSize(r.budget).padStart(10)} ` +
      `${status} ${percentage}%`
    );
  });

  const failed = results.filter(r => r.status === 'fail');
  if (failed.length > 0) {
    console.log('\n❌ Bundle size check failed!');
    process.exit(1);
  } else {
    console.log('\n✅ All bundles within budget');
  }
}

const results = checkBundleSizes('dist');
printReport(results);
```

2. **Set realistic budgets**
   - Based on target users and networks
   - Include buffer for growth
   - Review quarterly

3. **Lazy load aggressively**
   - Route-based splitting
   - Component-based splitting
   - Feature-based splitting

4. **Use modern formats**
   - ES modules for tree shaking
   - Dynamic imports for code splitting
   - Compression (gzip/brotli)

5. **Monitor continuously**
   - CI bundle size checks
   - Performance monitoring
   - User metrics tracking

### DON'T

1. **Don't bundle everything**
   - Avoid importing entire libraries
   - Don't include unused features
   - Don't inline large assets

2. **Don't skip analysis**
   - Don't guess at optimization impact
   - Don't optimize without measuring
   - Don't ignore CI warnings

3. **Don't sacrifice UX**
   - Don't over-split (loading states)
   - Don't break critical paths
   - Don't lazy load above fold content

4. **Don't ignore dependencies**
   - Don't add heavy deps without review
   - Don't keep unused dependencies
   - Don't use full builds when subsets exist
