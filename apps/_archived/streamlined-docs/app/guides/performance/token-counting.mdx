# Token Counting Performance Optimization

Token counting is a critical performance consideration in AI applications. This guide covers advanced optimization techniques to ensure token counting doesn't block your UI or degrade user experience.

## Overview

Token counting can be computationally expensive, especially for:
- Large documents (>100KB)
- Real-time counting as users type
- Multiple simultaneous counting operations
- Complex tokenization algorithms (GPT-4, Claude)

### Performance Impact

## 1. Worker Threads for Background Counting

Offload token counting to Web Workers to prevent UI blocking.

### Worker Implementation

```typescript
// utils/tokenization/token-counter.worker.ts
import { encode } from 'gpt-tokenizer';

interface CountRequest {
  id: string;
  text: string;
  model: string;
  options?: {
    cache?: boolean;
    stream?: boolean;
  };
}

interface CountResponse {
  id: string;
  count: number;
  cached: boolean;
  duration: number;
}

// Simple in-memory cache
const tokenCache = new Map<string, number>();

// Generate cache key
function getCacheKey(text: string, model: string): string {
  return `${model}:${text.length}:${text.slice(0, 100)}`;
}

// Handle token counting requests
self.onmessage = async (event: MessageEvent<CountRequest>) => {
  const { id, text, model, options } = event.data;
  const startTime = performance.now();

  // Check cache
  if (options?.cache) {
    const cacheKey = getCacheKey(text, model);
    const cached = tokenCache.get(cacheKey);

    if (cached !== undefined) {
      self.postMessage({
        id,
        count: cached,
        cached: true,
        duration: performance.now() - startTime
      } as CountResponse);
      return;
    }
  }

  try {
    // Count tokens based on model
    let count: number;

    switch (model) {
      case 'gpt-4':
      case 'gpt-3.5-turbo':
        count = encode(text).length;
        break;
      case 'claude-3':
        // Use Claude tokenizer
        count = await countClaudeTokens(text);
        break;
      default:
        // Fallback approximation
        count = Math.ceil(text.length / 4);
    }

    // Cache result
    if (options?.cache) {
      const cacheKey = getCacheKey(text, model);
      tokenCache.set(cacheKey, count);

      // Limit cache size
      if (tokenCache.size > 1000) {
        const firstKey = tokenCache.keys().next().value;
        tokenCache.delete(firstKey);
      }
    }

    self.postMessage({
      id,
      count,
      cached: false,
      duration: performance.now() - startTime
    } as CountResponse);
  } catch (error) {
    self.postMessage({
      id,
      error: error instanceof Error ? error.message : 'Unknown error',
      duration: performance.now() - startTime
    });
  }
};

// Claude token counting (placeholder)
async function countClaudeTokens(text: string): Promise<number> {
  // Implement Claude-specific tokenization
  // This is a simplified approximation
  return Math.ceil(text.length / 3.5);
}
```

### Worker Manager

```typescript
// hooks/useTokenWorker.ts
import { useEffect, useRef, useCallback, useState } from 'react';

interface TokenCountResult {
  count: number;
  cached: boolean;
  duration: number;
}

interface UseTokenWorkerOptions {
  model?: string;
  cache?: boolean;
  maxWorkers?: number;
}

export function useTokenWorker(options: UseTokenWorkerOptions = {}) {
  const {
    model = 'gpt-4',
    cache = true,
    maxWorkers = 2
  } = options;

  const workerPoolRef = useRef<Worker[]>([]);
  const nextWorkerRef = useRef(0);
  const pendingRequestsRef = useRef(new Map<string, {
    resolve: (result: TokenCountResult) => void;
    reject: (error: Error) => void;
  }>());

  const [isReady, setIsReady] = useState(false);

  // Initialize worker pool
  useEffect(() => {
    const workers: Worker[] = [];

    for (let i = 0; i < maxWorkers; i++) {
      const worker = new Worker(
        new URL('../utils/tokenization/token-counter.worker.ts', import.meta.url),
        { type: 'module' }
      );

      worker.onmessage = (event) => {
        const { id, count, cached, duration, error } = event.data;
        const pending = pendingRequestsRef.current.get(id);

        if (pending) {
          if (error) {
            pending.reject(new Error(error));
          } else {
            pending.resolve({ count, cached, duration });
          }
          pendingRequestsRef.current.delete(id);
        }
      };

      worker.onerror = (error) => {
        console.error('Token worker error:', error);
      };

      workers.push(worker);
    }

    workerPoolRef.current = workers;
    setIsReady(true);

    return () => {
      workers.forEach(w => w.terminate());
      workerPoolRef.current = [];
      setIsReady(false);
    };
  }, [maxWorkers]);

  // Count tokens using worker pool
  const countTokens = useCallback(async (
    text: string
  ): Promise<TokenCountResult> => {
    if (!isReady || workerPoolRef.current.length === 0) {
      throw new Error('Token workers not initialized');
    }

    const id = `${Date.now()}-${Math.random()}`;

    return new Promise((resolve, reject) => {
      pendingRequestsRef.current.set(id, { resolve, reject });

      // Round-robin worker selection
      const worker = workerPoolRef.current[nextWorkerRef.current];
      nextWorkerRef.current = (nextWorkerRef.current + 1) % workerPoolRef.current.length;

      worker.postMessage({
        id,
        text,
        model,
        options: { cache }
      });

      // Timeout after 10 seconds
      setTimeout(() => {
        const pending = pendingRequestsRef.current.get(id);
        if (pending) {
          pending.reject(new Error('Token counting timeout'));
          pendingRequestsRef.current.delete(id);
        }
      }, 10000);
    });
  }, [isReady, model, cache]);

  return { countTokens, isReady };
}
```

## 2. Incremental Counting

Count tokens incrementally as users type to provide real-time feedback without performance degradation.

### Debounced Incremental Counter

```typescript
// hooks/useIncrementalTokenCount.ts
import { useState, useEffect, useRef } from 'react';
import { useTokenWorker } from './useTokenWorker';

interface UseIncrementalTokenCountOptions {
  text: string;
  model?: string;
  debounceMs?: number;
  minLength?: number;
  onCountChange?: (count: number) => void;
}

export function useIncrementalTokenCount({
  text,
  model = 'gpt-4',
  debounceMs = 300,
  minLength = 10,
  onCountChange
}: UseIncrementalTokenCountOptions) {
  const [tokenCount, setTokenCount] = useState(0);
  const [isCountingTokens, setIsCountingTokens] = useState(false);
  const [lastCountDuration, setLastCountDuration] = useState(0);
  const [cacheHitRate, setCacheHitRate] = useState(0);

  const { countTokens, isReady } = useTokenWorker({ model, cache: true });
  const timeoutRef = useRef<NodeJS.Timeout>();
  const statsRef = useRef({ total: 0, cached: 0 });

  useEffect(() => {
    if (!isReady || text.length < minLength) {
      setTokenCount(0);
      return;
    }

    // Clear previous timeout
    if (timeoutRef.current) {
      clearTimeout(timeoutRef.current);
    }

    setIsCountingTokens(true);

    // Debounce token counting
    timeoutRef.current = setTimeout(async () => {
      try {
        const result = await countTokens(text);
        setTokenCount(result.count);
        setLastCountDuration(result.duration);

        // Update cache hit rate
        statsRef.current.total++;
        if (result.cached) {
          statsRef.current.cached++;
        }
        setCacheHitRate(statsRef.current.cached / statsRef.current.total);

        onCountChange?.(result.count);
      } catch (error) {
        console.error('Token counting error:', error);
      } finally {
        setIsCountingTokens(false);
      }
    }, debounceMs);

    return () => {
      if (timeoutRef.current) {
        clearTimeout(timeoutRef.current);
      }
    };
  }, [text, model, debounceMs, minLength, isReady, countTokens, onCountChange]);

  return {
    tokenCount,
    isCountingTokens,
    lastCountDuration,
    cacheHitRate
  };
}
```

## 3. Advanced Caching Strategies

Implement sophisticated caching to minimize redundant tokenization.

### Multi-Level Cache

```typescript
// utils/tokenization/token-cache.ts
interface CacheEntry {
  count: number;
  timestamp: number;
  hits: number;
  size: number;
}

export class TokenCache {
  private memoryCache = new Map<string, CacheEntry>();
  private maxMemorySize = 50 * 1024 * 1024; // 50MB
  private currentSize = 0;
  private maxAge = 3600000; // 1 hour

  // Generate cache key with content hash
  private getCacheKey(text: string, model: string): string {
    // Use first and last 100 chars + length for cache key
    const prefix = text.slice(0, 100);
    const suffix = text.slice(-100);
    return `${model}:${text.length}:${this.simpleHash(prefix + suffix)}`;
  }

  // Simple hash function
  private simpleHash(str: string): string {
    let hash = 0;
    for (let i = 0; i < str.length; i++) {
      const char = str.charCodeAt(i);
      hash = ((hash << 5) - hash) + char;
      hash = hash & hash;
    }
    return hash.toString(36);
  }

  // Get from cache
  get(text: string, model: string): number | null {
    const key = this.getCacheKey(text, model);
    const entry = this.memoryCache.get(key);

    if (!entry) return null;

    // Check if expired
    if (Date.now() - entry.timestamp > this.maxAge) {
      this.delete(key);
      return null;
    }

    // Update hit count
    entry.hits++;

    return entry.count;
  }

  // Set in cache
  set(text: string, model: string, count: number): void {
    const key = this.getCacheKey(text, model);
    const size = text.length * 2; // Approximate size in bytes

    // Evict if necessary
    while (this.currentSize + size > this.maxMemorySize && this.memoryCache.size > 0) {
      this.evictLeastUsed();
    }

    this.memoryCache.set(key, {
      count,
      timestamp: Date.now(),
      hits: 0,
      size
    });

    this.currentSize += size;
  }

  // Delete from cache
  private delete(key: string): void {
    const entry = this.memoryCache.get(key);
    if (entry) {
      this.currentSize -= entry.size;
      this.memoryCache.delete(key);
    }
  }

  // Evict least recently used entry
  private evictLeastUsed(): void {
    let leastUsedKey: string | null = null;
    let leastUsedHits = Infinity;
    let oldestTime = Infinity;

    for (const [key, entry] of this.memoryCache.entries()) {
      if (entry.hits < leastUsedHits ||
          (entry.hits === leastUsedHits && entry.timestamp < oldestTime)) {
        leastUsedHits = entry.hits;
        oldestTime = entry.timestamp;
        leastUsedKey = key;
      }
    }

    if (leastUsedKey) {
      this.delete(leastUsedKey);
    }
  }

  // Clear cache
  clear(): void {
    this.memoryCache.clear();
    this.currentSize = 0;
  }

  // Get cache stats
  getStats() {
    return {
      entries: this.memoryCache.size,
      sizeBytes: this.currentSize,
      sizeMB: (this.currentSize / 1024 / 1024).toFixed(2),
      hitRate: this.calculateHitRate()
    };
  }

  private calculateHitRate(): number {
    let totalHits = 0;
    let totalEntries = 0;

    for (const entry of this.memoryCache.values()) {
      totalHits += entry.hits;
      totalEntries++;
    }

    return totalEntries > 0 ? totalHits / totalEntries : 0;
  }
}

// Singleton instance
export const tokenCache = new TokenCache();
```

## 4. Streaming Tokenization

Process large texts in chunks for better perceived performance.

### Streaming Counter

```typescript
// utils/tokenization/streaming-counter.ts
interface StreamingOptions {
  chunkSize?: number;
  onProgress?: (progress: number, partialCount: number) => void;
  signal?: AbortSignal;
}

export async function countTokensStreaming(
  text: string,
  model: string,
  options: StreamingOptions = {}
): Promise<number> {
  const {
    chunkSize = 10000,
    onProgress,
    signal
  } = options;

  let totalCount = 0;
  const chunks: string[] = [];

  // Split into chunks
  for (let i = 0; i < text.length; i += chunkSize) {
    chunks.push(text.slice(i, i + chunkSize));
  }

  // Process chunks
  for (let i = 0; i < chunks.length; i++) {
    // Check for abort
    if (signal?.aborted) {
      throw new Error('Token counting aborted');
    }

    const chunk = chunks[i];
    const chunkCount = await countTokensForChunk(chunk, model);
    totalCount += chunkCount;

    // Report progress
    if (onProgress) {
      const progress = ((i + 1) / chunks.length) * 100;
      onProgress(progress, totalCount);
    }

    // Yield to main thread
    await new Promise(resolve => setTimeout(resolve, 0));
  }

  return totalCount;
}

async function countTokensForChunk(chunk: string, model: string): Promise<number> {
  // Implement chunk counting logic
  // This would use your tokenizer of choice
  return Math.ceil(chunk.length / 4);
}
```

## 5. Lazy Loading Tokenizer Dependencies

Load tokenizer libraries on-demand to reduce initial bundle size.

### Dynamic Tokenizer Loading

```typescript
// utils/tokenization/lazy-tokenizer.ts
type TokenizerModule = {
  encode: (text: string) => number[];
  decode: (tokens: number[]) => string;
};

class LazyTokenizer {
  private tokenizers = new Map<string, TokenizerModule>();
  private loading = new Map<string, Promise<TokenizerModule>>();

  async getTokenizer(model: string): Promise<TokenizerModule> {
    // Return cached tokenizer
    if (this.tokenizers.has(model)) {
      return this.tokenizers.get(model)!;
    }

    // Return in-progress load
    if (this.loading.has(model)) {
      return this.loading.get(model)!;
    }

    // Load tokenizer dynamically
    const loadPromise = this.loadTokenizer(model);
    this.loading.set(model, loadPromise);

    try {
      const tokenizer = await loadPromise;
      this.tokenizers.set(model, tokenizer);
      return tokenizer;
    } finally {
      this.loading.delete(model);
    }
  }

  private async loadTokenizer(model: string): Promise<TokenizerModule> {
    switch (model) {
      case 'gpt-4':
      case 'gpt-3.5-turbo':
        const gptTokenizer = await import('gpt-tokenizer');
        return {
          encode: gptTokenizer.encode,
          decode: gptTokenizer.decode
        };

      case 'claude-3':
        const claudeTokenizer = await import('@anthropic-ai/tokenizer');
        return {
          encode: (text) => claudeTokenizer.encode(text),
          decode: (tokens) => claudeTokenizer.decode(tokens)
        };

      default:
        throw new Error(`Unsupported model: ${model}`);
    }
  }

  async countTokens(text: string, model: string): Promise<number> {
    const tokenizer = await this.getTokenizer(model);
    return tokenizer.encode(text).length;
  }
}

export const lazyTokenizer = new LazyTokenizer();
```

```typescript
// utils/tokenization/__tests__/tokenizer-benchmark.test.ts
import { describe, it, expect } from 'vitest';
import { encode as gptEncode } from 'gpt-tokenizer';
import { tokenCache } from '../token-cache';

interface BenchmarkResult {
  name: string;
  avgTime: number;
  minTime: number;
  maxTime: number;
  tokensPerSecond: number;
  cacheHitRate?: number;
}

describe('Tokenizer Performance Benchmarks', () => {
  const testTexts = {
    small: 'Hello, world!'.repeat(10), // ~130 chars
    medium: 'Lorem ipsum dolor sit amet. '.repeat(100), // ~2.8KB
    large: 'The quick brown fox jumps over the lazy dog. '.repeat(1000), // ~45KB
    xlarge: 'A'.repeat(100000) // 100KB
  };

  async function benchmarkTokenizer(
    name: string,
    fn: (text: string) => Promise<number> | number,
    text: string,
    iterations = 100
  ): Promise<BenchmarkResult> {
    const times: number[] = [];

    // Warmup
    for (let i = 0; i < 5; i++) {
      await fn(text);
    }

    // Measure
    for (let i = 0; i < iterations; i++) {
      const start = performance.now();
      await fn(text);
      times.push(performance.now() - start);
    }

    const avgTime = times.reduce((a, b) => a + b, 0) / times.length;
    const minTime = Math.min(...times);
    const maxTime = Math.max(...times);
    const tokensPerSecond = (text.length * iterations) / (times.reduce((a, b) => a + b, 0) / 1000);

    return {
      name,
      avgTime,
      minTime,
      maxTime,
      tokensPerSecond
    };
  }

  it('should benchmark GPT tokenizer (main thread)', async () => {
    const results: BenchmarkResult[] = [];

    for (const [size, text] of Object.entries(testTexts)) {
      const result = await benchmarkTokenizer(
        `GPT (${size})`,
        (t) => gptEncode(t).length,
        text,
        size === 'xlarge' ? 10 : 100
      );
      results.push(result);
    }

    console.table(results);
    expect(results.every(r => r.avgTime >= 0)).toBe(true);
  });

    // Worker implementation benchmark
    const results: BenchmarkResult[] = [];

      return new Promise<number>((resolve) => {
        setTimeout(() => {

    let cacheHits = 0;
    let totalRequests = 0;

    const cachedTokenizer = (text: string): number => {
      totalRequests++;
      const cached = tokenCache.get(text, 'gpt-4');
      if (cached !== null) {
        cacheHits++;
        return cached;

      const count = gptEncode(text).length;
      tokenCache.set(text, 'gpt-4', count);

        ...approxResult,
```

## Best Practices

### 1. Choose the Right Strategy

```typescript
function selectTokenCountingStrategy(textLength: number, context: 'typing' | 'batch' | 'display') {
  // Small texts: main thread is fine
  if (textLength < 1000) {
    return 'mainThread';
  }

  // Real-time typing: debounced + cached
  if (context === 'typing') {
    return 'incrementalCached';
  }

  // Large batch processing: worker pool
  if (textLength > 50000 && context === 'batch') {
    return 'workerPool';
  }

  // Very large texts: streaming
  if (textLength > 100000) {
    return 'streaming';
  }

  // Default: worker + cache
  return 'workerCached';
}
```

### 2. Implement Progressive Enhancement

```tsx
function SmartTokenCounter({ text, model }: { text: string; model: string }) {
  // Start with fast approximation
  const approximateCount = Math.ceil(text.length / 4);
  const [accurateCount, setAccurateCount] = useState<number | null>(null);

  useEffect(() => {
    // Calculate accurate count in background
    const controller = new AbortController();

    countTokensAccurate(text, model, controller.signal)
      .then(setAccurateCount)
      .catch(() => {
        // Fall back to approximation
      });

    return () => controller.abort();
  }, [text, model]);

  return (
    <div>
      {accurateCount ?? approximateCount} tokens
      {accurateCount === null && (
        <span className="text-xs text-gray-400"> (approximate)</span>
      )}
    </div>
  );
}
```

### 3. Monitor Performance

```typescript
// utils/tokenization/performance-monitor.ts
export class TokenizationMonitor {
  private metrics: Array<{
    timestamp: number;
    duration: number;
    textLength: number;
    cached: boolean;
  }> = [];

  record(duration: number, textLength: number, cached: boolean) {
    this.metrics.push({
      timestamp: Date.now(),
      duration,
      textLength,
      cached
    });

    // Keep last 1000 entries
    if (this.metrics.length > 1000) {
      this.metrics.shift();
    }
  }

  getStats() {
    const recentMetrics = this.metrics.slice(-100);

    return {
      avgDuration: recentMetrics.reduce((sum, m) => sum + m.duration, 0) / recentMetrics.length,
      cacheHitRate: recentMetrics.filter(m => m.cached).length / recentMetrics.length,
      p95Duration: this.percentile(recentMetrics.map(m => m.duration), 0.95),
      throughput: recentMetrics.reduce((sum, m) => sum + m.textLength, 0) /
                  (recentMetrics.reduce((sum, m) => sum + m.duration, 0) / 1000)
    };
  }

  private percentile(values: number[], p: number): number {
    const sorted = [...values].sort((a, b) => a - b);
    const index = Math.ceil(sorted.length * p) - 1;
    return sorted[index];
  }
}
```
