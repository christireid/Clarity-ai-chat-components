  // app/page.tsx (Next.js) or src/App.tsx (Vite/CRA)

      </div>

  Create an API endpoint that streams responses. Here's an example for Next.js App Router:

  ```typescript
  // app/api/chat/route.ts
  import { OpenAI } from 'openai'

  const openai = new OpenAI({
    apiKey: process.env.OPENAI_API_KEY,
  })

  export async function POST(req: Request) {
    const { messages } = await req.json()

    const response = await openai.chat.completions.create({
      model: 'gpt-4',
      messages,
      stream: true,
    })

    // Convert OpenAI stream to SSE format
    const encoder = new TextEncoder()
    const stream = new ReadableStream({
      async start(controller) {
        for await (const chunk of response) {
          const content = chunk.choices[0]?.delta?.content || ''
          if (content) {
            controller.enqueue(
              encoder.encode(`data: ${JSON.stringify({ content })}\n\n`)
            )
          }
        }
        controller.enqueue(encoder.encode('data: [DONE]\n\n'))
        controller.close()
      },
    })

    return new Response(stream, {
      headers: {
        'Content-Type': 'text/event-stream',
        'Cache-Control': 'no-cache',
        Connection: 'keep-alive',
      },
    })
  }
  ```

  Enable conversation memory so the AI remembers context:

  ```tsx
  import { ClarityChat, MemoryProvider } from '@clarity-chat/react'

  export default function ChatPage() {
    return (
      <MemoryProvider>
        <div className="h-screen">
          <ClarityChat
            api="/api/chat"
            memoryStrategy="sliding-window"
            showHeader
            sessionTitle="AI Assistant"
            sessionSubtitle="Remembers our conversation"
          />
        </div>
      </MemoryProvider>
    )
  }
  ```

  Memory strategies:
  - `sliding-window`: Keeps the last N messages (best for most cases)
  - `semantic-chunks`: Groups related messages together
  - `vector-store`: Uses embeddings for long-term memory

    <h3 className="step-title">Handle Errors Gracefully</h3>
    <span className="step-time">~3 minutes</span>

  Add error handling to provide a good user experience:

  ```tsx
  import { ClarityChat, ErrorBoundary } from '@clarity-chat/react'
  import { toast } from '@clarity-chat/primitives'

  export default function ChatPage() {
    const handleError = (error: Error) => {
      console.error('Chat error:', error)

      // Show user-friendly toast notification
      toast({
        title: 'Something went wrong',
        description: error.message,
        variant: 'destructive',
      })
    }

    return (
      <ErrorBoundary
        fallback={
          <div className="flex items-center justify-center h-screen">
            <div className="text-center">
              <h2>Something went wrong</h2>
              <button onClick={() => window.location.reload()}>
                Try again
              </button>
            </div>
          </div>
        }
      >
        <ClarityChat
          api="/api/chat"
          onError={handleError}
          showNetworkStatus
        />
      </ErrorBoundary>
    )
  }
  ```

  The `showNetworkStatus` prop displays a connection indicator that helps users
  understand when they're offline.

  For complete control, use the `useClarityChat` hook:

  ```tsx
  import { useClarityChat, convertCoreMessagesToMessages } from '@clarity-chat/react'
  import { ChatWindow, Button } from '@clarity-chat/primitives'
  import { useMemo } from 'react'

  export default function CustomChatPage() {
    const {
      messages,
      input,
      handleInputChange,
      handleSubmit,
      isLoading,
      error,
      reload,
      stop,
      memoryInfo,
    } = useClarityChat({
      api: '/api/chat',
      memory: {
        enabled: true,
        strategy: 'sliding-window',
        maxTokens: 4000,
      },
      onFinish: (message) => {
        console.log('Response complete:', message)
      },
    })

    const formattedMessages = useMemo(
      () => convertCoreMessagesToMessages(messages),
      [messages]
    )

    return (
      <div className="h-screen flex flex-col">
        {/* Custom header */}
        <header className="p-4 border-b flex justify-between items-center">
          <h1 className="font-bold">My Custom Chat</h1>
          <div className="flex gap-2">
            {isLoading && (
              <Button variant="outline" onClick={stop}>
                Stop generating
              </Button>
            )}
            <Button variant="ghost" onClick={reload}>
              Regenerate
            </Button>
          </div>
        </header>

        {/* Chat window */}
        <div className="flex-1 overflow-hidden">
          <ChatWindow
            messages={formattedMessages}
            isLoading={isLoading}
            onSendMessage={async (content) => {
              await handleSubmit(undefined, { data: { content } })
            }}
          />
        </div>

        {/* Memory indicator */}
        {memoryInfo.enabled && (
          <div className="p-2 text-xs text-muted-foreground border-t">
            Memory: {memoryInfo.strategy} | Messages: {messages.length}
          </div>
        )}
      </div>
    )
  }
  ```

## Complete Example

Here's the full code for a production-ready chat application:

```tsx
// app/chat/page.tsx
'use client'

import {
  ClarityChat,
  MemoryProvider,
  ThemeProvider,
  ErrorBoundary,
  ToastProvider,
} from '@clarity-chat/react'
import { toast } from '@clarity-chat/primitives'

export default function ChatPage() {
  return (
    <ThemeProvider
      defaultTheme={{
        mode: 'system',
        preset: 'default',
        enableTransitions: true,
      }}
    >
      <ToastProvider>
        <MemoryProvider>
          <ErrorBoundary
            fallback={
              <div className="flex items-center justify-center h-screen">
                <div className="text-center p-8">
                  <h2 className="text-2xl font-bold mb-4">
                    Oops! Something went wrong
                  </h2>
                  <button
                    className="px-4 py-2 bg-primary text-primary-foreground rounded"
                    onClick={() => window.location.reload()}
                  >
                    Reload page
                  </button>
                </div>
              </div>
            }
          >
            <main className="h-screen bg-background">
              <ClarityChat
                api="/api/chat"
                memoryStrategy="sliding-window"
                showHeader
                sessionTitle="AI Assistant"
                sessionSubtitle="Powered by Clarity Chat"
                showMessageCount
                showTokenCounter
                showNetworkStatus
                onError={(error) => {
                  toast({
                    title: 'Error',
                    description: error.message,
                    variant: 'destructive',
                  })
                }}
                onExport={() => {
                  toast({
                    title: 'Exported',
                    description: 'Chat exported to clipboard',
                  })
                }}
              />
            </main>
          </ErrorBoundary>
        </MemoryProvider>
      </ToastProvider>
    </ThemeProvider>
  )
}
```
